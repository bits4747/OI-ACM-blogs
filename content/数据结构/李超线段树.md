#数据结构 #线段树

[详解](https://oiwiki.com/ds/li-chao-tree/)
## 简介
李超树支持：
1. 在平面上加入一条线段/函数。
2. 给定一个数 $k$，询问与直线 $x = k$ 相交的线段/函数图像中，交点纵坐标最值（线段/函数编号）。
### 时间复杂度
全局插入时间复杂度为 $O(\log n)$ ，区间插入为  $O(\log^2 n)$，查询为 $O(\log n)$。

查询时间复杂度显然。

全局插入（即插入一条直线/定义域不限的函数）时只需下传至左半区间或者右半区间。

而区间插入过程，需要将原线段拆分最多到 $O(\log n)$ 个线段树区间中，对于每个区间又需要花费 $O(\log n)$ 的时间递归下传，从而插入过程的时间复杂度为 $O(\log^2 n)$。
### 空间复杂度
李超树与一般的线段树不同，信息可以在非叶子结点保存，因此动态开点的李超树空间复杂度可以做到 $O(n)$。

具体方法是每次插入直线时遇到空结点就 return，这样每次只会新建一个结点。
### 模板
以下为全局插入动态开点的精简写法。
```cpp
#define F(i,o) -d[i]*o+f[i]
void upd(int &k,int l,int r){
	if(!k){s[k=++tot]=x;return;}
	int mi=(l+r)>>1;
	if(F(s[k],mi)>F(x,mi)) swap(s[k],x);
	if(l<r) F(x,l)<F(s[k],l)?upd(ls[k],l,mi):upd(rs[k],mi+1,r);
}
ll ask(int k,int l,int r){
	if(!k) return 1e18;
	int mi=(l+r)>>1;
	return min(F(s[k],x),x<=mi?ask(ls[k],l,mi):ask(rs[k],mi+1,r));
}
```

```cpp
#include<bits/stdc++.h>
using namespace std;
#define ls k<<1
#define rs k<<1|1 
const int N=1e5+5,X=39989,Y=1e9;
int n,las,cnt,ty,x,y,b0,b1,ll[N],rr[N],tr[X<<2];
double kk[N],b[N];
double f(int id,int xx){return (ll[id]<=xx&&xx<=rr[id])?kk[id]*xx+b[id]:-1.0;}
void add(int k,int l,int r,int i){
	int mi=(l+r)>>1;
	if(x<=l&&r<=y){
		if(f(tr[k],mi)<f(i,mi)) swap(tr[k],i);
		if(l<r) f(i,l)>f(tr[k],l)?add(ls,l,mi,i):add(rs,mi+1,r,i);
		return;
	}
	if(x<=mi) add(ls,l,mi,i);
	if(y>mi) add(rs,mi+1,r,i);
}
int ask(int k,int l,int r){
	if(l==r) return tr[k];
	int mi=(l+r)>>1,id=0;
	if(x<=mi) id=ask(ls,l,mi);
	else id=ask(rs,mi+1,r);
	return (fabs(f(id,x)-f(tr[k],x))<=1e-15)?
		   (id<tr[k]?id:tr[k]):(f(id,x)>f(tr[k],x)?id:tr[k]);
}
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i){
		scanf("%d",&ty);
		if(ty==0) scanf("%d",&x),x=(x+las-1)%X+1,printf("%d\n",las=ask(1,1,X));
		else{
			scanf("%d%d%d%d",&x,&b0,&y,&b1);
			ll[++cnt]=x=(x+las-1)%X+1,rr[cnt]=y=(y+las-1)%X+1,b0=(b0+las-1)%Y+1,b1=(b1+las-1)%Y+1;
			if(x>y) swap(x,y),swap(ll[cnt],rr[cnt]),swap(b0,b1);
			if(x==y) b[cnt]=max(b0,b1);
			else kk[cnt]=1.0*(b1-b0)/(1.0*(y-x)),b[cnt]=b0*1.0-kk[cnt]*x;
			add(1,1,X,cnt);
		}
	}
	return 0;
}
```
#### [P4254 [JSOI2008]Blue Mary开公司 ](https://www.luogu.com.cn/problem/P4254)
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=1e5+5,M=5e4;
char ty[10];
double b[N],kk[N];
int n,t,ans,tr[M<<2];
double get(int id,int xx){return kk[id]*xx+b[id];}
int ls(int k){return k<<1;}
int rs(int k){return k<<1|1;}
void change(int k,int l,int r,int i){//全局插入
	if(get(i,l)>get(tr[k],l)&&get(i,r)>get(tr[k],r)){tr[k]=i;return ;}
	if(get(i,l)<=get(tr[k],l)&&get(i,r)<=get(tr[k],r)) return ;
	int mi=(l+r)>>1;
	if(kk[i]>kk[tr[k]]){
		if(get(i,mi)>get(tr[k],mi)) change(ls(k),l,mi,tr[k]),tr[k]=i;
		else change(rs(k),mi+1,r,i);
	}
	else{
		if(get(i,mi)>get(tr[k],mi)) change(rs(k),mi+1,r,tr[k]),tr[k]=i;
		else change(ls(k),l,mi,i);
	}
}
double ask(int k,int l,int r){
	if(l==r) return get(tr[k],t);
	int mi=(l+r)>>1;
	if(t<=mi) return max(get(tr[k],t),ask(ls(k),l,mi));
	else return max(get(tr[k],t),ask(rs(k),mi+1,r));
}
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i){
		scanf(" %s",ty);
		if(ty[0]=='P') scanf("%lf%lf",&b[i],&kk[i]),b[i]-=kk[i],change(1,1,M,i);
		else scanf("%d",&t),printf("%d\n",int(ask(1,1,M)/100.0));
	}
	return 0;
}
```

#### [P2305 [NOI2014] 购票](https://www.luogu.com.cn/problem/P2305)
**线段树套李超树+出栈序**

![[Pasted image 20230913193322.png]]

**题解：**

$f_i​=min(f_j​+(d_i​−d_j​)*p_i​+q_i​)​=min(−d_j​*p_i​+f_j​)+d_i*p_i+q_i​$

$min(−d_j​*p_i​+f_j​)$ 明显可用李超线段树优化。

因为一个点只能走它到根的路，需要撤销操作。由于有 $l$ 的限制，需要线段树树套可撤销李超树。

这时，**出栈序**，也就是每个结点离开 dfs 的顺序，就派上用场了。

dfs 时在 $u$ 及其最远的合法祖先 $v$ 的出栈序对应区间上查询，区间内不在 $u$ 和 $v$ 的路径上的点都未被访问到，不会对答案产生贡献，因此不需要撤销操作。
```cpp
#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int N=2e5+5,M=N*20;
int n,T,fa,tim,ou[N],t,D[N],rt[N<<2],pos,x,tot,s[M],ls[M],rs[M];
ll Q[N],d[N],p[N],q[N],l[N],f[N];
vector<int>e[N];
#define F(i,o) -d[i]*o+f[i]
void upd(int &k,int l,int r){
	if(!k){s[k=++tot]=x;return;}
	int mi=(l+r)>>1;
	if(F(s[k],mi)>F(x,mi)) swap(s[k],x);
	if(l<r) F(x,l)<F(s[k],l)?upd(ls[k],l,mi):upd(rs[k],mi+1,r);
}
ll ask(int k,int l,int r){
	if(!k) return 1e18;
	int mi=(l+r)>>1;
	return min(F(s[k],x),x<=mi?ask(ls[k],l,mi):ask(rs[k],mi+1,r));
}
void upd(int k,int l,int r,int P){
	x=pos,upd(rt[k],0,1e6);
	if(l==r) return;
	int mi=(l+r)>>1;
	P<=mi?upd(k<<1,l,mi,P):upd(k<<1|1,mi+1,r,P);
}
ll ask(int k,int l,int r,int L,int R){
	if(L<=l&&r<=R) return ask(rt[k],0,1e6);
	int mi=(l+r)>>1;ll mn=1e18;
	if(L<=mi) mn=min(mn,ask(k<<1,l,mi,L,R));
	if(R>mi) mn=min(mn,ask(k<<1|1,mi+1,r,L,R));
	return mn;
}
void dfs0(int u){
	for(int v:e[u]) d[v]+=d[u],dfs0(v);
	ou[u]=++tim;
}
void dfs(int u){
	if(u!=1){
		int tmp=lower_bound(Q+1,Q+1+t,d[u]-l[u])-Q;
		tmp=(tmp==t+1?1:tmp);
		x=p[u],f[u]=ask(1,1,n,ou[u],ou[D[tmp]])+d[u]*p[u]+q[u];
	}
	pos=u,upd(1,1,n,ou[u]),Q[++t]=d[u],D[t]=u;
	for(int v:e[u]) dfs(v),--t;
}
int main(){
	scanf("%d%d",&n,&T);
	for(int i=2;i<=n;++i)
		scanf("%d%lld%lld%lld%lld",&fa,&d[i],&p[i],&q[i],&l[i]),e[fa].push_back(i);
	dfs0(1),dfs(1);
	for(int i=2;i<=n;++i) printf("%lld\n",f[i]);
	return 0;
}
```
### 将问题转化为函数最值
#### B. 22冲刺day13 最大子矩阵
![[Pasted image 20230926192252.png]]


容易想到，对于每个 $i$ ，求出 $[1,i-1]$ 和 $[i+1,n]$ 的最大子矩阵，即需处理前后缀最大子矩阵。

记录上一个比 $a[i]$ 小的数的位置为 $ll[i]$ ，后一个比 $a[i]$ 小的数的位置为 $rr[i]$ ，则 $(ll[i],rr[i])$ 这段区间子矩阵的值为简单的一次函数，可放入李超树求最值。

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=2e5+5;
int n,x,y,q[N],st,ll[N],rr[N],tr[N<<2];
long long a[N],b[N],lf[N],rf[N];
int ls(int k){return k<<1;}
int rs(int k){return k<<1|1;}
long long get(int id,int xx){return (ll[id]<=x&&x<=rr[id]?b[id]+a[id]*xx:-2e18);}
void add(int k,int l,int r,int i){
	int mi=(l+r)>>1;
	if(x<=l&&r<=y){
		if(get(i,l)>get(tr[k],l)&&get(i,r)>get(tr[k],r)){tr[k]=i;return ;}
		if(get(i,l)<=get(tr[k],l)&&get(i,r)<=get(tr[k],r)) return ;
		if(a[i]>a[tr[k]]){
			if(get(i,mi)>get(tr[k],mi)) add(ls(k),l,mi,tr[k]),tr[k]=i;
			else add(rs(k),mi+1,r,i);
		}
		else{
			if(get(i,mi)>get(tr[k],mi)) add(rs(k),mi+1,r,tr[k]),tr[k]=i;
			else add(ls(k),l,mi,i);
		}
		return ;
	}
	if(x<=mi) add(ls(k),l,mi,i);
	if(y>mi) add(rs(k),mi+1,r,i);
}
long long ask(int k,int l,int r){
	if(l==r) return get(tr[k],x);
	int mi=(l+r)>>1;
	if(x<=mi) return max(get(tr[k],x),ask(ls(k),l,mi));
	return max(get(tr[k],x),ask(rs(k),mi+1,r));
}
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i){
		scanf("%lld",&a[i]);
		while(st&&a[q[st]]>a[i]) --st;
		ll[i]=q[st]+1,q[++st]=i;
	}
	q[st=0]=n+1;
	for(int i=n;i;--i){
		while(st&&a[q[st]]>a[i]) --st;
		rr[i]=q[st]-1,q[++st]=i;
	}
	for(int i=1;i<=n;++i)
		b[i]=(a[i]-ll[i]*a[i]),x=ll[i],y=rr[i],add(1,1,n,i),x=i,lf[i]=max(lf[i-1],ask(1,1,n));
	memset(tr,0,sizeof(tr));
	for(int i=n;i>=1;--i)
		b[i]=(a[i]+rr[i]*a[i]),a[i]=-a[i],x=ll[i],y=rr[i],add(1,1,n,i),x=i,rf[i]=max(rf[i+1],ask(1,1,n));
	for(int i=1;i<=n;++i) printf("%lld\n",max(lf[i-1],rf[i+1]));
	return 0;
}
```
