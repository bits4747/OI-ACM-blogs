#数据结构 #线段树
## [P3919 【模板】可持久化线段树 1](https://www.luogu.com.cn/problem/P3919)
大佬的图，清晰明了（[转](https://www.luogu.com.cn/blog/juruohyfhaha/p3919-mu-ban-ke-chi-jiu-hua-shuo-zu-chu-bu-tan-jiu-zhu-xi-shu)）。
![[Pasted image 20230922120103.png]]

可以发现，只有橙颜色经过的结点才被修改过。因此，可持久化线段树（主席树）的基本思想就是只对进行修改的结点进行复制，再进一步修改。

![[Pasted image 20230922120119.png]]

动态开点以防MLE。

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=1e6+5;
int n,m,c[N],ti,ty,x,v,rt[N],tot;
struct tree{
	int ls,rs,a;
}s[N*40];
void build(int &k,int l,int r){
	s[++tot]=s[k],k=tot;
	if(l==r){s[k].a=c[l];return ;}
	int mi=(l+r)>>1;
	build(s[k].ls,l,mi),build(s[k].rs,mi+1,r);
}
void change(int &k,int l,int r){
	s[++tot]=s[k],k=tot;
	if(l==r){s[k].a=v;return ;}
	int mi=(l+r)>>1;
	if(x<=mi) change(s[k].ls,l,mi);
	else change(s[k].rs,mi+1,r);
}
int ask(int k,int l,int r){
	if(l==r) return s[k].a;
	int mi=(l+r)>>1;
	if(x<=mi) return ask(s[k].ls,l,mi);
	else return ask(s[k].rs,mi+1,r);
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<=n;++i) scanf("%d",&c[i]);
	build(rt[0],1,n);
	for(int i=1;i<=m;++i){
		scanf("%d%d%d",&ti,&ty,&x),rt[i]=rt[ti];
		if(ty==1) scanf("%d",&v),change(rt[i],1,n);
		else printf("%d\n",ask(rt[i],1,n));
	}
	return 0;
}
```

## [P3835 【模板】可持久化平衡树](https://www.luogu.com.cn/problem/P3835)
维护历史版本可持久化一下就可以了。每次操作复制先前版本再更新即可。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=5e5+5,inf=1e9+1;
int n,tot,ty,id,x,v,rt[N],tmp;
struct node{
	int ls,rs,cnt,ma,mi;
}s[N*32];
void change(int pre,int &k,int l,int r){
	s[++tot]=s[pre],k=tot;
	if(l==r){
		if(v==-1&&s[k].cnt==0) return ;
		s[k].cnt+=v,s[k].ma=(s[k].cnt?l:-inf),s[k].mi=(s[k].cnt?l:inf);return ;
	}
	int mid=(l+r)>>1;
	if(x<=mid) change(s[pre].ls,s[k].ls,l,mid);
	else change(s[pre].rs,s[k].rs,mid+1,r);
	s[k].cnt=s[s[k].ls].cnt+s[s[k].rs].cnt;
	s[k].ma=(s[s[k].rs].cnt?s[s[k].rs].ma:s[s[k].ls].ma);
	s[k].mi=(s[s[k].ls].cnt?s[s[k].ls].mi:s[s[k].rs].mi);
}
int ask(int k,int l,int r){
	int mid=(l+r)>>1;
	if(x==mid) return s[s[k].ls].cnt;
	if(x<mid) return ask(s[k].ls,l,mid);
	return ask(s[k].rs,mid+1,r)+s[s[k].ls].cnt;
}
int kth(int k,int l,int r){
	if(l==r) return l;
	int mid=(l+r)>>1;
	if(s[s[k].ls].cnt>=x) return kth(s[k].ls,l,mid);
	else{x-=s[s[k].ls].cnt;return kth(s[k].rs,mid+1,r);}
}
int MAX(int k,int l,int r){
	int mid=(l+r)>>1;
	if(x==mid) return s[s[k].ls].ma;
	if(x<mid) return MAX(s[k].ls,l,mid);
	return max(s[s[k].ls].ma,MAX(s[k].rs,mid+1,r));
}
int MIN(int k,int l,int r){
	int mid=(l+r)>>1;
	if(x==mid+1) return s[s[k].rs].mi;
	if(x>mid+1) return MIN(s[k].rs,mid+1,r);
	return min(s[s[k].rs].mi,MIN(s[k].ls,l,mid));
}
int main(){
//	freopen("data.in","r",stdin);
//	freopen("mine.out","w",stdout);
	scanf("%d",&n);
	s[0].ma=-inf,s[0].mi=inf;
	for(int i=1;i<=n;++i){
		scanf("%d%d%d",&id,&ty,&x);
		if(ty<=2){
			v=(ty==1?1:-1);
			change(rt[id],rt[i],-inf,inf);
			continue;
		} 
		rt[i]=rt[id];
		if(ty==3) --x,printf("%d\n",ask(rt[i],-inf,inf)+1);//查询 x 的排名，小于等于x-1的数+1 
		if(ty==4) printf("%d\n",kth(rt[i],-inf,inf));//查询排名为 x 的数
		if(ty==5) --x,printf("%d\n",((tmp=MAX(rt[i],-inf,inf))==-inf)?-2147483647:tmp);//前驱  -inf to x-1 max 
		if(ty==6) ++x,printf("%d\n",((tmp=MIN(rt[i],-inf,inf))==inf)?2147483647:tmp);//后继  x+1 to inf  min 
	}
	return 0;
}
```
