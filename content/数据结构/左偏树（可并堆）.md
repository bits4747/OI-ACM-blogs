#数据结构 
## 简介
左偏树是可在 $O(n\log n)$ 的时间内进行**合并**的**堆式**数据结构。

优点在于可以**高效合并**，但难以进行更为复杂的操作。

时间宽裕的话可以用 **STL + 启发式合并** 代替。

## 支持操作
### 合并
```cpp
int merge(int a,int b){
	if(!a||!b) return a+b;
	if(v[a]>v[b]) swap(a,b);
	rs[a]=merge(rs[a],b);
	if(d[ls[a]]<d[rs[a]]) swap(ls[a],rs[a]);
	d[a]=d[rs[a]]+1;
	return a;
}
```

### 树上可并堆
#### [P1552 [APIO2012] 派遣](https://www.luogu.com.cn/problem/P1552)
合并父亲和儿子，维护每个堆的大小和总花费，若堆的总花费超限，删除大根堆堆顶。

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=1e5+5;
int n,m,f[N],c[N],l[N],sz[N],rt[N],d[N],ls[N],rs[N];
long long ans,s[N];
int merge(int a,int b){
	if(!a||!b) return a+b;
	if(c[a]<c[b]) swap(a,b);
	rs[a]=merge(rs[a],b);
	if(d[ls[a]]<d[rs[a]]) swap(ls[a],rs[a]);
	d[a]=d[rs[a]]+1;
	return a;
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<=n;++i){
		scanf("%d%d%d",&f[i],&c[i],&l[i]),rt[i]=i,sz[i]=1,s[i]=c[i];
		if(c[i]<=m) ans=max(ans,1ll*l[i]);
	}
	for(int i=n;i>1;--i){
		rt[f[i]]=rt[i]=merge(rt[f[i]],rt[i]),sz[f[i]]+=sz[i],s[f[i]]+=s[i];
		while(s[f[i]]>m)
			--sz[f[i]],s[f[i]]-=c[rt[f[i]]],rt[f[i]]=merge(ls[rt[f[i]]],rs[rt[f[i]]]);
		ans=max(ans,1ll*sz[f[i]]*l[f[i]]);
	}
	printf("%lld",ans);
	return 0;
}
```
### ★整堆修改
不改变相对大小的操作可以通过类似线段树懒标记的方式进行。
#### [P3261 [JLOI2015]城池攻占](https://www.luogu.com.cn/problem/P3261)
树上可并堆+整堆修改
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=3e5+5;
int n,m,x,a[N],f[N],dep[N],pos[N],c[N],rt[N],d[N],ls[N],rs[N],dea[N];
long long h[N],v[N],s[N],ad[N],mu[N];
vector<int>vec[N];
void pd(int a,int b,int c){
	if(!ad[a]&&mu[a]==1) return ;
	if(b) s[b]*=mu[a],ad[b]*=mu[a],mu[b]*=mu[a],s[b]+=ad[a],ad[b]+=ad[a];
	if(c) s[c]*=mu[a],ad[c]*=mu[a],mu[c]*=mu[a],s[c]+=ad[a],ad[c]+=ad[a];
	ad[a]=0,mu[a]=1;
}
int merge(int a,int b){
	if(!a||!b) return a+b;
	if(s[a]>s[b]) swap(a,b);
	pd(a,ls[a],rs[a]),rs[a]=merge(rs[a],b);
	if(d[ls[a]]<d[rs[a]]) swap(ls[a],rs[a]);
	d[a]=d[rs[a]]+1;
	return a;
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<=n;++i) scanf("%lld",&h[i]);
	dep[1]=1;
	for(int i=2;i<=n;++i) scanf("%d%d%lld",&f[i],&a[i],&v[i]),dep[i]=dep[f[i]]+1;
	for(int i=1;i<=m;++i)
		scanf("%lld%d",&s[i],&c[i]),mu[i]=1,rt[c[i]]=merge(rt[c[i]],i);
	for(int i=n;i>=1;--i){
		while(rt[i]&&s[rt[i]]<h[i])
			++dea[i],pos[rt[i]]=i,pd(rt[i],ls[rt[i]],rs[rt[i]]),
			rt[i]=merge(ls[rt[i]],rs[rt[i]]);
		if(i==1) break;
		if(!a[i]) ad[rt[i]]+=v[i],s[rt[i]]+=v[i];
		else ad[rt[i]]*=v[i],mu[rt[i]]*=v[i],s[rt[i]]*=v[i];
		rt[f[i]]=merge(rt[f[i]],rt[i]);
	}
	for(int i=1;i<=n;++i) printf("%d\n",dea[i]);
	for(int i=1;i<=m;++i) printf("%d\n",dep[c[i]]-dep[pos[i]]);
	return 0;
}
```
### ★巧妙
#### [P4331 [BalticOI 2004]Sequence 数字序列](https://www.luogu.com.cn/problem/P4331)
**题意：**

给定一个整数序列 $a_1, a_2, ··· , a_n$，要求构造一个递增序列 $b$，使得序列 $a_i$ 和 $b_i$ 的各项之差的绝对值之和 $|a_1 - b_1| + |a_2 - b_2| + ··· + |a_n - b_n|$ 最小。

注：此题真·多倍经验，详见好题区

**题解：**

我们先弱化 $b$ 的限制（上升 $\to$ 不下降）。

考虑两种特殊情况：

1. $a[1]≤a[2]≤…≤a[n]$，最优解为 $b[i]=a[i]$；
2. $a[1]≥a[2]≥…≥a[n]$，最优解为 $b[i]=x$，$x$ 为数列 $a$ 的中位数。

因此，我们可以将原序列分成**单调不升的若干段**，每一段我们都取**中位数** $x$。这样对于每一段来说，都找到了最优解。

但是，如果后面段的解小于前面段的，就会导致 $b$ **失去不降性**。

于是需要对答案进行调整。当 $x_i>x_{i+1}$ 时，我们**合并第 $i,i+1$ 段**，重新求中位数 $x'_i$。由于 $x_{i+1}<x'_i<x_i$，可能需要继续合并修改 $x_{i-1,i-2...}$，需要开个**单调栈**维护。

现在，我们需要**高效**维护每段的**中位数**及**合并**后的**中位数**。

可以发现，进行合并操作后 $x'_i<x_i$，因此小于等于中位数的值都有可能成为答案（大于中位数则为废值）。不妨对每一段维护**大根堆**，将整段中**大于中位数的值全部踢掉**。这样既保留了**可能解**，又将当前答案放到**堆顶**，还可以**方便后续删除**大值。

由于需要**高效合并**，可以写可并堆（**左偏树**）。

最后，要维护 $b$ 单调上升。可以巧妙处理：$a_i-i$。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=1e6+5;
int n,d[N],sz[N],re[N],ls[N],rs[N],t,q[N],st,ans[N];
long long v[N],sum;
int merge(int a,int b){
	if(!a||!b) return a+b;
	if(v[a]<v[b]) swap(a,b);
	rs[a]=merge(rs[a],b);
	if(d[ls[a]]<d[rs[a]]) swap(ls[a],rs[a]);
	d[a]=d[rs[a]]+1;
	return a;
}
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i){
		scanf("%lld",&v[i]),v[i]-=i,q[++t]=i,sz[t]=re[t]=1;
		while(t>1&&v[q[t-1]]>v[q[t]]){
			--t,q[t]=merge(q[t],q[t+1]),sz[t]+=sz[t+1],re[t]+=re[t+1];
			while(re[t]>((sz[t]+1)>>1)) q[t]=merge(ls[q[t]],rs[q[t]]),--re[t];
		}
	}
	for(int i=1;i<=t;++i){
		for(int j=1;j<=sz[i];++j) ans[j+st]=v[q[i]]+j+st,sum+=abs(v[q[i]]-v[j+st]);
		st+=sz[i];
	}
	printf("%lld\n",sum);
	for(int i=1;i<=n;++i) printf("%d ",ans[i]);
	return 0;
}
```
