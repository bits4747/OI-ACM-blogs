#数据结构 #线段树
## [P5490 【模板】扫描线](https://www.luogu.com.cn/problem/P5490)
线段树在图上的小运用。

从上到下把图扫描一遍，每时刻截得的线段长度和即为图的面积。

用线段树维护每一时刻（每一单位 $y$ ）时覆盖的长度，由于数据过大，采用离散化。

总结一下：

1. 对 $x$ 离散化，并存贮结构体 $\{y,x1,x2,ty(1/-1)\}$ ，表示在 $y$ 时  $(x1,x2)$ 覆盖一次或减小覆盖一次。按 $y$ 升序排序。
2. **图上区间的经典操作：** 用线段树维护每一时刻（每一单位 $y$ ）时覆盖的长度 $len$ 。 **注意：由于维护的是区间覆盖长度，每次修改时为 $[x1,x2-1]$，整体范围也减 $1$ ，处理覆盖长度时为 $len[k]=a[r+1]-a[l]$  。**
3. 统计答案 $ans+=len[1]*(s[i+1].yy-s[i].yy)$ 。

一个“偷懒”的点：由于只需维护单位时刻全图线段覆盖长度，**无需 pushdown 。**

**注意：要区分求的是点还是线段，当 $l==r$ 时为点，不能 $len[k]=len[ls(k)]+len[rs(k)]$ ，不然会进入空节点，空间会越界！**

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=1e5+5;
int n,n2,m,x,y,v,xa[N],ya[N],xb[N],yb[N],tr[N<<3];
long long ans,a[N<<1],len[N<<3];
struct line{
	long long yy;
	int x1,x2,ty;
}s[N<<1];
bool cmp(line u,line v){
	return u.yy<v.yy;
}
int ls(int k){return k<<1;}
int rs(int k){return k<<1|1;}
void change(int k,int l,int r){
	if(x<=l&&r<=y){
		tr[k]+=v;
		if(tr[k]) len[k]=a[r+1]-a[l];
		else len[k]=(l==r)?0:len[ls(k)]+len[rs(k)];
		return ;
	}
	int mid=(l+r)>>1;
	if(x<=mid) change(ls(k),l,mid);
	if(y>mid) change(rs(k),mid+1,r);
	if(tr[k]) len[k]=a[r+1]-a[l];
	else len[k]=(l==r)?0:len[ls(k)]+len[rs(k)];
}
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i){
		scanf("%d%d%d%d",&xa[i],&ya[i],&xb[i],&yb[i]);
		a[i]=xa[i],a[i+n]=xb[i];
	}
	sort(a+1,a+1+n+n);
	m=unique(a+1,a+1+n+n)-(a+1);
	for(int i=1;i<=n;++i){
		s[i].yy=ya[i],s[i].x1=lower_bound(a+1,a+1+m,xa[i])-a,s[i].x2=lower_bound(a+1,a+1+m,xb[i])-a,s[i].ty=1;
		s[i+n].yy=yb[i],s[i+n].x1=s[i].x1,s[i+n].x2=s[i].x2,s[i+n].ty=-1;
	}
	n=n<<1;
	sort(s+1,s+1+n,cmp);
	for(int i=1;i<n;++i){
		x=s[i].x1,y=s[i].x2-1,v=s[i].ty;
		change(1,1,n-1);
		ans+=len[1]*(s[i+1].yy-s[i].yy);
	}
	printf("%lld",ans);
	return 0;
}
```

## [P1856 [IOI1998] [USACO5.5] 矩形周长Picture](https://www.luogu.com.cn/problem/P1856)
容易想到从上到下扫描答案更新为 $ans+=Δlen$ 。

思路简单代码双倍的写法：从上到下、从左到右都扫一遍。

当然，也可同时处理。线段树多维护每一时刻的线段段数 $sum$ ，标记区间左右端点是否覆盖即可合并。答案更新为 $ans+=Δlen+2*sum*(s[i+1].yy-s[i].yy)$ 。

**注意：当 $y$ 相同时，加边比减边要先处理，否则一条本不该有贡献的线段会多算一次或多次。**

## [P3722 [AH2017/HNOI2017]影魔](https://www.luogu.com.cn/problem/P3722)
记录每个点 $i$ 左边第一个比它大的点 $L[i]$，和右边第一个比它大的点 $R[i]$。

1. $L[i],R[i]$ 都在区间 $[l,r]$ 内时有 $p1$ 贡献。
2. $L[i]$ 在区间 $[l,r]$ 内时 $(i,min(R[i]-1,r)]$ 有 $p2$ 贡献。
3. $R[i]$ 在区间 $[l,r]$ 内时 $(max(L[i]+1,l),i]$ 有 $p2$ 贡献。

分别按 $L,R$ 排序各用扫描线扫一遍。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=2e5+5;
#define ll long long
int n,m,p1,p2,a[N],h,t,qu[N],L[N],R[N],x,y;
ll v,tr[N<<2],la[N<<2],ans[N];
struct que{int l,r,id;}q[N];
bool cmp1(que x,que y){return x.l==y.l?x.r>y.r:x.l<y.l;}
bool cmp2(que x,que y){return x.r==y.r?x.l<y.l:x.r>y.r;}
struct node{int p,lp,rp;}s[N];
bool CMP1(node x,node y){return x.lp<y.lp;}
bool CMP2(node x,node y){return x.rp>y.rp;}
#define ls k<<1
#define rs k<<1|1
void pd(int k,int l,int r,int mi){
	tr[ls]+=(mi-l+1)*la[k],tr[rs]+=(r-mi)*la[k];
	la[ls]+=la[k],la[rs]+=la[k],la[k]=0;
}
void change(int k,int l,int r){
	if(x<=l&&r<=y){tr[k]+=(r-l+1)*v,la[k]+=v;return ;}
	int mi=(l+r)>>1;
	if(la[k]) pd(k,l,r,mi);
	if(x<=mi) change(ls,l,mi);
	if(y>mi) change(rs,mi+1,r);
	tr[k]=tr[ls]+tr[rs];
}
ll ask(int k,int l,int r){
	if(x<=l&&r<=y) return tr[k];
	int mi=(l+r)>>1;
	if(la[k]) pd(k,l,r,mi);
	if(x<=mi&&y>mi) return ask(ls,l,mi)+ask(rs,mi+1,r);
	if(x<=mi) return ask(ls,l,mi);
	return ask(rs,mi+1,r);
}
void add1(int k,int ty){
	if(s[k].rp<=n) x=y=s[k].rp,v=ty*p1,change(1,1,n);
	x=s[k].p+1,y=s[k].rp-1,v=ty*p2;
	if(x<=y) change(1,1,n);
}
void add2(int k,int ty){
	x=s[k].lp+1,y=s[k].p-1,v=ty*p2;
	if(x<=y) change(1,1,n);
}
int main(){
	scanf("%d%d%d%d",&n,&m,&p1,&p2);
	for(int i=1;i<=n;++i) scanf("%d",&a[i]),s[i].p=i;
	for(int i=1;i<=n;++i){
		while(t&&a[i]>a[qu[t]]) --t;
		s[i].lp=qu[t],qu[++t]=i;
	}
	qu[t=1]=n+1,a[n+1]=n+1;
	for(int i=n;i;--i){
		while(t&&a[i]>a[qu[t]]) --t;
		s[i].rp=qu[t],qu[++t]=i;
	}
	for(int i=1;i<=m;++i) scanf("%d%d",&q[i].l,&q[i].r),q[i].id=i;
	sort(s+1,s+1+n,CMP1),sort(q+1,q+1+m,cmp1);
	h=t=1;
	for(int i=1;i<=m;++i){
		while(t<=n&&s[t].lp<=q[i].r) add1(t,1),++t;
		while(h<t&&s[h].lp<q[i].l) add1(h,-1),++h;
		x=q[i].l,y=q[i].r,ans[q[i].id]+=ask(1,1,n);
	}
	while(h<t) add1(h,-1),++h;
	sort(s+1,s+1+n,CMP2),sort(q+1,q+1+m,cmp2);
	h=t=1;
	for(int i=1;i<=m;++i){
		while(t<=n&&s[t].rp>=q[i].l) add2(t,1),++t;
		while(h<=t&&s[h].rp>q[i].r) add2(h,-1),++h;
		x=q[i].l,y=q[i].r,ans[q[i].id]+=ask(1,1,n)+(q[i].r-q[i].l)*p1;
	}
	for(int i=1;i<=m;++i) printf("%lld\n",ans[i]);
	return 0;
}
```
