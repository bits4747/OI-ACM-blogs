### [P3372 【模板】线段树 1](https://www.luogu.com.cn/problem/P3372)
```cpp
#include<bits/stdc++.h>
using namespace std;
int n,m,ty,x,y; 
long long a[400005],tr[400005],la[400005],v;//4*N
int ls(int k){return k<<1;}
int rs(int k){return (k<<1)|1;}
int mid(int l,int r){return (l+r)>>1;} 
void bui(int k,int l,int r){
	if(l==r){tr[k]=a[l];return ;}
	int mi=mid(l,r);
	bui(ls(k),l,mi);
	bui(rs(k),mi+1,r);
	tr[k]=tr[ls(k)]+tr[rs(k)];//!!!!!!!!!!!!!!
}
void pud(int k,int l,int r){
	int mi=mid(l,r);
	tr[ls(k)]+=(mi-l+1)*la[k],la[ls(k)]+=la[k];
	tr[rs(k)]+=(r-mi)*la[k],la[rs(k)]+=la[k];
	la[k]=0;
}
void add(int k,int l,int r){
	if(x<=l&&r<=y){
		tr[k]+=(r-l+1)*v,la[k]+=v;
		return ;
	}
	pud(k,l,r);//!!!
	int mi=mid(l,r);
	if(x<=mi) add(ls(k),l,mi);
	if(mi<y) add(rs(k),mi+1,r);
	tr[k]=tr[ls(k)]+tr[rs(k)];//!!!!!!!!!!!!!!
}
long long ask(int k,int l,int r){
	if(x<=l&&r<=y) return tr[k];
	pud(k,l,r);//!!!
	int mi=mid(l,r);
	long long sum=0;
	if(x<=mi) sum+=ask(ls(k),l,mi);
	if(mi<y) sum+=ask(rs(k),mi+1,r);
	return sum;
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<=n;++i) scanf("%lld",&a[i]);
	bui(1,1,n);
	for(int i=1;i<=m;++i){
		scanf("%d%d%d",&ty,&x,&y);
		if(ty==1){
			scanf("%lld",&v);
			add(1,1,n);
		}
		else printf("%lld\n",ask(1,1,n));
	}
	return 0;
}
```

## 权值线段树+动态开点
### [CF431E Chemistry Experiment](https://www.luogu.com.cn/problem/CF431E)
这题比较容易想到二分答案，利用区间权值和及数量和判断，但 $1e9$ 的数据范围让人不知如何维护。

引入更高级的线段树应用：**权值线段树**。

- 基本线段树每个点维护一段序列区间的相关值。
- 权值线段树每个点维护一段数值区间的相关值。

通常由于数值范围过大，在可以离线的情况下常采用离散化，若无法离线就要用**动态开点**。

**动态开点**：需要用到此节点才建，可以省去朴素写法若干空结点的空间。

1. 用 $x$ 表示值域，每一次操作最多消耗 $log(x)$ 的空间，所以开 $m·log(x)$ 即可。
1. 无需 $build$ 。
1. $add$ 函数巧妙运用引用调用建新点并更新父节点的左右儿子。

具体看看代码：
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=1e5+5;
const double eps=1e-5;
int n,m,a[N],ty,xx,x,v1,v2,rt,tot;
long long v;
double ll,rr,mid;
struct node{
	int ls,rs,cnt;//cnt--多少个数
	long long sum;//sum--数值之和
}tr[N<<5];
void add(int &k,int l,int r){
	if(!k) k=++tot;
	if(l==r){
		tr[k].sum+=v1,tr[k].cnt+=v2;
		return ;
	}
	int mi=(l+r)>>1;
	if(x<=mi) add(tr[k].ls,l,mi);
	if(x>mi) add(tr[k].rs,mi+1,r);
	tr[k].sum=tr[tr[k].ls].sum+tr[tr[k].rs].sum,tr[k].cnt=tr[tr[k].ls].cnt+tr[tr[k].rs].cnt;
}
long long ask(int k,int l,int r,int t){
	if(!k) return 0;
	if(0<=l&&r<=mid) return t?tr[k].cnt:tr[k].sum;
	int mi=(l+r)>>1;
	long long tmp=0;
	if(0<=mi) tmp+=ask(tr[k].ls,l,mi,t);
	if(mid>mi) tmp+=ask(tr[k].rs,mi+1,r,t);
	return tmp;
}
double solve(){
	ll=0,rr=1e15+1e9;
	while(ll+eps<rr){
		mid=(ll+rr)/2;
		if(ask(rt,0,1e9,0)+v<=ask(rt,0,1e9,1)*mid) rr=mid;
		else ll=mid;
	}
	return rr;
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<=n;++i) scanf("%d",&a[i]),x=a[i],v1=a[i],v2=1,add(rt,0,1e9);
	for(int i=1;i<=m;++i){
		scanf("%d",&ty);
		if(ty==1){
			scanf("%d%lld",&xx,&v);
			v1=-a[xx],v2=-1,x=a[xx],add(rt,0,1e9);
			a[xx]=v1=v,v2=1,x=a[xx],add(rt,0,1e9);
		}
		else{
			scanf("%lld",&v);
			printf("%lf\n",solve());
		}
	}
	return 0;
}
```
##  权值线段树实现平衡树
### [P3369 【模板】普通平衡树](https://www.luogu.com.cn/problem/P3369)
用了动态开点，离散化也行。

维护权值区间出现数的个数（用来处理排名）、最大值（用来处理前驱）、最小值（用来处理后继）。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=1e5+5,inf=1e7+1;
int n,tot,ty,x,v,rt;
struct node{
	int ls,rs,cnt,ma,mi;
}s[N*32];
void change(int &k,int l,int r){
	if(!k) s[++tot]=s[k],k=tot;
	if(l==r){s[k].cnt+=v,s[k].ma=(s[k].cnt?l:-inf),s[k].mi=(s[k].cnt?l:inf);return ;}
	int mid=(l+r)>>1;
	if(x<=mid) change(s[k].ls,l,mid);
	else change(s[k].rs,mid+1,r);
	s[k].cnt=s[s[k].ls].cnt+s[s[k].rs].cnt;
	s[k].ma=(s[s[k].rs].cnt?s[s[k].rs].ma:s[s[k].ls].ma);
	s[k].mi=(s[s[k].ls].cnt?s[s[k].ls].mi:s[s[k].rs].mi);
}
int ask(int k,int l,int r){
	int mid=(l+r)>>1;
	if(x==mid) return s[s[k].ls].cnt;
	if(x<mid) return ask(s[k].ls,l,mid);
	return ask(s[k].rs,mid+1,r)+s[s[k].ls].cnt;
}
int kth(int k,int l,int r){
	if(l==r) return l;
	int mid=(l+r)>>1;
	if(s[s[k].ls].cnt>=x) return kth(s[k].ls,l,mid);
	else{x-=s[s[k].ls].cnt;return kth(s[k].rs,mid+1,r);}
}
int MAX(int k,int l,int r){
	int mid=(l+r)>>1;
	if(x==mid) return s[s[k].ls].ma;
	if(x<mid) return MAX(s[k].ls,l,mid);
	return max(s[s[k].ls].ma,MAX(s[k].rs,mid+1,r));
}
int MIN(int k,int l,int r){
	int mid=(l+r)>>1;
	if(x==mid+1) return s[s[k].rs].mi;
	if(x>mid+1) return MIN(s[k].rs,mid+1,r);
	return min(s[s[k].rs].mi,MIN(s[k].ls,l,mid));
}
int main(){
	scanf("%d",&n);
	s[0].ma=-inf,s[0].mi=inf;
	for(int i=1;i<=n;++i){
		scanf("%d%d",&ty,&x);
		if(ty<=2){
			v=(ty==1?1:-1);
			change(rt,-inf,inf);
		} 
		if(ty==3) --x,printf("%d\n",ask(1,-inf,inf)+1);//查询 x 的排名，小于等于x-1的数+1 
		if(ty==4) printf("%d\n",kth(1,-inf,inf));//查询排名为 x 的数
		if(ty==5) --x,printf("%d\n",MAX(1,-inf,inf));//前驱  -inf to x-1 max 
		if(ty==6) ++x,printf("%d\n",MIN(1,-inf,inf));//后继  x+1 to inf  min 
	}
	return 0;
}
```

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=3e5+5;
char c;
int n,m,a[N],ti[N],fa[N],f[19][N],sz[N],tim,dfn[N],id[N],x,y,v,mx[N<<2],la[N<<2],ad;
vector<int>vec[N];
struct op{int ty,x,y;}s[N];
int find(int u){return fa[u]==u?u:fa[u]=find(fa[u]);}
void dfs(int u,int fa){
	f[0][u]=fa,sz[u]=1,dfn[u]=++tim,id[tim]=u;//区分fa和f，用dfn给节点重新编号 
	for(int i=1;i<19;++i) f[i][u]=f[i-1][f[i-1][u]];//更新父亲倍增信息 
	for(int v:vec[u]) dfs(v,u),sz[u]+=sz[v];
}
#define ls k<<1
#define rs k<<1|1
void build(int k,int l,int r){//线段树区间最值板子
	if(l==r){mx[k]=a[id[l]];return ;}
	int mi=(l+r)>>1;build(ls,l,mi),build(rs,mi+1,r);
	mx[k]=max(mx[ls],mx[rs]);
}
void pd(int k){mx[ls]+=la[k],mx[rs]+=la[k],la[ls]+=la[k],la[rs]+=la[k],la[k]=0;}
void add(int k,int l,int r){
	if(x<=l&&r<=y){mx[k]+=v,la[k]+=v;return ;}
	if(la[k]) pd(k);
	int mi=(l+r)>>1;
	if(x<=mi) add(ls,l,mi);
	if(y>mi) add(rs,mi+1,r);
	mx[k]=max(mx[ls],mx[rs]);
}
int ask(int k,int l,int r){
	if(x<=l&&r<=y) return mx[k];
	if(la[k]) pd(k);
	int mi=(l+r)>>1;
	if(x<=mi&&y>mi) return max(ask(ls,l,mi),ask(rs,mi+1,r));
	if(x<=mi) return ask(ls,l,mi);
	return ask(rs,mi+1,r);
} 
void getxy(int t,int u){//找x所属连通块在时间t内对应的dfn区间 
	if(ti[u]<t){//找到当前连通块的根
		for(int i=18;~i;--i)if(ti[f[i][u]]<t) u=f[i][u];
		u=f[0][u];//!!!

		mi=(l+r)>>1;
		if(ti[vec[u][mi]]<t) l=mi;
		else r=mi;
	}
	x=y=dfn[u];
	if(l!=-1) y=dfn[vec[u][l]]+sz[vec[u][l]]-1;//把儿子整个加入 
}
int main(){

	scanf("%d",&m);

			if(x!=y) ti[y]=i,fa[y]=x,vec[x].push_back(y);//记录并入其他连通块的时间
			//并查集还可以保证u的儿子连u的时间都是在u连接fa之前

		if(c=='A'){

		if(c=='F'){
			scanf(" %c",&c);
			if(c=='1') scanf("%d",&s[i].x),s[i].ty=4;
			if(c=='2') scanf("%d",&s[i].x),s[i].ty=5;
```
