#数据结构 #线段树
## 例题：[P3437 [POI2006]TET-Tetris 3D](https://www.luogu.com.cn/problem/P3437)
### 四分树线段树
![[Pasted image 20230922113515.png]]
如图，容易想到把矩阵四分，用“二维线段树”维护。

但是，**复杂度危险！**

一次询问复杂度可达 $O(n)$ ，总复杂度为 $O(nq)=2e7$ ，加上线段树总所周知的大常数就危险了。

记 $ma$ 矩阵区间内最大值， $la$ 覆盖整个矩阵区间的最大值。

20年前良心主题人不卡人，通过以下优化可过：

1. 由于答案的单调递增性，可以类似标记永久化，不pushdown，每次更新时 $v=ma[xl,xr,yd,yu]+h$ ，即取最大值。
2. ask时，当 $sum>=ma[k]$ ，剪枝。

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=16*1005*1005;
int n,D,S,a,b,v,xl,xr,yd,yu,la[N],ma[N],sum;//ma区间内最大值，la覆盖这个区间的最大值 
int dls(int k){return k<<2;}
int drs(int k){return k<<2|1;}
int uls(int k){return k<<2|2;}
int urs(int k){return k<<2|3;}
void change(int k,int l,int r,int d,int u){
	if(xl<=l&&r<=xr&&yd<=d&&u<=yu){ma[k]=v,la[k]=v;return ;}
	if(l==r){
		int miy=(d+u)>>1;
		if(yd<=miy) change(dls(k),l,r,d,miy);
		if(yu>miy) change(uls(k),l,r,miy+1,u);
		ma[k]=max(ma[dls(k)],ma[uls(k)]);
		return;
	}
	if(d==u){
		int mix=(l+r)>>1;
		if(xl<=mix) change(dls(k),l,mix,d,u);
		if(xr>mix) change(drs(k),mix+1,r,d,u);
		ma[k]=max(ma[dls(k)],ma[drs(k)]);
		return;
	}
	int mix=(l+r)>>1,miy=(d+u)>>1;
	if(xl<=mix&&yd<=miy) change(dls(k),l,mix,d,miy);
	if(xl<=mix&&yu>miy) change(uls(k),l,mix,miy+1,u);
	if(xr>mix&&yd<=miy) change(drs(k),mix+1,r,d,miy);
	if(xr>mix&&yu>miy) change(urs(k),mix+1,r,miy+1,u);
	ma[k]=max(max(ma[dls(k)],ma[uls(k)]),max(ma[drs(k)],ma[urs(k)]));
}
int ask(int k,int l,int r,int d,int u){
	if(sum>=ma[k]) return sum;
	if(xl<=l&&r<=xr&&yd<=d&&u<=yu){sum=max(sum,ma[k]);return sum;}
	sum=max(sum,la[k]);
	if(l==r){
		int miy=(d+u)>>1;
		if(yd<=miy) sum=max(sum,ask(dls(k),l,r,d,miy));
		if(yu>miy) sum=max(sum,ask(uls(k),l,r,miy+1,u));
		return sum;
	}
	if(d==u){
		int mix=(l+r)>>1;
		if(xl<=mix) sum=max(sum,ask(dls(k),l,mix,d,u));
		if(xr>mix) sum=max(sum,ask(drs(k),mix+1,r,d,u));
		return sum;
	}
	int mix=(l+r)>>1,miy=(d+u)>>1;
	if(xl<=mix&&yd<=miy) sum=max(sum,ask(dls(k),l,mix,d,miy));
	if(xl<=mix&&yu>miy) sum=max(sum,ask(uls(k),l,mix,miy+1,u));
	if(xr>mix&&yd<=miy) sum=max(sum,ask(drs(k),mix+1,r,d,miy));
	if(xr>mix&&yu>miy) sum=max(sum,ask(urs(k),mix+1,r,miy+1,u));
	return sum;
}
int main(){
	scanf("%d%d%d",&D,&S,&n);
	for(int i=1;i<=n;++i){
		scanf("%d%d%d%d%d",&a,&b,&v,&xl,&yd);
		xr=xl+a,yu=yd+b,++xl,++yd;
		sum=0,v+=ask(1,1,D,1,S);
		change(1,1,D,1,S);
	}
	printf("%d",ma[1]);
	return 0;
}
```
### 真 · 二维线段树（树套树）★
以 $x$ 轴为二维线段树的外维， $y$ 轴为二维线段树的内维，建立一颗二维线段树 $tdseg$ (Two-dimensional segment tree)。

对于第一维线段树的每一个节点都建立两颗线段树， $ma$ （矩阵区间内任意一点的最大值）和 $la$ （覆盖矩阵区间的最大值）。如图：
![[Pasted image 20230922113526.png]]
所以 $ma.ma$ 表示 $x,y$ 矩阵区间内任意一点的最大值， $ma.la$ 表示 $x$ 范围内任意位置但完全覆盖了 $y$ 的最大值， $la.ma$ 表示完全覆盖了 $x$ 范围的 $y$ 范围内任意位置的最大值， $la.la$ 表示覆盖矩阵区间 $x,y$ 的最大值。

```cpp
/*① maxx=la ：
	误区一：线段树内任意一点ma都不小于la
	因为没有pushdown，ma的值不下传（在change函数里l,r被覆盖就return了），只能保证从线段树树根开始往下走时max(ma)不小于max(la)。
	所以，ask里maxx要max(la)。
*/
/*②
	误区二：v永远大于ma、la
	回顾ma.ma, ma.la, la.ma, la.la的定义，v为矩阵区间（xl,xr,yl,yr）内任意一点的最大值，只能保证v>la.la!!!
	所以要max！
*/
#include<bits/stdc++.h>
using namespace std;
const int N=4005;
int n,D,S,a,b,v,xl,xr,yl,yr;
int ls(int k){return k<<1;}
int rs(int k){return k<<1|1;}
struct segy{
	int ma[N],la[N];
	void change(int k,int l,int r){
		ma[k]=max(ma[k],v);
		if(yl<=l&&r<=yr){la[k]=max(la[k],v);return ;}//②!
		int mi=(l+r)>>1;
		if(yl<=mi) change(ls(k),l,mi);
		if(yr>mi) change(rs(k),mi+1,r);
	}
	int ask(int k,int l,int r){
		if(yl<=l&&r<=yr) return max(ma[k],la[k]);
		int mi=(l+r)>>1,maxx=la[k];//①!
		if(yl<=mi) maxx=max(maxx,ask(ls(k),l,mi));
		if(yr>mi) maxx=max(maxx,ask(rs(k),mi+1,r));
		return maxx;
	}
};
struct segx{
	segy ma[N],la[N];
	void change(int k,int l,int r){
		ma[k].change(1,1,S);
		if(xl<=l&&r<=xr){la[k].change(1,1,S);return ;}
		int mi=(l+r)>>1;
		if(xl<=mi) change(ls(k),l,mi);
		if(xr>mi) change(rs(k),mi+1,r);
	}
	int ask(int k,int l,int r){
		if(xl<=l&&r<=xr) return ma[k].ask(1,1,S);
		int mi=(l+r)>>1,maxx=la[k].ask(1,1,S);//①!
		if(xl<=mi) maxx=max(maxx,ask(ls(k),l,mi));
		if(xr>mi) maxx=max(maxx,ask(rs(k),mi+1,r));
		return maxx;
	}
}tdseg;
int main(){
	scanf("%d%d%d",&D,&S,&n);
	for(int i=1;i<=n;++i){
		scanf("%d%d%d%d%d",&a,&b,&v,&xl,&yl);
		xr=xl+a,yr=yl+b,++xl,++yl;//处理x,y的l,r
		v+=tdseg.ask(1,1,D);//加上矩阵区间内任意一点的最大值（由题意得：立体俄罗斯方块）
		tdseg.change(1,1,D);//维护二维线段树
	}
	xl=1,xr=D,yl=1,yr=S;
	printf("%d",tdseg.ask(1,1,D));
	return 0;
}
```
