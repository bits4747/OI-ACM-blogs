## 历史最值
### 例题
#### [【模板】线段树 3](https://www.luogu.com.cn/problem/P6242)
**题意：**

给出一个长度为 $n$ 的数列 $A$，同时定义一个辅助数组 $B$，$B$ 开始与 $A$ 完全相同。接下来进行了 $m$ 次操作，操作有五种类型，按以下格式给出：

- `1 l r k`：对于所有的 $i\in[l,r]$，将 $A_i$ 加上 $k$（$k$ 可以为负数）。
- `2 l r v`：对于所有的 $i\in[l,r]$，将 $A_i$ 变成 $\min(A_i,v)$。
- `3 l r`：求 $\sum_{i=l}^{r}A_i$。
- `4 l r`：对于所有的 $i\in[l,r]$，求 $A_i$ 的最大值。
- `5 l r`：对于所有的 $i\in[l,r]$，求 $B_i$ 的最大值。

在每一次操作后，我们都进行一次更新，让 $B_i\gets\max(B_i,A_i)$。

对于全部测试数据，保证 $1\leq n,m\leq 5\times 10^5$，$-5\times10^8\leq A_i\leq 5\times10^8$，$op\in[1,5]$，$1 \leq l\leq r \leq n$，$-2000\leq k\leq 2000$，$-5\times10^8\leq v\leq 5\times10^8$。

**题解：**

吉司机线段树板子。

区间加，区间和，求区间最大值为线段树基本操作。

> **区间取 $\min$ 转化为区间加：**
> 
> 把区间内大于 $v$ 的数都减去一个数，使得这些数都变为 $v$。
> 
> 结合之前区间最值操作的思路，当 $se<v<mx$ 时对 $mx$ 下手，当 $se>=v$ 时因不确定涉及数量，暴力下传。
> 
> 由于我们把区间取 $\min$ 转化为了区间加，则导致区间最值与非最值的区间加 tag 有所不同，因此需要分别维护。

> **历史最值：**
> 
> 记录标记 $ad$ 的最大值 $adm$，为 $ad$ 下传前这段时间内的最大值，用以更新 $mxb$。 

注意各个值转移的顺序，~~建议全文背诵。~~

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=5e5+5,inf=-2e9;
int n,m,ty,x,y,v,a[N];
struct node{
	long long sum;
	//sum当前区间和，cnt最大值数量，mxa当前最大值，mxb历史最大值，se当前次大值，ls rs左右儿子 
	int ls,rs,mxa,cnt,mxb,se,ad1,ad1m,ad2,ad2m;
	//ad1 当前最大值的标记,ad1m最大值的标记的历史最大值，2--非最大值的... 
};
struct jlssegtree{
	node s[N<<2];
	void pushup(int k){
		s[k].mxb=max(s[s[k].ls].mxb,s[s[k].rs].mxb),s[k].sum=s[s[k].ls].sum+s[s[k].rs].sum;
		if(s[s[k].ls].mxa==s[s[k].rs].mxa)
			s[k].mxa=s[s[k].ls].mxa,s[k].cnt=s[s[k].ls].cnt+s[s[k].rs].cnt,s[k].se=max(s[s[k].ls].se,s[s[k].rs].se);
		if(s[s[k].ls].mxa>s[s[k].rs].mxa)
			s[k].mxa=s[s[k].ls].mxa,s[k].cnt=s[s[k].ls].cnt,s[k].se=max(s[s[k].ls].se,s[s[k].rs].mxa);
		if(s[s[k].ls].mxa<s[s[k].rs].mxa)
			s[k].mxa=s[s[k].rs].mxa,s[k].cnt=s[s[k].rs].cnt,s[k].se=max(s[s[k].rs].se,s[s[k].ls].mxa);
	}
	void upd(int k,int l,int r,int ad1,int ad1m,int ad2,int ad2m){
		s[k].sum+=1ll*s[k].cnt*ad1+1ll*(r-l+1-s[k].cnt)*ad2,
		s[k].mxb=max(s[k].mxb,s[k].mxa+ad1m),s[k].mxa+=ad1,s[k].se+=(s[k].se==inf)?0:ad2,
		s[k].ad1m=max(s[k].ad1m,s[k].ad1+ad1m),s[k].ad1+=ad1,
		s[k].ad2m=max(s[k].ad2m,s[k].ad2+ad2m),s[k].ad2+=ad2;
	}
	void pushdown(int k,int l,int r){
		int maxx=max(s[s[k].ls].mxa,s[s[k].rs].mxa),mi=(l+r)>>1;
		if(maxx==s[s[k].ls].mxa) upd(s[k].ls,l,mi,s[k].ad1,s[k].ad1m,s[k].ad2,s[k].ad2m); 
		else upd(s[k].ls,l,mi,s[k].ad2,s[k].ad2m,s[k].ad2,s[k].ad2m); 
		if(maxx==s[s[k].rs].mxa) upd(s[k].rs,mi+1,r,s[k].ad1,s[k].ad1m,s[k].ad2,s[k].ad2m); 
		else upd(s[k].rs,mi+1,r,s[k].ad2,s[k].ad2m,s[k].ad2,s[k].ad2m); 
		s[k].ad1=s[k].ad1m=s[k].ad2=s[k].ad2m=0;//!!!!!!!!!
	}
	void build(int k,int l,int r){
		if(l==r){s[k].mxa=s[k].mxb=s[k].sum=a[l],s[k].cnt=1,s[k].se=inf;return ;}
		s[k].ls=k<<1,s[k].rs=k<<1|1;
		int mi=(l+r)>>1;
		build(s[k].ls,l,mi),build(s[k].rs,mi+1,r);
		pushup(k);
	}
	void add(int k,int l,int r){
		if(x<=l&&r<=y){upd(k,l,r,v,v,v,v);return ;}
		pushdown(k,l,r);
		int mi=(l+r)>>1;
		if(x<=mi) add(s[k].ls,l,mi);
		if(y>mi) add(s[k].rs,mi+1,r);
		pushup(k);
	}
	void minn(int k,int l,int r){
		if(s[k].mxa<=v) return ;
		if(x<=l&&r<=y&&s[k].se<v){upd(k,l,r,v-s[k].mxa,v-s[k].mxa,0,0);return ;}
		pushdown(k,l,r);
		int mi=(l+r)>>1;
		if(x<=mi) minn(s[k].ls,l,mi);
		if(y>mi) minn(s[k].rs,mi+1,r);
		pushup(k);
	}
	long long getsum(int k,int l,int r){
		if(x<=l&&r<=y) return s[k].sum;
		pushdown(k,l,r);
		int mi=(l+r)>>1;long long sum=0;
		if(x<=mi) sum+=getsum(s[k].ls,l,mi);
		if(y>mi) sum+=getsum(s[k].rs,mi+1,r);
		return sum;
	}
	int getmxa(int k,int l,int r){
		if(x<=l&&r<=y) return s[k].mxa;
		pushdown(k,l,r);
		int mi=(l+r)>>1,sum=inf;
		if(x<=mi) sum=max(sum,getmxa(s[k].ls,l,mi));
		if(y>mi) sum=max(sum,getmxa(s[k].rs,mi+1,r));
		return sum;
	}
	int getmxb(int k,int l,int r){
		if(x<=l&&r<=y) return s[k].mxb;
		pushdown(k,l,r);
		int mi=(l+r)>>1,sum=inf;
		if(x<=mi) sum=max(sum,getmxb(s[k].ls,l,mi));
		if(y>mi) sum=max(sum,getmxb(s[k].rs,mi+1,r));
		return sum;
	}
}seg;
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<=n;++i) scanf("%d",&a[i]);
	seg.build(1,1,n);
	for(int i=1;i<=m;++i){
		scanf("%d%d%d",&ty,&x,&y);
		if(ty==1) scanf("%d",&v),seg.add(1,1,n);
		if(ty==2) scanf("%d",&v),seg.minn(1,1,n);
		if(ty==3) printf("%lld\n",seg.getsum(1,1,n));
		if(ty==4) printf("%d\n",seg.getmxa(1,1,n));
		if(ty==5) printf("%d\n",seg.getmxb(1,1,n));
	}
	return 0;
}
```
