#数据结构 #线段树
## 合并
目标：将两个线段树合并成一颗新线段树。

核心部分（未新建合并树，重复利用k1）：
```cpp
void merge(int &k1,int k2,int l,int r){//把k2合并进k1
	if(!k1||!k2){k1+=k2;return ;}//如果到当前点任意一颗线段树为空，直接把存在的部分放入合并树，重复利用k1
	if(l==r){/*把k2相关值并入k1，如*/s[k1].cnt+=s[k2].cnt;return ;}//叶子节点直接合并
	int mi=(l+r)>>1;
	merge(s[k1].ls,s[k2].ls,l,mi),merge(s[k1].rs,s[k2].rs,mi+1,r);//合并左右儿子
	pushup(k1);
}
```
不破坏任何树，每次合并新开节点（保守起见空间开上面的 $2$ 倍）：
```cpp
int merge(int k1,int k2,int l,int r){
	if(!k1||!k2) return k1+k2;
	int mi=(l+r)>>1,nk=++cnt;
	if(l==r){tr[nk]=tr[k1]+tr[k2];return nk;}
	ls[nk]=merge(ls[k1],ls[k2],l,mi),rs[nk]=merge(rs[k1],rs[k2],mi+1,r);
	tr[nk]=tr[ls[nk]]+tr[rs[nk]];
	return nk;
}
```
复杂度：

均摊单次复杂度都为 $log(|x|)$ ，总时空复杂度为 $n*log(|x|)$ 。
## 分裂
![[Pasted image 20230922123135.png]]

目标：将橙色线段覆盖的部分（值域区间 $[x,y]$ ）从 $rt[k1]$ 分裂出来到新树 $rt[k2]$ 。

核心部分：
```cpp
void split(int &k1,int &k2,int l,int r){
	if(!k1) return ;//若k1为空，无需继续分裂。
	if(x<=l&&r<=y){k2=k1,k1=0;return ;}
	//若 [l,r] 被 [x,y] 全覆盖，把k1直接给k2，清空k1（引用调用巧妙删边，图中红色）。
	if(!k2) k2=++tot;
	int mi=(l+r)>>1;
	if(x<=mi) split(s[k1].ls,s[k2].ls,l,mi);
	if(y>mi) split(s[k1].rs,s[k2].rs,mi+1,r);
	pushup(k1),pushup(k2);//两棵树都要pushup
}
```
复杂度：

一次操作新增点为左右两条链，时空复杂度都为 $2*\log(|x|)$ ，总时空复杂度为 $2*n*\log(|x|)$ 。
## 模板：[P5494 【模板】线段树分裂](https://www.luogu.com.cn/problem/P5494)（合并+分裂纯纯的板子）
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=2e5+5;
int n,m,ma,ty,id,x,y,v,rt[N],tot,srt=1;
struct node{
	int ls,rs;
	long long cnt;
}s[N*68];
void change(int &k,int l,int r){
	if(!k) k=++tot;
	if(l==r){s[k].cnt+=v;return ;}
	int mi=(l+r)>>1;
	if(x<=mi) change(s[k].ls,l,mi);
	else change(s[k].rs,mi+1,r);
	s[k].cnt=s[s[k].ls].cnt+s[s[k].rs].cnt;
}
void merge(int &k1,int k2,int l,int r){
	if(!k1||!k2){k1+=k2;return ;}
	if(l==r){s[k1].cnt+=s[k2].cnt;return ;}
	int mi=(l+r)>>1;
	merge(s[k1].ls,s[k2].ls,l,mi),merge(s[k1].rs,s[k2].rs,mi+1,r);
	s[k1].cnt=s[s[k1].ls].cnt+s[s[k1].rs].cnt;
}
void split(int &k1,int &k2,int l,int r){
	if(!k1) return ;
	if(x<=l&&r<=y){k2=k1,k1=0;return ;}
	if(!k2) k2=++tot;
	int mi=(l+r)>>1;
	if(x<=mi) split(s[k1].ls,s[k2].ls,l,mi);
	if(y>mi) split(s[k1].rs,s[k2].rs,mi+1,r);
	s[k1].cnt=s[s[k1].ls].cnt+s[s[k1].rs].cnt,s[k2].cnt=s[s[k2].ls].cnt+s[s[k2].rs].cnt;
}
long long ask(int k,int l,int r){
	if(x<=l&&r<=y) return s[k].cnt;
	long long sum=0;
	int mi=(l+r)>>1;
	if(x<=mi) sum+=ask(s[k].ls,l,mi);
	if(y>mi) sum+=ask(s[k].rs,mi+1,r);
	return sum;
}
int kth(int k,int l,int r){
	if(l==r) return l;
	int mi=(l+r)>>1;
	if(s[s[k].ls].cnt>=x) return kth(s[k].ls,l,mi);
	x-=s[s[k].ls].cnt;return kth(s[k].rs,mi+1,r);
}
int main(){
	scanf("%d%d",&n,&m),ma=max(n,m);
	for(int i=1;i<=n;++i) scanf("%d",&v),x=i,change(rt[1],1,ma);
	for(int i=1;i<=m;++i){
		scanf("%d%d%d",&ty,&id,&x);
		if(ty==0){
			scanf("%d",&y),++srt;
			split(rt[id],rt[srt],1,ma);
		}
		if(ty==1) merge(rt[id],rt[x],1,ma);
		if(ty==2) v=x,scanf("%d",&x),change(rt[id],1,ma);
		if(ty==3) scanf("%d",&y),printf("%lld\n",ask(rt[id],1,ma));
		if(ty==4) printf("%d\n",(s[rt[id]].cnt>=x)?kth(rt[id],1,ma):-1);
	}
	return 0;
}
```
## 应用
### [P4556 [Vani有约会]雨天的尾巴 /【模板】线段树合并](https://www.luogu.com.cn/problem/P4556)
运用差分思想，$u-v$ 上发放一个数字变为 $u$ 到根放一个数字，$v$ 到根放一个数字，$u,v$ 的 $lca$ 到根撤回一个数字 ，$f[lca]$ 到根撤回一个数字。对树的每一个点都建立一颗动态开点线段树维护区间类型救济粮的数量，最后线段树合并统计答案。

注意：把 $1$ 个操作分成 $4$ 个，空间复杂度多带一个常数 $4$。

### ★[P5327 [ZJOI2019]语言](https://www.luogu.com.cn/problem/P5327)（合并）
树链剖分+树上差分+线段树合并 的毒瘤题。
#### 题解：
- 简化：求出可以和 $u$ 点贸易的点数。
    
    显然，可以和 $u$ 点贸易的点都在一个连通块内，考虑对 $u$ 维护一棵线段树，把与 $u$ 语言相通的点打上标记，那么 $u$ 的贡献就是线段树内有标记的点数 $-1$。（除去 $u$ 本身）。

现在考虑如何高效维护 $n$ 个点。

可以发现，每次操作需要把**一条树上路径** $(a,b)$ 上的点的线段树加上**同样**的标记。

直接暴力要修改 $n$ 次，由于**一条树上路径**+**同样**，可以差分，统计贡献时再线段树合并。
#### 技巧：
1. 对于一棵只需维护有标记的点数，即线段树上每一个点只有 $0$ 和非 $0$ 的区别，无需 $pushdown$，可以标记永久化，只需记录当前点代表区间**有标记的个数** $siz$ 和**整个区间同时加上的值** $val$。
2. 由于线段树上每一个点只有 $0$ 和非 $0$ 的区别，做差分时只需再 $a,b$ 上 $+1$，在 $f[lca]$（如果有）上 $-2$，不用像一般差分一样进行四次操作。
3. 由于线段树合并单次均摊的时空复杂度为 $log(|x|)$，所以**先合并儿子再进行当前点的修改操作**可以大幅减少时空复杂度。因此，需**记录**下每个点的线段树的**修改操作**，且差分时**从下到上合并**。
4. 在树剖求 $lca$ 时可以**记录路径**，避免处理 $a,b,f[lca]$ 时多次重复跳 $top$ 的过程。
5. 对于每个点 $u$，为方便编写先统一**强制**在线段树上**标记自己**（否则不知道是否有路径覆盖到 $u$，不知统计答案时当不当 $-1$），统计答案时再减去。
#### bug：
1. 求 $lca$ 时**记录路径**要清空。
2. 由于标记永久化，在合并操作中每一步都要 $s[k1].val+=s[k2].val$。
3. 由于标记永久化，在修改操作中不能写成：
    ```cpp
    if(x<=l&&r<=y){s[k].val+=v,s[k].siz=(s[k].val==0?0:r-l+1);return ;}
    ```
   因为当 $val=0$ 时代表整个区间没有同时加上的标记，不代表区间内都无标记，要写成：
    ```cpp
    if(x<=l&&r<=y){s[k].val+=v,s[k].siz=(s[k].val==0?s[s[k].ls].siz+s[s[k].rs].siz:r-l+1);return ;}
    ```

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=1e5+5,M=3e7;
#define pb push_back
#define pair pair<int,int>
#define fi first
#define se second
int n,m,a,b,siz[N],d[N],f[N],son[N],tim,dfn[N],top[N],cnt,rt[N],lca,x,y,v;
long long ans;
vector<int>vec[N];
struct seg{int ls,rs,val,siz;}s[M];
struct op{int x,y,v;};
vector<op>q[N];
vector<pair>pa;
void dfs1(int u,int fa){
	siz[u]=1,d[u]=d[fa]+1,f[u]=fa;
	int maxx=0;
	for(int v:vec[u]){
		if(v==fa) continue;
		dfs1(v,u),siz[u]+=siz[v];
		if(maxx<siz[v]) maxx=siz[v],son[u]=v;
	}
}
void dfs2(int u,int tp){
	dfn[u]=++tim,top[u]=tp;
	if(son[u]) dfs2(son[u],tp);
	for(int v:vec[u]) if(v!=f[u]&&v!=son[u]) dfs2(v,v);
}
int LCA(int a,int b){
	pa.clear();//!!!
	while(top[a]!=top[b]){
		if(d[top[a]]<d[top[b]]) swap(a,b);
		pa.pb(make_pair(dfn[top[a]],dfn[a])),a=f[top[a]];
	}
	if(d[a]>d[b]) swap(a,b);
	return pa.pb(make_pair(dfn[a],dfn[b])),a; 
}
void pu(int k,int l,int r){s[k].siz=(s[k].val==0?s[s[k].ls].siz+s[s[k].rs].siz:r-l+1);}
void add(int &k,int l,int r){
	if(!k) k=++cnt;
	if(x<=l&&r<=y) s[k].val+=v;//!!!
	else{
		int mi=(l+r)>>1;
		if(x<=mi) add(s[k].ls,l,mi);
		if(y>mi) add(s[k].rs,mi+1,r);
	}
	pu(k,l,r);
}
void merge(int &k1,int &k2,int l,int r){
	if(!k1||!k2){k1+=k2;return ;}
	s[k1].val+=s[k2].val;//!!!
	if(l!=r){
		int mi=(l+r)>>1;
		merge(s[k1].ls,s[k2].ls,l,mi),merge(s[k1].rs,s[k2].rs,mi+1,r);
	}
	pu(k1,l,r);
}
void dfs(int u){
	for(int v:vec[u]) if(v!=f[u]) dfs(v),merge(rt[u],rt[v],1,n);
	for(auto i:q[u]) x=i.x,y=i.y,v=i.v,add(rt[u],1,n);
	ans+=s[rt[u]].siz-1;
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<n;++i) scanf("%d%d",&a,&b),vec[a].pb(b),vec[b].pb(a);
	dfs1(1,0),dfs2(1,1);
	for(int i=1;i<=n;++i){
		q[i].pb((op){dfn[i],dfn[i],1});
		if(f[i]) q[f[i]].pb((op){dfn[i],dfn[i],-1});
	}
	for(int i=1;i<=m;++i){
		scanf("%d%d",&a,&b),lca=LCA(a,b);
		for(auto j:pa){
			q[a].pb((op){j.fi,j.se,1}),q[b].pb((op){j.fi,j.se,1});
			if(f[lca]) q[f[lca]].pb((op){j.fi,j.se,-2});
		}
	}
	dfs(1);
	printf("%lld",ans/2);
	return 0;
}
```
