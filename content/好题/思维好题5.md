## [CF1223F Stack Exterminable Arrays](https://www.luogu.com.cn/problem/CF1223F)
记 $(nxt_i,i)$ 为 $i$ 结尾能消掉的最短的串，$f_i$ 为 $A_{1,i}$ 的答案。

当处理到 $i$ 时，判断 $a_{nxt_i-1}$ 是否等于 $a_i$，不等于则找 $nxt_{nxt_i-1}-1……$，直到最近的位置 $la$，若 $la$ 存在，则 $f_i=f_{la}+1$。

但直接暴力跳会 T，巧妙地用 $map_{i,j}$ 表示 $i$ 位前，$a=j$ 的最近位置，巧妙运用 swap $O(1)$ 转移。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=3e5+5;
#define ll long long
int T,n,a,la;
ll f[N],ans;
map<int,int>mp[N];
void solve(){
	scanf("%d",&n),ans=0;
	for(int i=1;i<=n;++i){
		scanf("%d",&a),la=mp[i-1][a];
		if(la>1) swap(mp[la-1],mp[i]);
		f[i]=(la?f[la-1]+1:0),ans+=f[i],mp[i][a]=i;
	}
	for(int i=1;i<=n;++i) mp[i].clear();
	printf("%lld\n",ans);
}
int main(){
	scanf("%d",&T);
	while(T--) solve();
	return 0;
}
```
## [CF1334F Strange Function](https://www.luogu.com.cn/problem/CF1334F)

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=5e5+5;
#define ll long long
int n,a[N],p[N],m,b[N],mt[N];
ll tr[N],la,pr;
void add(int x,ll v){for(++x;x<=m;x+=x&-x) tr[x]+=v;}
ll ask(int x){ll re=0;for(++x;x;x-=x&-x) re+=tr[x];return re;}
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i) scanf("%d",&a[i]);
	for(int i=1;i<=n;++i) scanf("%d",&p[i]);
	scanf("%d",&m),++m;
	for(int i=1;i<m;++i) scanf("%d",&b[i]),mt[b[i]]=i;
	add(1,1e17);
	for(int i=1;i<=n;++i){
		if(mt[a[i]]) la=ask(mt[a[i]]-1);
		add(0,p[i]);
		if(p[i]>0) add(lower_bound(b+1,b+m,a[i])-b,-p[i]);
		if(mt[a[i]]&&(pr=ask(mt[a[i]]))>la)
			add(mt[a[i]],la-pr),add(mt[a[i]]+1,pr-la);
	}
	if((pr=ask(m-1))>=1e16) printf("NO\n"); 
	else printf("YES\n%lld",pr);
	return 0;
}
```

## [CF744C Hongcow Buys a Deck of C](https://www.luogu.com.cn/problem/CF744C)

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=16,M=1<<16,Y=125;
#define pa pair<int,int>
#define fi first
#define se second
char c[N];
int n,m,R,B,r[N],b[N],f[M][Y],u,ans;
pa h[M];
int main(){
	scanf("%d",&n),m=1<<n;
	for(int i=0;i<n;++i) scanf(" %c%d%d",&c[i],&r[i],&b[i]),R+=r[i],B+=b[i];
	for(int i=1;i<m;++i)
		for(int j=0;j<n;++j)if((i>>j)&1){
			if(c[j]=='R') ++h[i].fi;
			else ++h[i].se;
		}
	memset(f,-1,sizeof(f)),f[0][0]=0;
	for(int i=0;i<m;++i)for(int j=0;j<=105;++j)if(f[i][j]!=-1)for(int k=0;k<n;++k)if(!((i>>k)&1))
		u=i|(1<<k),f[u][j+min(r[k],h[i].fi)]=max(f[u][j+min(r[k],h[i].fi)],f[i][j]+min(b[k],h[i].se));
	ans=max(B,R);
	for(int i=0;i<=120;++i)if(~f[m-1][i]) ans=min(ans,max(B-f[m-1][i],R-i));
	printf("%d\n",ans+n);
	return 0;
```

## [CF1332F Independent Set](https://www.luogu.com.cn/problem/CF1332F)
定义 $f_{i,0/1/2}$ 表示 $i$ 与父亲有连边且在独立集、$i$ 与父亲有连边且不在独立集、$i$ 与父亲无连边的方案数。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=3e5+5,p=998244353;
int n,x,y;
long long f[N][3];
vector<int>e[N];
void dfs(int u,int fa){
	f[u][0]=f[u][1]=f[u][2]=1;
	for(int v:e[u])if(v!=fa){
		dfs(v,u);
		f[u][0]=f[u][0]*(f[v][0]+f[v][1]+f[v][2])%p;
		f[u][1]=f[u][1]*(f[v][0]+f[v][2])%p;
		f[u][2]=f[u][2]*f[v][2]%p;
	}
	f[u][2]=(f[u][0]+f[u][1]+p-f[u][2])%p;//u与父亲无连边时怎么选都行，但是要减去儿子都不与u连边的方案数，因为u被孤立时不能选入独立集
}
int main(){
	scanf("%d",&n);
	for(int i=1;i<n;++i) scanf("%d%d",&x,&y),e[x].push_back(y),e[y].push_back(x);
	dfs(1,1);
	printf("%lld",f[1][2]-1);
	return 0;
}
```
## [CF1733E Conveyor](https://www.luogu.com.cn/problem/CF1733E)
![[Pasted image 20230408122447.png]]
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=125;
#define ll long long
int q,x,y;
ll t,b,f[N][N];
ll solve(ll k){
	f[0][0]=k-x-y+1;
	for(int i=0;i<=x;++i)for(int j=(i?0:1);j<=y;++j){
		f[i][j]=0;
		if(i) f[i][j]+=f[i-1][j]/2;
		if(j) f[i][j]+=(f[i][j-1]+1)/2;
	}
	return f[x][y];
}
int main(){
	scanf("%d",&q);
	while(q--){
		scanf("%lld%d%d",&t,&x,&y);
		if(solve(t)!=solve(t-1))printf("YES\n");
		else printf("NO\n");
	}
	return 0;
}
```

## [CF1295F Good Contest](https://www.luogu.com.cn/problem/CF1295F)
![[Pasted image 20230423155859.png]]
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=55,p=998244353;
int n,nn,a[N],b[N],c[N<<1],inv[N],g[N],f[N][N<<1],ans;
int poww(int x,int y){int e=1;while(y){if(y&1)e=1ll*e*x%p; x=1ll*x*x%p,y>>=1;}return e;}
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i) scanf("%d%d",&a[i],&b[i]),++b[i],c[i]=a[i],c[i+n]=b[i]; 
	sort(c+1,c+1+n+n),nn=unique(c+1,c+1+n+n)-(c+1);
	for(int i=1;i<=n;++i) a[i]=lower_bound(c+1,c+1+nn,a[i])-c,b[i]=lower_bound(c+1,c+1+nn,b[i])-c;
	inv[0]=inv[1]=1;
	for(int i=2;i<=n;++i) inv[i]=1ll*(p-p/i)*inv[p%i]%p;
	for(int i=1;i<=nn;++i) f[0][i]=1;
	for(int i=1;i<=n;++i){
		for(int j=a[i];j<b[i];++j){
			g[1]=c[j+1]-c[j];
			for(int k=2;k<=i;++k) g[k]=1ll*g[k-1]*(c[j+1]-c[j]+k-1)%p*inv[k]%p;
			for(int k=i-1;~k;--k){
				(f[i][j]+=1ll*f[k][j+1]*g[i-k]%p)%=p;
				if(a[k]>j||b[k]<=j) break;
			}
		}
		for(int j=nn-1;j;--j) (f[i][j]+=f[i][j+1])%=p;
	}
	ans=f[n][1];
	for(int i=1;i<=n;++i) ans=1ll*ans*poww(c[b[i]]-c[a[i]],p-2)%p;
	printf("%d",ans);
	return 0;
}
```
## [P3643 [APIO2016] 划艇](https://www.luogu.com.cn/problem/P3643)
![[Pasted image 20230423171141.png]]
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=505,p=1e9+7;
int n,nn,a[N],b[N],c[N<<1],inv[N],g[N],f[N],ans;
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i) scanf("%d%d",&a[i],&b[i]),++b[i],c[i]=a[i],c[i+n]=b[i]; 
	sort(c+1,c+1+n+n),nn=unique(c+1,c+1+n+n)-(c+1);
	for(int i=1;i<=n;++i) a[i]=lower_bound(c+1,c+1+nn,a[i])-c,b[i]=lower_bound(c+1,c+1+nn,b[i])-c;
	inv[0]=inv[1]=1;
	for(int i=2;i<=n;++i) inv[i]=1ll*(p-p/i)*inv[p%i]%p;
	f[0]=1;
	for(int j=1;j<nn;++j){
		g[1]=c[j+1]-c[j];
		for(int k=2;k<=n;++k) g[k]=1ll*g[k-1]*max(0,c[j+1]-c[j]+k-1)%p*inv[k]%p;
		for(int i=n;i;--i)if(a[i]<=j&&j<b[i])for(int o=1,k=i-1;~k;--k)
			(f[i]+=1ll*f[k]*g[o]%p)%=p,o+=(a[k]<=j&&j<b[k]);//不在区间内的只能为0
	}
	for(int i=1;i<=n;++i) (ans+=f[i])%=p;
	printf("%d",ans);
	return 0;
}
```
