## [P3942 将军令（弱化版）](https://www.luogu.com.cn/problem/P3942)|| [P3523 [POI2011] DYN-Dynamite（强化版）](https://www.luogu.com.cn/problem/P3523)
以强化版为例。

**题意：** 给定一棵 $n$ 个点的树，一个点可能是黑点($d_i=1$)或者白点($d_i=0$)，试确定 $m$ 个关键点，使得对于所有的黑点，都存在一条长度不大于 $K$ 的路径到达某个关键点，求 $K$ 的最小值。

首先，题面暗示二分答案，问题转化为黑点到关键点距离不超过 $K$ 时最少需要多少个关键点。

$f_{u,0}$ 表示 $u$ 子树内目前最远的没被关键点覆盖的点与 $u$ 的距离，$f_{u,1}$ 表示 $u$ 子树内离 $u$ 最近的点与 $u$ 的距离。

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=3e5+5;
int rd(){
	int x=0;char ch=getchar();
	while(ch<'0'||ch>'9') ch=getchar();
	while('0'<=ch&&ch<='9') x=x*10+ch-'0',ch=getchar();
	return x;
}
int n,m,x,y,l,r,mi,d[N],f[N][2],sum;
vector<int>vec[N];
void dfs(int u,int fa){
	f[u][0]=-n,f[u][1]=n;//初始化
	for(int v:vec[u])
		if(v!=fa) dfs(v,u),f[u][0]=max(f[u][0],f[v][0]+1),f[u][1]=min(f[u][1],f[v][1]+1);
	if(d[u]&&f[u][1]>mi) f[u][0]=max(f[u][0],0);//u为黑点且子树内覆盖不了u，丢给u的父亲
	if(f[u][1]+f[u][0]<=mi) f[u][0]=-n;//u被覆盖
	if(f[u][0]==mi) f[u][0]=-n,f[u][1]=0,++sum;//子树内存在与u距离为mi的未被覆盖的点，必须覆盖
}
int main(){
	n=rd(),m=rd(),l=-1,r=(n+1)>>1;
	for(int i=1;i<=n;++i) d[i]=rd();
	for(int i=1;i<n;++i) x=rd(),y=rd(),vec[x].push_back(y),vec[y].push_back(x);
	while(l+1<r){
		mi=(l+r)>>1;
		sum=0,dfs(1,0);
		if(sum+(f[1][0]>=0)<=m) r=mi;//特判根
		else l=mi;
	}
	printf("%d",r);
	return 0;
}
```
## [CF1146F Leaf Partition](https://www.luogu.com.cn/problem/CF1146F)
**题意：** 设 $L$ 是一个叶子节点的集合，$f(L)$ 为这些集合的最小连通子图。求把叶子分成若干个集合，使得任意$f(x)$和$f(y)$ 都有 $f(x)\cap f(y)=\emptyset$ 的方案数。

**题解：** 定义 $f[u][0]$ 为 $u$ 子树不与外界在同一个最小连通子图的方案数，$f[u][1]$ 为 $u$ 子树与外界在同一个最小连通子图的方案数，$f[u][2]$ 为 $u$ 子树与或不与外界在同一个最小连通子图的都可行方案数。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=2e5+5,p=998244353;
int n,x;
long long f[N][3];//0-no 1-yes 2-either
vector<int>e[N];
void dfs(int u){
	if(!e[u].size()) f[u][0]=f[u][1]=1;
	else f[u][0]=1;
	for(int v:e[u]){
		dfs(v);
		f[u][2]=(f[u][2]*(f[v][0]+f[v][1]+2*f[v][2])+f[u][1]*(f[v][1]+f[v][2]))%p;//*2表示可与也可不与
		f[u][1]=(f[u][0]*(f[v][1]+f[v][2])+f[u][1]*(f[v][0]+f[v][2]))%p;
		f[u][0]=f[u][0]*(f[v][0]+f[v][2])%p;
	}
}
int main(){
	scanf("%d",&n);
	for(int i=2;i<=n;++i) scanf("%d",&x),e[x].push_back(i);
	dfs(1);
	printf("%lld",(f[1][0]+f[1][2])%p);
	return 0;
}
```
## [CF1016F Road Projects](https://www.luogu.com.cn/problem/CF1016F)
**题意：** 给定一棵树，有 $q$ 次询问，每次询问：给定一条长度为 $l$ 的边，任选两个没有边直接相连的点接上这条边， 最大化连边后 $mindis(1,n)$ 。每次询问独立。

**题解：** 

定义 $dis_{i,j}$ 为原树上点 $i$ 到点 $j$ 的距离。

显然，要最大化使连边后新增的路径（比 $dis_{1,n}$ 大就行了），只需找到不相连 $x,y$ 使得 $dis{1,x}+dis_{x,y}$ 最大。

我们把 $1\to n$ 的路径拎出来，把原树变成一条链及若干与链相连的子树。

1. 对于链上的一个点，如果有 $2$ 棵及以上的子树或一棵子树的大小大于等于 $3$，就可以直接选任两个合法点连边（相当于连了废边，走原路显然更小）。
2. 否则，找 $max\{dis{1,x}+dis_{x,y}\}$，注意原来连了边的不能为 $x,y$，详见代码。
 
答案为 $min(dis_{1,n},dis{1,x}+dis_{x,y}+l)$。

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=3e5+5;
#define ll long long
#define mp make_pair
#define pb push_back
#define pa pair<int,ll>
#define fi first
#define se second
int n,m,x,y,dis,b[N],sz[N],fl;
ll d[N],mx[N],pmx=-1,nmx=-1,ans;
vector<pa>e[N];
int dfs(int u,int fa){
	int ffl=0;
	for(pa v:e[u]){
		if(v.fi==fa) continue;
		d[v.fi]=d[u]+v.se,b[u]|=dfs(v.fi,u);
		if(!b[v.fi]) sz[u]+=sz[v.fi],ffl|=(sz[v.fi]>1),mx[u]=max(mx[u],mx[v.fi]+v.se);//对子树大小>=2的标记，记录与链上点的最大距离
	}
	if(u==n) b[u]=1;
	sz[u]+=(!b[u])+(!ffl);//特殊处理标记，方便写代码
	return b[u];
}
void dfs1(int u,int fa){
	//要找x,y时，链上只会剩下没有子树和只有一个大小为1的子树的点
	//pmx表示前缀最大dis[1][x],nmx表示算上与u相连的前缀最大dis[1][x]
	if(~pmx) ans=max(ans,pmx+mx[u]+d[n]-d[u]);
	if(~nmx&&mx[u]) ans=max(ans,nmx+mx[u]+d[n]-d[u]);
	pmx=max(pmx,nmx);
	if(mx[u]) pmx=max(pmx,d[u]+mx[u]);
	else nmx=max(nmx,d[u]+mx[u]);
	for(pa v:e[u])if(v.fi!=fa&&b[v.fi]) dfs1(v.fi,u);
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<n;++i) scanf("%d%d%d",&x,&y,&dis),e[x].pb(mp(y,dis)),e[y].pb(mp(x,dis));
	dfs(1,0);
	for(int i=1;i<=n;++i)if(sz[i]>2){ans=d[n],fl=1;break;}
	if(!fl) dfs1(1,0);
	while(m--){
		scanf("%d",&dis);
		printf("%lld\n",min(d[n],ans+dis));
	}
	return 0;
}
```
## [P4253 [SCOI2015]小凸玩密室](https://www.luogu.com.cn/problem/P4253)
**题意：**
给定一棵有点权边权的完全二叉树，要点亮树上的所有灯泡，要求任意时刻点亮的灯泡必须连通，且点亮一个灯泡后必须先点亮其子树。点亮第一个灯泡不需要钱，之后每点亮一个新的灯泡 $v$ 的为上一个被点亮的灯泡 $u$ 到这个点 $v$ 的距离 $D_{u,v}$，乘以这个点的权值 $a_v$。求最少花费。

**题解：**
由于题目规定，点亮灯泡的过程为：

1. 选定第一个点亮的灯泡 $u$。
2. 点亮它的一个子树，再点亮它另外的子树。
3. 把 $u$ 的父节点变为下一过程的 $u$，递归操作 $2,3$。

$f[i][j]$ 表示点亮 $i$ 之后回到它第 $j$ 个祖先的最小花费，$g[i][j]$ 表示点亮 $i$ 之后回到它第 $j$ 个祖先的另一个儿子的最小花费，转移见代码。

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=2e5+5;
int n,x,y,a[N],d[N];
long long dis[N][19],f[N][19],g[N][19],sum,ans=2e18;//f--fa		g--fa's other son	注：第k祖先从0开始
#define ls (u<<1)
#define rs (u<<1|1)
#define br ((u>>i)^1)
void dfs(int u){
	if(u>n) return ;
	d[u]=d[u>>1]+1;
	for(int i=1;i<d[u];++i) dis[u][i]=dis[u][i-1]+dis[u>>i][0];//处理u到祖先的距离
	dfs(ls),dfs(rs);
	for(int i=0;i<=d[u];++i){
		if(ls>n) g[u][i]=(dis[u][i]+dis[br][0])*a[br],f[u][i]=dis[u][i]*a[u>>(i+1)];//u为叶子
		else if(rs>n) g[u][i]=dis[ls][0]*a[ls]+g[ls][i+1],f[u][i]=dis[ls][0]*a[ls]+f[ls][i+1];//u只有一个儿子
		else g[u][i]=min(dis[ls][0]*a[ls]+g[ls][0]+g[rs][i+1],dis[rs][0]*a[rs]+g[rs][0]+g[ls][i+1]),
			 f[u][i]=min(dis[ls][0]*a[ls]+g[ls][0]+f[rs][i+1],dis[rs][0]*a[rs]+g[rs][0]+f[ls][i+1]);
	} 
} 
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i) scanf("%d",&a[i]);
	for(int i=2;i<=n;++i) scanf("%lld",&dis[i][0]);
	d[0]=-1,dfs(1);
	for(int u=1;u<=n;++u){
		sum=f[u][0];//选u第一个点亮
		for(int i=0;i<d[u];++i){
			if(br>n) sum+=dis[u>>1][0]*a[u>>2];
			else sum+=dis[br][0]*a[br]+f[br][1];
		}
		ans=min(ans,sum);
	}
	printf("%lld",ans);
	return 0;
}
```
## [P3554 [POI2013]LUK-Triumphal arch](https://www.luogu.com.cn/problem/P3554)
**题意：** 给一颗树，$1$ 号节点已经被染黑，其余是白的。两个人轮流操作，每轮 A 选择 $k$ 个点染黑，B 走一步（B起始在 $1$），当 B 走到白点 B 胜，当 A 染完全部的点A胜。求能让 A 获胜的最小的 $k$。

**题解：** 先二分答案，判断 A 一次染 $mid$ 个点能否获胜。$f_i$ 表示 $i$ 子树在 B 走到 $i$ 之前需要染色的个数。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=3e5+5;
int n,x,y,l,r,mi,f[N];
vector<int>e[N];
void dfs(int u,int fa){
	f[u]=-mi;
	for(int v:e[u])if(v!=fa) dfs(v,u),f[u]+=max(0,f[v])+1;	
}
int main(){
	scanf("%d",&n),l=-1,r=n-1;
	for(int i=1;i<n;++i) scanf("%d%d",&x,&y),e[x].push_back(y),e[y].push_back(x);
	while(l+1<r){
		mi=(l+r)>>1,dfs(1,0);
		if(f[1]<=0) r=mi;
		else l=mi;
	}
	printf("%d",r);
	return 0;
}
```
## [CF1175E Minimal Segment Cover](https://www.luogu.com.cn/problem/CF1175E)
**题意：** 给定若干个线段，多次询问覆盖一段区间需要最少几条线段。

先预处理出从一个左端点用一条线段最远能覆盖到哪个右端点，然后倍增处理出每个左端点用 $i$ 条线段最远能覆盖到哪个右端点。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=5e5+5;
int n,m,mx,l,r,p=1,lg,f[N][20],x,y;
int ask(){
	int re=0;
	for(int i=lg;~i;--i)if(f[x][i]<y) x=f[x][i],re+=1<<i;
	return f[x][0]>=y?re+1:-1;
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<=n;++i)
		scanf("%d%d",&l,&r),mx=max(mx,r),f[l][0]=max(f[l][0],r);
	for(int i=1;i<=mx;++i) f[i][0]=max(f[i][0],f[i-1][0]);
	lg=log2(mx);
	for(int i=mx;~i;--i)for(int j=1;j<=lg;++j) f[i][j]=f[f[i][j-1]][j-1];
	for(int i=1;i<=m;++i){
		scanf("%d%d",&x,&y);
		printf("%d\n",ask());
	}
	return 0;
}
```

## [CF543D Road Improvement](https://www.luogu.com.cn/problem/CF543D)
换根 DP。

设 $f_u$ 表示以 $u$ 为根的子树中改良道路的方案数，则 $f_{u}=\prod _{v\in son}\left( f_{v}+1\right)$（$E(u,v)$ 改良为 $v$ 子树的方案，再加上 $E(u,v)$ 不改良）。叶子节点为 $1$。

换根 $u,V$：$f_{u}=\prod _{v\in son,v\ne V}\left( f_{v}+1\right),f_v*=f_u+1$。

因此，多维护前后缀积即可。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=2e5+5,m=1e9+7;
int n,x,p[N],f[N],ans[N];
vector<int>e[N],f1[N],f2[N];
void dfs(int u){
	int sum=1,l=e[u].size()-1;
	for(int v:e[u]) dfs(v),sum=1ll*sum*(f[v]+1)%m,f1[u].push_back(sum);
	f[u]=sum,sum=1;
	for(int i=l;~i;--i) sum=1ll*sum*(f[e[u][i]]+1)%m,f2[u].push_back(sum);
	reverse(f2[u].begin(),f2[u].end());
}
void dfs2(int u){
	ans[u]=1ll*f[u]*(f[p[u]]+1)%m;
	int l=e[u].size();
	for(int i=0;i<l;++i)
		f[u]=1ll*(i?f1[u][i-1]:1)*(i+1==l?1:f2[u][i+1])%m*(f[p[u]]+1)%m,dfs2(e[u][i]);
}
int main(){
	scanf("%d",&n);
	for(int i=2;i<=n;++i) scanf("%d",&p[i]),e[p[i]].push_back(i);
	dfs(1),dfs2(1);
	for(int i=1;i<=n;++i) printf("%d ",ans[i]);
	return 0;
}
```
## [CF592D Super M](https://www.luogu.com.cn/problem/CF592D)
dfs 标记，保留关键点。问题变成到达树上每一个点的最少移动距离。

可以发现，树上一条链上的边可以只走一次，其余要走两次。因此最短距离为 $2*(n-1)-len$，$len$ 为直径的长度。

起始点在直径端点较小处（注意多个直径的情况，细节见代码）。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=123460;
int n,m,x,y,b[N],mx[N],cmx[N],id[N],cid[N],ans,len,iid=N;
vector<int>e[N];
void dfs1(int u,int fa){
	for(int v:e[u])if(v!=fa) dfs1(v,u),b[u]|=b[v];
}
void dfs2(int u,int fa){
	id[u]=cid[u]=n+1,ans+=2;
	for(int v:e[u]){
		if(v==fa||!b[v]) continue;
		dfs2(v,u);
		if(mx[v]+1>mx[u]||(mx[v]+1==mx[u]&&id[u]>id[v])){
			if(mx[u]==cmx[u]) cid[u]=min(cid[u],id[u]);
			else cmx[u]=mx[u],cid[u]=id[u];
			mx[u]=mx[v]+1,id[u]=id[v];
		}
		else if(mx[v]+1>cmx[u]||(mx[v]+1==cmx[u]&&cid[u]>id[v]))
			cmx[u]=mx[v]+1,cid[u]=id[v];
	}
	if(!mx[u]) id[u]=u;
	if(!cmx[u]) cid[u]=u;
	if(mx[u]+cmx[u]==len) iid=min(iid,min(id[u],cid[u]));
	if(mx[u]+cmx[u]>len) len=mx[u]+cmx[u],iid=min(id[u],cid[u]);
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<n;++i) scanf("%d%d",&x,&y),e[x].push_back(y),e[y].push_back(x);
	for(int i=1;i<=m;++i) scanf("%d",&x),b[x]=1;
	dfs1(x,x),dfs2(x,x),ans=ans-2-len;
	printf("%d\n%d",iid,ans);
	return 0;
}
```

```cpp
#include<bits/stdc++.h>
```

## [CF1305G Kuroni and Antihype](https://www.luogu.com.cn/problem/CF1305G)
**题意**：

$n$ 个点，每个点有点权 $a_i$​，$i,j$ 之间有边当且仅当 $a_i \&a_j​=0$。

执行以下操作：
1.  选择一个点 $u$ 染色。
2.  选择一个与染色点 $u$ 相连的且未染色的点 $v$，将 $v$​ 染色，贡献为 $a_u$。

输出最大的答案。

$1\le n\le2×10^5,0\le a_i\le2×10^5$

**题解：**

设 $a_{n+1}​=0$，并且默认 $n+1$ 一开始已经被染色。

如果将每个点在 $2$ 过程中选择的点与它之间连一条边，会得到一棵树。如果将边权定义为 $a_i​+a_j​$，那么最终的答案即为边权之和减去点权之和（每条边定向由父亲连向儿子，每个点只有一个入度，只会多算 $\sum 点权$），于是问题变为求解最大生成树。

那么考虑从大到小枚举边权 $i$，再枚举 $i$ 的子集 $j$，将 $j$ 与 $i\oplus j$ 连边（不用显式建边），跑 Kruskal。

复杂度 $O(3^{18}\alpha(n))$。

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=(1<<18)+5;
int n,mx,a[N],t[N],f[N];
long long ans;
int F(int x){return x==f[x]?x:f[x]=F(f[x]);}
void merge(int a,int b,int x){
	if(!(t[a]&&t[b])) return ;
	a=F(a),b=F(b);
	if(a!=b) ans+=1ll*x*(t[a]+t[b]-1),f[b]=a,t[a]=1;
}
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i)
		scanf("%d",&a[i]),mx=max(mx,a[i]),++t[a[i]],f[a[i]]=a[i],ans-=a[i];
	mx=1<<((int)log2(mx)+1),++t[0];
	for(int i=mx-1;i;--i)
		for(int j=i;j;j=(j-1)&i) merge(j,i^j,i);
	printf("%lld",ans);
	return 0;
}
```
