## 双hash
### [#2427. 「POI2010」珍珠项链 Beads](https://loj.ac/p/2427)
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=5e5+5,mod=1e9+123,MOD=999999929,p=49999;
int n,a[N],ans[N],cnt,maxx,sum;
long long po[N],sum1[N],sum2[N],s1,s2,PO[N],SUM1[N],SUM2[N],S1,S2;
int read(){
	int x=0,f=1;char c=getchar();
	while(c<'0'||c>'9'){if(c=='-') f=-1;c=getchar();}
	while(c>='0'&&c<='9'){x*=10;x+=c-48;c=getchar();}
	return x*f;
}
long long M(long long x,long long y){return x>y?x-y:x;}
long long hs1(int x,int y){return M(sum1[y]+mod-sum1[x-1]*po[y-x+1]%mod,mod);}
long long hs2(int x,int y){return M(sum2[x]+mod-sum2[y+1]*po[y-x+1]%mod,mod);}
long long HS1(int x,int y){return M(SUM1[y]+MOD-SUM1[x-1]*PO[y-x+1]%MOD,MOD);}
long long HS2(int x,int y){return M(SUM2[x]+MOD-SUM2[y+1]*PO[y-x+1]%MOD,MOD);}
int main(){
	n=read(),po[0]=1,PO[0]=1;
	for(int i=1;i<=n;++i)
		a[i]=read(),po[i]=po[i-1]*p%mod,sum1[i]=M(sum1[i-1]*p%mod+a[i],mod),PO[i]=PO[i-1]*p%MOD,SUM1[i]=M(SUM1[i-1]*p%MOD+a[i],MOD);
	for(int i=n;i>=1;--i) sum2[i]=M(sum2[i+1]*p%mod+a[i],mod),SUM2[i]=M(SUM2[i+1]*p%MOD+a[i],MOD);
	for(int k=1;k<=n;++k){
		map<int,bool>mp,MP;
		sum=0;
		for(int i=1;i+k-1<=n;i+=k){
			s1=hs1(i,i+k-1),s2=hs2(i,i+k-1);
			S1=HS1(i,i+k-1),S2=HS2(i,i+k-1);
			if((!mp[s1]&&!mp[s2])||(!MP[S1]&&!MP[S2])) ++sum;
			mp[s1]=mp[s2]=MP[S1]=MP[S2]=1;
		}
		if(sum==maxx) ans[++cnt]=k;
		if(sum>maxx){maxx=sum,ans[cnt=1]=k;}
	}
	printf("%d %d\n",maxx,cnt);
	for(int i=1;i<=cnt;++i) printf("%d ",ans[i]);
	return 0;
}
```
