## [P1368 【模板】最小表示法](https://www.luogu.com.cn/problem/P1368)
![[Pasted image 20230925144721.png]]

复杂度：$O(N)$
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=3e5+5;
int n,a[N],x,y=1,k;
int main(){
	scanf("%d",&n);
	for(int i=0;i<n;++i) scanf("%d",&a[i]);
	while(k<n&&x<n&&y<n){
		if(a[(x+k)%n]==a[(y+k)%n]) ++k;
		else{
			if(a[(x+k)%n]>a[(y+k)%n]) x=x+k+1;
			else y=y+k+1;
			if(x==y) ++y;//！！！
			k=0;
		}
	}
	x=min(x,y);
	for(int i=x;i<n;++i) printf("%d ",a[i]);
	for(int i=0;i<x;++i) printf("%d ",a[i]);
	return 0;
}
```
