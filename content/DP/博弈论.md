## 入门
### Bash Game：

$n$ 个物品，最少取 $1$ 个，最多取 $m$ 个，先取完者胜。

给对手留下 $(m+1)$ 的倍数肯定获胜。若 $n\%(m+1)==0$，先手必败。

### Nim Game：

$n$ 堆物品，取某一堆的若干个，至少取一个，多者不限，先取完者胜。

当 $x_1$ ^ $x_2$ ^ ... ... ^ $x_n \ne 0$ 先手必胜，反之。

核心：第一步取石子使得剩下的异或值为 $0$，然后不断模仿后手。

### Wythoff's Game：

两堆若干个，轮流从某一堆取任意个或同时从两堆取同样多任意个，最少一个，多者不限。先取完者胜。

局势：$(ak,bk)$

只要 $ak=(int) (bk - ak) * ( \sqrt5 + 1 ) / 2$ 先手必输，否则先手必胜

- 模板：多个有向图游戏：[#10243. 「一本通 6.7 例 3」移棋子游戏](https://loj.ac/p/10243)

```cpp
#include<bits/stdc++.h>
using namespace std;
int n,m,k,x,y,a,ans,sg[2005];
vector<int>vec[2005];
bool bo[2005];
void dfs(int k){
	if(bo[k]) return;
	bo[k]=1;
	int len=vec[k].size();
	bool vis[2005];
	for(int i=0;i<=n;++i) vis[i]=0;
	for(int i=0;i<len;++i) dfs(vec[k][i]),vis[sg[vec[k][i]]]=1;
	for(int i=0;i<=n;++i) if(!vis[i]){sg[k]=i;break;}
}
int main(){
	scanf("%d%d%d",&n,&m,&k);
	for(int i=1;i<=m;++i) scanf("%d%d",&x,&y),vec[x].push_back(y);
	for(int i=1;i<=n;++i) if(!bo[i]) dfs(i);
	for(int i=1;i<=k;++i) scanf("%d",&a),ans^=sg[a];
	if(!ans) printf("lose");
	else printf("win");
	return 0;
}
```
- [P3185 [HNOI2007]分裂游戏](https://www.luogu.com.cn/problem/P3185)

![[Pasted image 20230925144421.png]]

**重点：有时不把堆看成子游戏，而可以把单个石子看成子游戏。**

$$SG(x)=mex\{SG(j)⊕SG(k)∣j∈[0,x−1],k∈[0,j]\}$$

```cpp
#include<bits/stdc++.h>
using namespace std;
int t,n,p[25],sg[25],mex,ans,sum;
bool vis[505];
int main(){
	scanf("%d",&t);
	sg[1]=0;
	for(int i=2;i<=21;++i){
		memset(vis,0,sizeof(vis));
		for(int j=1;j<i;++j)
			for(int k=1;k<=j;++k) vis[sg[j]^sg[k]]=1;
		mex=0;
		while(vis[mex]) ++mex;
		sg[i]=mex;
	}
	while(t--){
		scanf("%d",&n),ans=sum=0;
		for(int i=n;i>=1;--i){
			scanf("%d",&p[i]);
			ans^=(p[i]&1)?sg[i]:0;
		}
		if(!ans){printf("%d %d %d\n%d\n",-1,-1,-1,0);continue;}
		for(int i=n;i>=1;--i){
			if(!p[i]) continue;
			for(int j=i-1;j>=1;--j){
				for(int k=j;k>=1;--k){
					if(ans^sg[i]^sg[j]^sg[k]) continue;
					++sum;
					if(sum==1) printf("%d %d %d\n",n-i,n-j,n-k);
				}
			}
		}
		printf("%d\n",sum);
	}
	return 0;
}
```

- [P5363 [SDOI2019]移动金币](https://www.luogu.com.cn/problem/P5363)

题解：[click](https://www.luogu.com.cn/blog/C-r-e-e-d/solution-p5363)

```cpp
#include<bits/stdc++.h>
using namespace std;
const int mod=1e9+9;
int n,m,lgn,p,c[150055][55],om,cb;
long long f[20][150005],ans;//放到第i位，用了j个石子 
int main(){
	scanf("%d%d",&n,&m);//n个石子，m堆 
	n-=m,++m,om=m/2,cb=m-om-1;
	lgn=log(n)/log(2);
	for(int i=0;i<=n+m;++i) c[i][0]=1;
	for(int i=1;i<=n+m;++i)
		for(int j=1;j<=min(i,m);++j) c[i][j]=(c[i-1][j-1]+c[i-1][j])%mod;
	f[0][n]=1;
	for(int i=0;i<=lgn;++i){
		p=1<<i;
		for(int j=n;j>=0;--j){
			if(!f[i][j]) continue;
			for(int k=0;k*p<=j&&k<=om;k+=2)
				f[i+1][j-k*p]=(f[i+1][j-k*p]+f[i][j]*c[om][k]%mod)%mod;
		}
	}
	for(int j=0;j<=n;++j) ans=(ans+f[lgn][j]*c[j+cb][cb]%mod)%mod;
	printf("%lld",(c[n+m-1][m-1]+mod-ans)%mod);
	return 0;
}
```

- 例题：[P4279 [SHOI2008]小约翰的游戏](https://www.luogu.com.cn/problem/P4279)

```cpp
#include<bits/stdc++.h>
using namespace std;
int t,n,ans,maxx,x;
int main(){
	scanf("%d",&t);
	while(t--){
		scanf("%d",&n),ans=maxx=0;
		for(int i=1;i<=n;++i)
			scanf("%d",&x),maxx=max(maxx,x),ans^=x;
		if((ans==0&&maxx<=1)||(ans!=0&&maxx>1)) printf("John\n");
		else printf("Brother\n");
	}
	return 0;
}
```
