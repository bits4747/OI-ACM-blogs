```cpp
#include<bits/stdc++.h>
using namespace std;
#define ll long long
int n,m=49;
ll x,a[50],ans;
void insert(){
	for(int i=m;~i;--i){
		if(!((x>>i)&1)) continue;
		if((a[i]>>i)&1){x^=a[i];continue;}
		for(int j=i-1;~j;--j)if((x>>j)&1) x^=a[j];
		for(int j=m;j>i;--j)if((a[j]>>i)&1) a[j]^=x;//保证如果第i位可以异或出1，则有且只有a[i]的第i位为1
		a[i]=x;break;
	}
	/*
	for(int i=m;~i;--i)if((x>>i)&1){
		if((a[i]>>i)&1){x^=a[i];continue;}
		a[i]=x;return x;
	}
	*/
}
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i) scanf("%lld",&x),insert();
	for(int i=0;i<=m;++i) ans=ans^a[i];
	printf("%lld",ans);
	return 0;
}
```

### [CF724G Xor-matic Number of the Graph](https://www.luogu.com.cn/problem/CF724G)
结合上一题可以得到两点间所有路径的异或值。

考虑如何快速求两点间所有路径异或值之和。不妨按位考虑，若线性基 $S$ 里第 $i$ 位存在 $1$，则存在 $2^{|S|-1}$ 种异或方式使得第 $i$ 位为 $1$。

考虑快速统计多点之间所有路径异或之和。随意生成一棵树，求出任意点前缀异或值。若线性基 $S$ 里第 $i$ 位存在 $1$，则任意起点终点都可以造成 $2^{|S|-1}*2^i$ 的贡献；若线性基 $S$ 里第 $i$ 位不存在 $1$，则任意起点终点有且只有一点 $d_i$ 的第 $i$ 位为 $1$ 可造成 $2^{|S|}*2^i$ 的贡献。

## 实数线性基
### [P3265 [JLOI2015]装备购买](https://www.luogu.com.cn/problem/P3265)
```cpp
#include<bits/stdc++.h>
using namespace std;
#define db long double
const int N=505;
const db eps=1e-10;
int n,m,cnt,ans;
db a[N][N],d;
struct eqp{db x[N];int v;}s[N];
bool cmp(eqp a,eqp b){return a.v<b.v;}
int insert(db x[]){
	for(int i=m;i;--i){
		if(abs(x[i])<eps) continue;
		if(abs(a[i][i])<eps){for(int j=1;j<=m;++j)a[i][j]=x[j];return 1;}
		d=x[i]/a[i][i];for(int j=1;j<=m;++j) x[j]-=d*a[i][j];
	}
	return 0;
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<=n;++i)for(int j=1;j<=m;++j) scanf("%Lf",&s[i].x[j]);
	for(int i=1;i<=n;++i) scanf("%d",&s[i].v);
	sort(s+1,s+1+n,cmp);
	for(int i=1;i<=n;++i)if(insert(s[i].x)) ++cnt,ans+=s[i].v;
	printf("%d %d\n",cnt,ans);
	return 0;
}
```
