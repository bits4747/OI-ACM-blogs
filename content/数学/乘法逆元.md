## 离线求逆元（离线处理+单个逆元）
### ecgcd
```cpp
#include<bits/stdc++.h>
using namespace std;
long long n,p,k,kk=1,a,fz,fm=1,x,y,t;
long long exgcd(long long a,long long b){
	if(!b){x=1,y=0;return 1;}
	exgcd(b,a%b);
	t=x,x=y,y=t-a/b*y;
	return x;
}
int main(){
	scanf("%lld%lld%lld",&n,&p,&k);
	for(int i=1;i<=n;++i){
		scanf("%lld",&a);
		kk=kk*k%p,fz=(fz*a%p+kk*fm%p)%p,fm=fm*a%p;//fz的式子是重点，处理前后缀积
	}
	printf("%lld",fz*(exgcd(fm,p)%p+p)%p);
	//注意exgcd求解x,y可能为负，要求正数解要稍加处理 
	return 0;
}
```

## 线性求逆元(1--n)
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=3e6+5;
long long n,p,inv[N];
int main(){
	scanf("%lld%lld",&n,&p);
	inv[0]=inv[1]=1;//0无逆元，方便计算把inv[0]赋为1
	printf("1\n");
	for(int i=2;i<=n;++i) inv[i]=(p-p/i)*inv[p%i]%p,printf("%lld\n",inv[i]);
	return 0;
}
```
