```cpp
#include<bits/stdc++.h>
using namespace std;
int t; 
long long a,b,c,x,y,tt,gcd,sm,bi;
void exgcd(long long a,long long b){
	if(!b){x=1,y=0,gcd=a;return ;}
	exgcd(b,a%b);
	tt=x,x=y,y=tt-(a/b)*y;
}
int main(){
	scanf("%d",&t);
	while(t--){
		scanf("%lld%lld%lld",&a,&b,&c);
		exgcd(a,b);
		if(c%gcd){printf("-1\n");continue;}
		x*=c/gcd,y*=c/gcd,a/=gcd,b/=gcd;
		bi=(y>0)?(y-1)/a:y/a-1,sm=(1-x)>0?(1-x)/b+((1-x)%b!=0):(1-x)/b;
		if(sm>bi) printf("%lld %lld\n",x+sm*b,y-bi*a);
		else printf("%lld %lld %lld %lld %lld\n",bi-sm+1,x+sm*b,y-bi*a,x+bi*b,y-sm*a);
	}
	return 0;
}
```

## CRT 中国剩余定理
给定  $n$ 组非负整数  $a_i, m_i$，其中 $m$ <font color=red size=5>两两互质</font>，求解关于  $x$ 的方程组的最小非负整数解。
$$\begin{cases} x \equiv b_1\ ({\rm mod}\ m_1) \\ x\equiv b_2\ ({\rm mod}\ m_2) \\ ... \\ x \equiv b_n\ ({\rm mod}\ m_n)\end{cases}$$

$$M=\prod ^{n}_{i=1}m_{i}~,~M_{i}=\dfrac{M}{m_{i}}~,~invM_{i}为M_{i}在模m_{i}意义下的逆元$$

$$x=\sum ^{n}_{i=1}b_{i}\cdot M_{i}\cdot invM_{i}$$

很好理解，对于第 $i$ 个同余方程式，第$i$项外都含有$m_{i}$的因子，第$i$项$M_{i}\cdot invM_{i}=1$，使之满足 $x \equiv b_i\ ({\rm mod}\ m_i)$ 。

### [P1495 【模板】中国剩余定理（CRT）](https://www.luogu.com.cn/problem/P1495)

```cpp
#include<bits/stdc++.h>
using namespace std;
int n;
long long m[11],b[11],M=1,x,y,t,ans;
void exgcd(long long a,long long b){
	if(!b){x=1,y=0;return ;}
	exgcd(b,a%b);
	t=x,x=y,y=t-a/b*y;
	return ;
}
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i) scanf("%lld%lld",&m[i],&b[i]),M*=m[i];
	for(int i=1;i<=n;++i){
		exgcd(M/m[i],m[i]);
		ans=(ans+M/m[i]*b[i]*(x<0?x+m[i]:x))%M;
	}
	printf("%lld",ans);
	return 0;
}
```

### [P4777 【模板】扩展中国剩余定理（EXCRT）](https://www.luogu.com.cn/problem/P4777)
```cpp
#include<bits/stdc++.h>
using namespace std;
int n;
__int128 b,m,M,x,y,lcm=1,t,gcd,ans;
__int128 read(){
	__int128 xx=0;
	char ch=getchar();
	while(ch<'0'||ch>'9') ch=getchar();
	while(ch>='0'&&ch<='9') xx=xx*10+ch-'0',ch=getchar();
	return xx;
}
void print(__int128 xx){
	if(xx>9) print(xx/10);
	putchar(xx%10+'0');
}
void exgcd(long long a,long long b){
	if(!b) return x=1,y=0,gcd=a,void();
	exgcd(b,a%b);
	t=x,x=y,y=t-a/b*y;
}
int main(){
	scanf("%d",&n);
	lcm=read(),ans=read();
	for(int i=2;i<=n;++i){
		m=read(),b=read(),M=(b-ans%m+m)%m;
		exgcd(lcm,m);
		if(M%gcd) return 0;
		x=(x*(M/gcd)%m+m)%m,ans+=x*lcm,lcm*=m/gcd,ans=(ans+lcm)%lcm;
	}
	print(ans);
	return 0;
}
```

## BSGS 
$$a^{x}\equiv b\left( mod~p\right)$$
$$a^{x}=a^{i\sqrt{p}-j}\left( \lceil \sqrt{p}\rceil ,0\leq i\leq \sqrt{p},0\leq j <\sqrt{p}\right)$$ 
$$a^{i\sqrt{p} }\equiv a^{j}b\left( mod~m\right)$$
预处理所有 $a^{j}b$ 放入hash表中，枚举 $i$ 即可。

<font color=red size=5> 注意：$a$ , $p$要互质</font>

### [P4195 【模板】扩展 BSGS/exBSGS](https://www.luogu.com.cn/problem/P4195)
```cpp
#include<bits/stdc++.h>
using namespace std;
long long a,aa,b,p,g,t,x,y,k,ans,sp,s,ss;
map<long long,int>mp;
long long gcd(long long a,long long b){return (b?gcd(b,a%b):a);}
void exgcd(long long a,long long b){
	if(!b) return x=1,y=0,void();
	exgcd(b,a%b),t=x,x=y,y=t-a/b*y;
}
void bsgs(){
	sp=ceil(sqrt(p)),s=ss=1,mp.clear(),ans=-1;
	for(int i=0;i<sp;++i) mp[s*b%p]=i,s=s*a%p;
	for(int i=0;i<=sp;++i){
		if(mp.find(ss)!=mp.end()&&i*sp-mp[ss]>=0) return ans=i*sp-mp[ss],void();
		ss=ss*s%p;
	}
}
bool exbsgs(){
	b%=p;
	if(b==1||p==1) return ans=0,1;
	k=0,aa=1;
	while((g=gcd(a,p))>1){
		if(b%g) return 0;
		b/=g,p/=g,aa=aa*(a/g)%p,++k;//注意：先p/=g再更新aa！！！ 
		if(aa==b) return ans=k,1;
	}
	exgcd(aa,p),x=(x%p+p)%p,b=b*x%p;
	bsgs();
	if(ans==-1) return 0;
	return ans+=k,1;
}
int main(){
	while(scanf("%lld%lld%lld",&a,&p,&b)){
		if(!a&&!b&&!p) return 0;
		if(!exbsgs()) printf("No Solution\n");
		else printf("%lld\n",ans);
	}
	return 0;
}
```
