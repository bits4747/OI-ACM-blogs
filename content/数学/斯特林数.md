### [P4609 [FJOI2016]建筑师](https://www.luogu.com.cn/problem/P4609)

![[Pasted image 20230922185020.png]]

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=5e4+1,M=201,p=1e9+7;
int s[N][M],c[M][M],t,n,a,b;
int main(){
	s[0][0]=s[1][1]=1;
	for(int i=2;i<N;++i) s[i][1]=1ll*(i-1)*s[i-1][1]%p;
	for(int i=1;i<N;++i)for(int j=1;j<min(M,i+1);++j)
		s[i][j]=(s[i-1][j-1]+1ll*s[i-1][j]*(i-1))%p;
	for(int i=0;i<M;++i) c[i][0]=1;
	for(int i=1;i<M;++i)for(int j=1;j<=i;++j)
		c[i][j]=(c[i-1][j-1]+c[i-1][j])%p;
	scanf("%d",&t);
	while(t--){
		scanf("%d%d%d",&n,&a,&b);
		printf("%lld\n",1ll*s[n-1][a+b-2]*c[a+b-2][a-1]%p);
	}
	return 0;
}
```

### [CF1278F Cards](https://www.luogu.com.cn/problem/CF1278F)

![[Pasted image 20230922185058.png]]


```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=5005,P=998244353;
int n,m,k,s[N][N],fn,p,ans;
int poww(int x,int y){
	int re=1;
	while(y){if(y&1)re=1ll*re*x%P; x=1ll*x*x%P,y>>=1;}
	return re;
}
int main(){
	scanf("%d%d%d",&n,&m,&k);
	s[0][0]=1;
	for(int i=1;i<=k;++i)for(int j=1;j<=i;++j)
		s[i][j]=(s[i-1][j-1]+1ll*s[i-1][j]*j)%P;
	fn=p=1,m=poww(m,P-2);
	for(int i=0;i<=k;++i)
		ans=(ans+1ll*s[k][i]*fn%P*p%P)%P,fn=1ll*fn*(n-i)%P,p=1ll*p*m%P;
	printf("%d",ans);
	return 0;
}
```
