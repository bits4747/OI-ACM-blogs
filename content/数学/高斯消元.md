```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=55;
int n,m,s;
double a[N][N],d;
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i)
		for(int j=1;j<=n+1;++j) scanf("%lf",&a[i][j]);
	s=1;
	for(int i=1;i<=n;++i){//以第i项为主元
		m=s;//处理到第a[s]个方程（因为会出现某项主元为0的特殊情况，跳过此项不处理，s<=i）
		for(int j=s+1;j<=n;++j) if(fabs(a[m][i])<fabs(a[j][i])) m=j;//找到第i项最大的
		if(fabs(a[m][i])<=1e-15) continue;//主元为0，跳过此项不处理
		swap(a[m],a[s]);//把a[s]变成最大的
		for(int j=1;j<=n;++j){//消掉a[s]以外的方程
			if(j==s) continue;
			d=a[j][i]/a[s][i];
			for(int k=1;k<=n+1;++k) a[j][k]-=d*a[s][k];
		}
		++s;
	}
	if(s<=n){//a[s--n]中系数都为0
		while(s<=n){
			if(a[s][n+1]){printf("-1");return 0;}//若常数项不为0则无解
			++s;
		}
		printf("0");//反之，则有无穷解
		return 0;
	}
	for(int i=1;i<=n;++i) printf("x%d=%.2lf\n",i,a[i][n+1]/a[i][i]);//有唯一解
	return 0;
}
```
