对拍：
```cpp
#include<bits/stdc++.h>
using namespace std;
int main(){
    for(int t=1;t<=10000;++t){
    	system("in.exe");
    	double st=clock();
    	system("mine.exe");
    	double en=clock();
    	system("ans.exe");
    	if(system("fc 1.out 1.ans")){
    		printf("#%d WA!",t);
    		return 0;
		}
		if(en-st>2000)
			printf("#%d TLE time:%.2fms\n",t,en-st);
		else
			printf("#%d AC time:%.2fms\n",t,en-st);
	}
}
```
随机数据：
```cpp
#include<bits/stdc++.h>
using namespace std;
int n;
int main(){
	freopen("1.in","w",stdout);
	srand(time(NULL));
	n=15;
	printf("%d\n",n);
	for(int i=1;i<=n;++i){
		printf("%d ",rand()%2);
	}
	printf("\n1\n2 1 15\n");
	return 0;
}
```
分别在程序里加上
```cpp
freopen("1.in","r",stdin);
freopen("1.out","w",stdout);
```
和
```cpp
freopen("1.in","r",stdin);
freopen("1.ans","w",stdout);
```
运行生成 mine.exe 和 ans.exe，并把对拍程序与里面出现的东西放在一个文件夹里。
