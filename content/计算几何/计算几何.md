#### 随贪乱搞
整体旋转随即角度，以横纵坐标为第一第二关键字排序，取一点前 $5$（小常数）个点计算距离。
```cpp
#include<bits/stdc++.h>
using namespace std;
#define db double
const int N=2e5+5;
const db pi=acos(-1);
int n;
db d,ans=2e18;
struct vec{db x,y;}s[N];
bool cmp(vec a,vec b){return a.x==b.x?a.y<b.y:a.x<b.x;}
vec ar(vec a){return (vec){a.x*cos(d)-a.y*sin(d),a.x*sin(d)+a.y*cos(d)};}
void turn(){
	d=1.0*(rand()%359+1)/180*pi;
	for(int i=1;i<=n;++i) s[i]=ar(s[i]);
}
db dis(vec a,vec b){return (b.x-a.x)*(b.x-a.x)+(b.y-a.y)*(b.y-a.y);}
void solve(){
	sort(s+1,s+1+n,cmp);
	for(int i=1;i<=n;++i)for(int j=min(n,i+5);j>i;--j) ans=min(ans,dis(s[i],s[j]));
}
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i) scanf("%lf%lf",&s[i].x,&s[i].y);
	solve();
	for(int i=1;i<=5;++i) turn(),solve();
	printf("%.4lf",sqrt(ans));
	return 0;
}
```
#### 巧用 set
先把 $S$ 按照 $(x,y)$ 排序，并建一个按 $(y,x)$ 排序的 multiset（无重点 set 就行）。

我们依次把点插入 multiset。

当进行到 $i$ 时，前 $i-1$ 个点的答案为 $ans$，把 multiset 中与 $s_i$ 横坐标差大于 $ans$ 的删掉，并把 multiset 中与 $s_i$ 纵坐标差小于 $ans$ 的点拿出来更新答案，最后把 $s_i$ 放入 multiset。

显然，这样正确性有保证，考虑复杂度。

![[Pasted image 20230329151617.png]]

除去 $i$ 本身，最多有 $5$ 个点，更新答案的复杂度是线性的。总复杂度为 $O(n*log~n)$。
```cpp
#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int N=4e5+5;
int n,h=1;
ll ans;
struct vec{int x,y;}s[N];
bool cmp(vec a,vec b){return a.x==b.x?a.y<b.y:a.x<b.x;}
ll dis(vec a,vec b){return 1ll*(b.x-a.x)*(b.x-a.x)+1ll*(b.y-a.y)*(b.y-a.y);}
struct cmp1{
	bool operator() (vec a,vec b) const {
		return a.y==b.y?a.x<b.x:a.y<b.y;
	}
};
set<vec,cmp1>q;
set<vec>::iterator it; 
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i) scanf("%d%d",&s[i].x,&s[i].y);
	sort(s+1,s+1+n,cmp),ans=dis(s[1],s[2]),q.insert(s[1]),q.insert(s[2]);
	for(int i=3;i<=n;++i){
		while(h<i&&1ll*(s[i].x-s[h].x)*(s[i].x-s[h].x)>=ans) q.erase(s[h++]);
		it=q.lower_bound((vec){s[i].x,ceil(s[i].y-sqrt(ans)+1e-10)});//向上取整下面才能加= 
		for(;it!=q.end();++it){
			if(1ll*((*it).y-s[i].y)*((*it).y-s[i].y)>=ans) break;//不然可能一开始循环就结束了 
			ans=min(ans,dis(*it,s[i]));
		}
		q.insert(s[i]); 
	}
	printf("%lld",ans);
	return 0;
}
```

### [【模板】二维凸包](https://www.luogu.com.cn/problem/P2742)
#### Graham 扫描法
算法过程：

![[Pasted image 20230328200847.png]]

找出左下角的点，其余点按与起点的极角（或与起点连线的斜率）排序，再用叉积正负判断维护凸包。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=1e5+5;
int n,t,st[N];
double ans;
struct node{double x,y;}s[N];
double dis(node a,node b){return sqrt((a.x-b.x)*(a.x-b.x)+(a.y-b.y)*(a.y-b.y));}
double cross(node a,node b,node c){return (b.x-a.x)*(c.y-a.y)-(b.y-a.y)*(c.x-a.x);}
bool cmp(node a,node b){
	return (a.y-s[1].y)*(b.x-s[1].x)==(b.y-s[1].y)*(a.x-s[1].x)?
	dis(s[1],a)<dis(s[1],b):(a.y-s[1].y)*(b.x-s[1].x)<(b.y-s[1].y)*(a.x-s[1].x);
}
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i){
		scanf("%lf%lf",&s[i].x,&s[i].y);
		if(s[i].x<s[1].x||(s[i].x==s[1].x&&s[i].y<s[1].y)) swap(s[i],s[1]);
	}
	sort(s+2,s+1+n,cmp);st[t=1]=1;
	for(int i=2;i<=n;++i){
		while(t>1&&cross(s[st[t-1]],s[st[t]],s[i])<=0) --t;
		st[++t]=i;
	}
	for(int i=2;i<=t;++i) ans+=dis(s[st[i-1]],s[st[i]]);
	ans+=dis(s[1],s[st[t]]);
	printf("%.2lf",ans);
	return 0;
}
```
### [【模板】旋转卡壳](https://www.luogu.com.cn/problem/P1452)
![[Pasted image 20230328200438.png]]
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=5e4+5;
int n,id=1,t=2,st[N],p=2,ans;
struct node{int x,y;}s[N];
int dis(node a,node b){return (a.x-b.x)*(a.x-b.x)+(a.y-b.y)*(a.y-b.y);}
bool cmp(node a,node b){
	return (a.y-s[1].y)*(b.x-s[1].x)==(b.y-s[1].y)*(a.x-s[1].x)?
	dis(s[1],a)<dis(s[1],b):(a.y-s[1].y)*(b.x-s[1].x)<(b.y-s[1].y)*(a.x-s[1].x);
}
int cross(node o,node a,node b){return (a.x-o.x)*(b.y-o.y)-(a.y-o.y)*(b.x-o.x);}
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i){
		scanf("%d%d",&s[i].x,&s[i].y);
		if(s[i].x<s[id].x||(s[i].x==s[id].x&&s[i].y<s[id].y)) id=i;
	}
	swap(s[1],s[id]),sort(s+2,s+1+n,cmp),st[1]=1,st[2]=2;
	for(int i=3;i<=n;++i){
		while(t>1&&cross(s[st[t-1]],s[st[t]],s[i])<=0) --t;
		st[++t]=i;
	}
	st[t+1]=st[1];
	for(int i=1;i<=t;++i){//不能加=，不然比如只有两个点时会死循环 
		while(cross(s[st[p+1]],s[st[i]],s[st[i+1]])>cross(s[st[p]],s[st[i]],s[st[i+1]])) p=p%t+1;
		ans=max(ans,max(dis(s[st[i]],s[st[p]]),dis(s[st[i+1]],s[st[p]])));
	}
	printf("%d",ans);
	return 0;
}
```
###  [【模板】半平面交](https://www.luogu.com.cn/problem/P4196)
首先以原点为极点，将所有凸多边形的边按极角排序。（凸多边形的边按逆时针方向记录）。

显然相交的部分为凸多边形，考虑用一个 deque 来维护这个凸壳。

我们将有效半平面定义为左半平面。

去队尾：

![[Pasted image 20230329154003.png]]

去队首：
![[Pasted image 20230329154222.png]]

只需要通过叉积正负判断首尾两条线段的交点是否在新加入的向量的 “右” 侧即可。

```cpp
#include<bits/stdc++.h>
using namespace std;
#define db double
const int N=505;
int n,m,t,l,r,st[N];
db ta,ans;
struct vec{
	db x,y;
	vec operator +(const vec &b)const{return (vec){x+b.x,y+b.y};}
	vec operator -(const vec &b)const{return (vec){x-b.x,y-b.y};}
	vec operator *(const db &b)const{return (vec){x*b,y*b};}
}fr,a,b,o;
struct line{vec u,v;}s[N];
db cro(vec a,vec b){return a.x*b.y-a.y*b.x;}
bool cmp(line a,line b){
	double ka=atan2(a.v.y-a.u.y,a.v.x-a.u.x),kb=atan2(b.v.y-b.u.y,b.v.x-b.u.x);
	return abs(ka-kb)<1e-10?cro(a.u-b.u,b.v-a.u)>0:ka<kb;
}
vec O(line a,line b){return a.u+(a.v-a.u)*(cro(b.u-a.u,b.v-b.u)/cro(a.v-a.u,b.v-b.u));}
void input(){
	scanf("%d%lf%lf",&m,&a.x,&a.y),fr=a;
	for(int i=2;i<=m;++i) scanf("%lf%lf",&b.x,&b.y),s[++t]=(line){a,b},a=b;
	s[++t]=(line){a,fr};
}
int main(){
	scanf("%d",&n);
	while(n--) input();
	sort(s+1,s+1+t,cmp),st[l=r=1]=1;
	ta=atan2(s[1].v.y-s[1].u.y,s[1].v.x-s[1].u.x);
	while(abs(ta-atan2(s[st[1]+1].v.y-s[st[1]+1].u.y,s[st[1]+1].v.x-s[st[1]+1].u.x))<1e-10) ++st[1];
	for(int i=st[1]+1;i<=t;++i){
		while(l<r&&cro(s[i].u-O(s[st[r]],s[st[r-1]]),s[i].v-O(s[st[r]],s[st[r-1]]))<1e-10) --r;
		while(l<r&&cro(s[i].u-O(s[st[l]],s[st[l+1]]),s[i].v-O(s[st[l]],s[st[l+1]]))<1e-10) ++l;
		st[++r]=i;
	}
	while(l<r&&cro(s[st[l]].u-O(s[st[r]],s[st[r-1]]),s[st[l]].v-O(s[st[r]],s[st[r-1]]))<1e-10) --r;
	o=O(s[st[l]],s[st[r]]);
	for(int i=l+1;i<r;++i) ans+=cro(O(s[st[i-1]],s[st[i]])-o,O(s[st[i]],s[st[i+1]])-o);
	printf("%.3lf",ans/2);
	return 0;
}
```
