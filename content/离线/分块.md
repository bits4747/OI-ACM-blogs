#离线 #分块

部分摘自：[分块思想](https://oi-wiki.org//ds/decompose/)

大佬：[分块相关杂谈](https://www.luogu.com.cn/blog/command-block/fen-kuai-xiang-guan-za-tan)
## 简介
### 分块思想总的来讲就是暴力优化
分块的基本思想是，通过对原数据的适当划分，并在划分后的每一个块上预处理部分信息，从而较一般的暴力算法取得更优的时间复杂度。

分块的时间复杂度主要取决于分块的块长，一般可以通过均值不等式求出某个问题下的最优块长，以及相应的时间复杂度。

分块是一种很灵活的思想，相较于树状数组和线段树，分块的优点是通用性更好，可以维护很多树状数组和线段树无法维护的信息。

当然，分块的复杂度相较于线段树和树状数组不够好。
## 例题
### [#6280. 数列分块入门 4](https://loj.ac/p/6280)
区间加法，区间求和
```cpp
#include<bits/stdc++.h>
using namespace std;
long long n,a[100325],s[320],b[100325],fk,add[320],op,l,r,c,ans;
void ad(){
	if(b[l]==b[r]){
		for(int j=l;j<=r;++j) a[j]+=c;
		s[b[l]]+=c*(r-l+1);
		return ;
	}
	for(int j=b[l]+1;j<b[r];++j) add[j]+=c,s[j]+=fk*c;
	for(int j=l;j<=b[l]*fk;++j) a[j]+=c;
	s[b[l]]+=c*(b[l]*fk-l+1);
	for(int j=(b[r]-1)*fk+1;j<=r;++j) a[j]+=c;
	s[b[r]]+=c*(r-(b[r]-1)*fk);
}
long long ask(){
	ans=0;
	if(b[l]==b[r]){
		for(int j=l;j<=r;++j) ans+=a[j]+add[b[l]],ans%=c+1;
		return ans;
	}
	for(int j=b[l]+1;j<b[r];++j) ans+=s[j],ans%=c+1;
	for(int j=l;j<=b[l]*fk;++j) ans+=a[j]+add[b[l]],ans%=c+1;
	for(int j=(b[r]-1)*fk+1;j<=r;++j) ans+=a[j]+add[b[r]],ans%=c+1;
	return ans;
}
int main(){
	scanf("%lld",&n);
	fk=sqrt(n)+1;
	for(int i=1;i<=n;++i) scanf("%lld",&a[i]),b[i]=(i-1)/fk+1,s[b[i]]+=a[i];
	for(int i=1;i<=n;++i){
		scanf("%lld%lld%lld%lld",&op,&l,&r,&c);
		if(!op) ad();
		else printf("%lld\n",ask());
	}
	return 0;
}
```
### [P4168 [Violet]蒲公英](https://www.luogu.com.cn/problem/P4168)
```cpp
#include<bits/stdc++.h>
using namespace std;
int n,M,a[40005],b[40005],fk,l,r,rea[40005],num[40005],sum,maxx,mid,kma[205][205],tmp,xxx;
map<int,int>m;//离散化 
vector<int>col[40005];
bool vis[40005];
void work(int x){//预处理每块之间（b[i]~b[n]）的众数
	for(int i=1;i<=sum;++i) num[i]=0;
	maxx=mid=0;
	for(int i=(x-1)*fk+1;i<=n;++i){
		++num[a[i]];
		if(num[a[i]]>maxx||(num[a[i]]==maxx&&rea[mid]>rea[a[i]]))
			mid=a[i],maxx=num[a[i]];
		kma[x][b[i]]=mid;
	}
}
int get(int x){return upper_bound(col[x].begin(),col[x].end(),r)-lower_bound(col[x].begin(),col[x].end(),l);}
//二分查找l~r中颜色x的数量 
int ask(){//区间众数只可能是整块的众数或左右散块里的数 
	for(int i=1;i<=sum;++i) vis[i]=0;//记录i是否算过 
	mid=kma[b[l]+1][b[r]-1],vis[mid]=1,maxx=get(mid);
	for(int i=l;i<=b[l]*fk;++i){//左 
		if(vis[a[i]]) continue;
		tmp=get(a[i]);
		if(tmp>maxx||(tmp==maxx&&rea[mid]>rea[a[i]])) mid=a[i],maxx=tmp;
	}
	for(int i=(b[r]-1)*fk+1;i<=r;++i){//右 
		if(vis[a[i]]) continue;
		tmp=get(a[i]);
		if(tmp>maxx||(tmp==maxx&&rea[mid]>rea[a[i]])) mid=a[i],maxx=tmp;
	}
	return xxx=rea[mid];
}
int main(){
	scanf("%d%d",&n,&M);
	fk=sqrt(n)+1;
	for(int i=1;i<=n;++i){
		scanf("%d",&a[i]),b[i]=(i-1)/fk+1;
		if(!m[a[i]]) m[a[i]]=++sum,rea[m[a[i]]]=a[i];//记录颜色 
		a[i]=m[a[i]],col[a[i]].push_back(i);//记录颜色a[i]出现位置 
	}
	for(int i=1;i<=b[n];++i) work(i);
	for(int i=1;i<=M;++i){
		scanf("%d%d",&l,&r);
		l=((l+xxx-1)%n)+1,r=((r+xxx-1)%n)+1;//强制在线 
		if(l>r) swap(l,r);
		printf("%d\n",ask());
	}
	return 0;
}
```
### [P3203 [HNOI2010]弹飞绵羊](https://www.luogu.com.cn/problem/P3203)
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=2e5+5;
int n,m,a[N],b[N],fk,ck[N],to[N],op,pos,k,len,ans;//ck出块步数，to抵达位置 
int ask(){
	ans=0;
	while(pos<=n) ans+=ck[pos],pos=to[pos];
	return ans;
}
void chan(){
	len=(b[pos]-1)*fk+1,a[pos]=k;
	for(int i=pos;i>=len;--i){
		if(b[i]!=b[i+a[i]]) ck[i]=1,to[i]=i+a[i];
		else ck[i]=1+ck[i+a[i]],to[i]=to[i+a[i]];
	}
}
int main(){
	scanf("%d",&n);
	fk=sqrt(n);
	for(int i=1;i<=n;++i) scanf("%d",&a[i]),b[i]=(i-1)/fk+1;
	for(int i=n;i>=1;--i){
		if(b[i]!=b[i+a[i]]) ck[i]=1,to[i]=i+a[i];
		else ck[i]=1+ck[i+a[i]],to[i]=to[i+a[i]];
	}
	scanf("%d",&m);
	for(int i=1;i<=m;++i){
		scanf("%d%d",&op,&pos),++pos;
		if(op==1) printf("%d\n",ask());
		else scanf("%d",&k),chan();
	}
	return 0;
}
```
