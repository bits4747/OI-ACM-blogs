#离线 
## 简介
整体二分是一种优秀的**离线**算法，能够替代很多复杂的数据结构（如树套树）。

对于可二分答案的问题，我们可以采用整体二分的方法处理多组询问。

> 可以使用整体二分解决的题目需要满足以下性质：
> 
> 1. 询问的答案具有**可二分性**
> 
> 2. 修改对判定答案的贡献**互相独立**，修改之间互不影响效果
> 
> 3. 修改如果对判定答案有贡献，则贡献为一确定的与判定标准无关的值
> 
> 4. 贡献满足**交换律，结合律，具有可加性**
> 
> 5. 题目允许使用**离线**算法
## 查询区间第K小
### [P1527 [国家集训队]矩阵乘法](https://www.luogu.com.cn/problem/P1527) 
用二维树状数组维护。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=505,Q=6e4+5;
int n,m,tr[N][N],ans[Q],tmp;
struct node{int x,y,v;}s[N*N];
bool cmp(node a,node b){return a.v<b.v;}
struct que{int xa,ya,xb,yb,k,id;}q[Q],ql[Q],qr[Q];
void add(int x,int y,int v){for(int i=x;i<=n;i+=i&-i)for(int j=y;j<=n;j+=j&-j) tr[i][j]+=v;}
int ask(int x,int y){int re=0;for(int i=x;i;i-=i&-i)for(int j=y;j;j-=j&-j) re+=tr[i][j];return re;}
int get(int lx,int ly,int rx,int ry){return ask(rx,ry)-ask(lx-1,ry)-ask(rx,ly-1)+ask(lx-1,ly-1);}
void solve(int l,int r,int L,int R){//l,r表询问/操作，L,R表值域
	if(L==R){for(int i=l;i<=r;++i) ans[q[i].id]=L;return ;}
	int M=(L+R)>>1,cl=0,cr=0;
	for(int i=L;i<=M;++i) add(s[i].x,s[i].y,1);//加入
	for(int i=l;i<=r;++i){
		if((tmp=get(q[i].xa,q[i].ya,q[i].xb,q[i].yb))<q[i].k) qr[++cr]=q[i],qr[cr].k-=tmp;
		else ql[++cl]=q[i];
	}
	for(int i=L;i<=M;++i) add(s[i].x,s[i].y,-1);//清空
	for(int i=1;i<=cl;++i) q[i+l-1]=ql[i];
	for(int i=1;i<=cr;++i) q[i+l-1+cl]=qr[i];
	solve(l,l+cl-1,L,M),solve(l+cl,r,M+1,R);
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<=n;++i)for(int j=1;j<=n;++j) scanf("%d",&s[++tmp].v),s[tmp].x=i,s[tmp].y=j;
	sort(s+1,s+1+tmp,cmp);
	for(int i=1;i<=m;++i) scanf("%d%d%d%d%d",&q[i].xa,&q[i].ya,&q[i].xb,&q[i].yb,&q[i].k),q[i].id=i;
	solve(1,m,1,tmp);
	for(int i=1;i<=m;++i) printf("%d\n",s[ans[i]].v); 
	return 0;
}
```

### [P2617 Dynamic Rankings](https://www.luogu.com.cn/problem/P2617) 
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=3e5+5;
char C;
int n,m,c,a[N],b[N],cnt,tmp,tr[N],ans[N];
struct op{int ty,x,y,k,id;}s[N],sl[N],sr[N];
void add(int u,int v){for(;u<=n;u+=u&-u) tr[u]+=v;}
int ask(int u){int re=0;for(;u;u-=u&-u) re+=tr[u];return re;}
void solve(int l,int r,int L,int R){
	if(L==R){for(int i=l;i<=r;++i)if(s[i].ty) ans[s[i].id]=a[L];return ;}
	int M=(L+R)>>1,cl=0,cr=0;
	for(int i=l;i<=r;++i){
		if(s[i].ty){
			if((tmp=ask(s[i].y)-ask(s[i].x-1))<s[i].k) sr[++cr]=s[i],sr[cr].k-=tmp;
			else sl[++cl]=s[i];
		}
		else{
			if(s[i].y>M) sr[++cr]=s[i];
			else sl[++cl]=s[i],add(s[i].x,s[i].k);//合法加入
		}
	}
	for(int i=1;i<=cl;++i)if(!sl[i].ty)	add(sl[i].x,-sl[i].k);//清空
	for(int i=1;i<=cl;++i) s[i+l-1]=sl[i];
	for(int i=1;i<=cr;++i) s[i+l-1+cl]=sr[i];
	solve(l,l+cl-1,L,M),solve(l+cl,r,M+1,R);
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<=n;++i) scanf("%d",&a[i]),b[i]=a[i],s[i].x=i,s[i].y=a[i],s[i].k=1;
	c=cnt=n;
	for(int i=1;i<=m;++i){
		scanf(" %c",&C),s[++cnt].id=i;
		if(C=='Q') s[cnt].ty=1,scanf("%d%d%d",&s[cnt].x,&s[cnt].y,&s[cnt].k);
		else{
			scanf("%d%d",&s[cnt].x,&s[cnt].y),s[cnt].k=1,s[cnt+1]=s[cnt];
			s[cnt].k=-1,s[cnt].y=b[s[cnt].x],++cnt,a[++c]=s[cnt].y,b[s[cnt].x]=s[cnt].y;	
		}
	}
	sort(a+1,a+1+c),c=unique(a+1,a+1+c)-(a+1);
	for(int i=1;i<=cnt;++i)if(!s[i].ty) s[i].y=lower_bound(a+1,a+1+c,s[i].y)-a;
	solve(1,cnt,1,c);
	for(int i=1;i<=m;++i)if(ans[i]) printf("%d\n",ans[i]);
	return 0;
}
```

### [P4602 [CTSC2018]混合果汁](https://www.luogu.com.cn/problem/P4602)   
整体二分美味度+线段树上二分

将果汁按 $d_i$ 排序，二分确定好美味度 $D$ 后，将大于 $D$ 的放入线段树维护。

贪心地看，先把线段树里 $p_i$ 小的果汁加入，直到钱不够时比较体积来划分询问。
```cpp
bool cmp(op a,op b){return a.d>b.d;}
void add(int k,int l,int r){
	if(l==r) return pri[k]+=x*v,vol[k]+=v,void();
	int mi=(l+r)>>1;
	if(x<=mi) add(ls,l,mi);
	else add(rs,mi+1,r);
	pri[k]=pri[ls]+pri[rs],vol[k]=vol[ls]+vol[rs];
}
ll ask(int k,int l,int r,ll res){//线段树上二分
	if(l==r) return res*l;
	int mi=(l+r)>>1;
	if(vol[ls]>=res) return ask(ls,l,mi,res);
	return pri[ls]+ask(rs,mi+1,r,res-vol[ls]);
}
void solve(int l,int r,int L,int R){
	if(l>r||L>R) return ;
	if(L==R){for(int i=l;i<=r;++i) ans[s[i].d]=L;return ;}
	int M=(L+R)>>1,cl=0,cr=0;
	while(p&&j[p].d<=M) x=j[p].g,v=-j[p].l,add(1,1,md),--p;
	while(j[p+1].d>M) ++p,x=j[p].g,v=j[p].l,add(1,1,md);//维护线段树
	for(int i=l;i<=r;++i)
		if(vol[1]<s[i].l||(tmp=ask(1,1,md,s[i].l))>s[i].g) sl[++cl]=s[i];
		else sr[++cr]=s[i];//划分询问
	for(int i=1;i<=cl;++i) s[i+l-1]=sl[i];
	for(int i=1;i<=cr;++i) s[i+l-1+cl]=sr[i];
	solve(l,l+cl-1,L,M),solve(l+cl,r,M+1,R);
}
```
## 树上
### [P3250 [HNOI2016] 网络](https://www.luogu.com.cn/problem/P3250) 
整体二分+差分+LCA+树状数组

二分重要程度，需判断是否重要程度大于 $M$ 的路径都经过 $x$ 点。

路径 $(a,b)$ 在 $a,b$ 处 $+1$，在 $lca,f[lca]$ 处 $-1$。

经过 $x$ 点的路径数为路径一端在 $x$ 子树内的数量，即 $ask(dfn[s[i].x]+siz[s[i].x]-1)-ask(dfn[s[i].x]-1)$。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=1e5+5,M=2e5+5;
int n,m,x,y,t,d[N],f[N],siz[N],son[N],tim,dfn[N],top[N],cnt,a[M],lca,tot,tmp,tr[N],ans[M];
vector<int>vec[N];
struct op{int ty,x,y,v,id;}s[M],sl[M],sr[M];
void dfs1(int u,int fa){
	d[u]=d[fa]+1,f[u]=fa,siz[u]=1;
	int maxx=0;
	for(int v:vec[u]){
		if(v==fa) continue;
		dfs1(v,u),siz[u]+=siz[v];
		if(siz[v]>maxx) maxx=siz[v],son[u]=v;
	}
}
void dfs2(int u,int tp){
	dfn[u]=++tim,top[u]=tp;
	if(son[u]) dfs2(son[u],tp);
	for(int v:vec[u])if(v!=f[u]&&v!=son[u]) dfs2(v,v);
}
int LCA(int a,int b){
	while(top[a]!=top[b]){
		if(d[top[a]]>d[top[b]]) a=f[top[a]];
		else b=f[top[b]];
	}
	return d[a]<d[b]?a:b;
}
void add(int u,int v){for(;u<=n;u+=u&-u) tr[u]+=v;}
int ask(int u){int re=0;for(;u;u-=u&-u) re+=tr[u];return re;}
void change(int a,int b){
	lca=LCA(a,b);
	add(dfn[a],tmp),add(dfn[b],tmp),add(dfn[lca],-tmp);
	if(lca!=1) add(dfn[f[lca]],-tmp);
}
void solve(int l,int r,int L,int R){
	if(l>r||L>R) return ;
	if(L==R){for(int i=l;i<=r;++i)if(s[i].ty==2) ans[s[i].id]=L;return ;}
	int M=(L+R)>>1,cl=0,cr=0;
	for(int i=l;i<=r;++i){
		if(s[i].ty<2){
			if(s[i].v<=M) sl[++cl]=s[i];
			else sr[++cr]=s[i],tmp=(s[i].ty?-1:1),tot+=tmp,change(s[i].x,s[i].y);
		}
		else{
			if((tmp=ask(dfn[s[i].x]+siz[s[i].x]-1)-ask(dfn[s[i].x]-1))!=tot) sr[++cr]=s[i];
			else sl[++cl]=s[i];
		}
	}
	for(int i=1;i<=cr;++i)if(sr[i].ty!=2) tmp=(sr[i].ty?1:-1),tot+=tmp,change(sr[i].x,sr[i].y);
	for(int i=1;i<=cl;++i) s[i+l-1]=sl[i];
	for(int i=1;i<=cr;++i) s[i+l-1+cl]=sr[i];
	solve(l,l+cl-1,L,M),solve(l+cl,r,M+1,R);
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<n;++i) scanf("%d%d",&x,&y),vec[x].push_back(y),vec[y].push_back(x);
	dfs1(1,1),dfs2(1,1);
	for(int i=1;i<=m;++i){
		scanf("%d",&s[i].ty),s[i].id=i;
		if(s[i].ty==0) scanf("%d%d%d",&s[i].x,&s[i].y,&s[i].v),a[++cnt]=s[i].v;
		if(s[i].ty==1) scanf("%d",&t),s[i]=s[t],s[i].ty=1;
		if(s[i].ty==2) scanf("%d",&s[i].x);
	}
	sort(a+1,a+1+cnt),cnt=unique(a+1,a+1+cnt)-(a+1),a[0]=-1;
	for(int i=1;i<=m;++i) if(s[i].ty<2) s[i].v=lower_bound(a+1,a+1+cnt,s[i].v)-a;
	memset(ans,-1,sizeof(ans));
	solve(1,m,0,cnt);
	for(int i=1;i<=m;++i) if(ans[i]!=-1) printf("%d\n",a[ans[i]]);
	return 0;
}
```

### [P3242 [HNOI2015] 接水果](https://www.luogu.com.cn/problem/P3242) 
**题意：** 给出 $A$ 类带权路径和 $B$ 类路径，问包含于 $B_i$ 路径中的第 $k_i$ 小的 $A$ 类路径。

又是树上整体二分，难点在于如何判断两路径的包含关系。

设路径 $A_i$ 为 $(x,y)$（$dfn[x]<dfn[y]$，记 $l->dfn,r->dfn+siz-1$），被 $(X,Y)$ 包含：
1. $lca\ne x$：$X,Y$ 在矩阵 $\left\{ [l_x,r_x],[l_y,r_y] \right\}$ 里。
2. $lca=x$：记 $z$ 为路径 $(x,y)$ 上离 $x$ 最近的点，$X,Y$ 在矩阵 $\left\{ [1,l_z),[l_y,r_y] \right\}$ 或 $\left\{ [l_y,r_y],(r_z,n] \right\}$ 里。

树状数组维护扫描线。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=4e4+5,M=2e5+5;
int n,p,pp,q,x,y,z,a[N],d[N],f[N],siz[N],son[N],tim,dfn[N],top[N],lca,tmp,tr[N],ans[N],cnt;
vector<int>vec[N];
struct op{int ty,t,l,r,v,id;}s[M],sl[M],sr[M];
bool cmp(op a,op b){return (a.t==b.t?a.ty<b.ty:a.t<b.t);}
void dfs1(int u,int fa){
	d[u]=d[fa]+1,f[u]=fa,siz[u]=1;
	int maxx=0;
	for(int v:vec[u]){
		if(v==fa) continue;
		dfs1(v,u),siz[u]+=siz[v];
		if(siz[v]>maxx) maxx=siz[v],son[u]=v;
	}
}
void dfs2(int u,int tp){
	dfn[u]=++tim,top[u]=tp;
	if(son[u]) dfs2(son[u],tp);
	for(int v:vec[u]) if(v!=f[u]&&v!=son[u]) dfs2(v,v);
}
int LCA(int a,int b){
	while(top[a]!=top[b]){
		if(d[top[a]]<d[top[b]]) swap(a,b);
		a=f[top[a]];
	}
	return d[a]<d[b]?a:b;
}
int SON(int a,int b){
	while(top[a]!=top[b]){
		if(f[top[b]]==a) return top[b];
		b=f[top[b]];
	}
	return son[a];
}
void add(int u,int v){for(;u<=n;u+=u&-u) tr[u]+=v;}
int ask(int u){int re=0;for(;u;u-=u&-u) re+=tr[u];return re;}
void solve(int l,int r,int L,int R){
	if(l>r||L>R) return ;
	if(L==R){for(int i=l;i<=r;++i)if(s[i].ty) ans[s[i].id]=L;return ;}
	int M=(L+R)>>1,cl=0,cr=0;
	for(int i=l;i<=r;++i){
		if(!s[i].ty){
			if(s[i].v>M) sr[++cr]=s[i];
			else sl[++cl]=s[i],add(s[i].l,s[i].id),add(s[i].r+1,-s[i].id);
		}
		else{
			if((tmp=ask(s[i].l))<s[i].v) sr[++cr]=s[i],sr[cr].v-=tmp;
			else sl[++cl]=s[i];
		}
	}
	for(int i=1;i<=cl;++i)if(!sl[i].ty) add(sl[i].l,-sl[i].id),add(sl[i].r+1,sl[i].id);
	for(int i=1;i<=cl;++i) s[i+l-1]=sl[i];
	for(int i=1;i<=cr;++i) s[i+l-1+cl]=sr[i];
	solve(l,l+cl-1,L,M),solve(l+cl,r,M+1,R);
}
int main(){
	scanf("%d%d%d",&n,&p,&q);
	for(int i=1;i<n;++i) scanf("%d%d",&x,&y),vec[x].push_back(y),vec[y].push_back(x);
	dfs1(1,1),dfs2(1,1);
	for(int i=1;i<=p;++i){
		scanf("%d%d%d",&x,&y,&a[i]);
		if(dfn[x]>dfn[y]) swap(x,y);
		lca=LCA(x,y);
		if(x==lca){
			z=SON(x,y);
			s[++cnt]=(op){0,1,dfn[y],dfn[y]+siz[y]-1,a[i],1},
			s[++cnt]=(op){0,dfn[z],dfn[y],dfn[y]+siz[y]-1,a[i],-1};
			if(dfn[z]+siz[z]-1<n) 
				s[++cnt]=(op){0,dfn[y],dfn[z]+siz[z],n,a[i],1},
				s[++cnt]=(op){0,dfn[y]+siz[y],dfn[z]+siz[z],n,a[i],-1};
		}
		else
			s[++cnt]=(op){0,dfn[x],dfn[y],dfn[y]+siz[y]-1,a[i],1},
			s[++cnt]=(op){0,dfn[x]+siz[x],dfn[y],dfn[y]+siz[y]-1,a[i],-1};
	}
	for(int i=1;i<=q;++i){
		scanf("%d%d%d",&x,&y,&z);
		if(dfn[x]>dfn[y]) swap(x,y);
		s[++cnt]=(op){1,dfn[x],dfn[y],0,z,i};
	}
	sort(s+1,s+1+cnt,cmp),sort(a+1,a+1+p),pp=unique(a+1,a+1+p)-(a+1);
	for(int i=1;i<=cnt;++i)if(!s[i].ty) s[i].v=lower_bound(a+1,a+1+pp,s[i].v)-a;
	solve(1,cnt,1,pp);
	for(int i=1;i<=q;++i) printf("%d\n",a[ans[i]]);
	return 0;
}
```
