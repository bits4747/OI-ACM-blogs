#图论 #网络流

## SSP 算法
Dinic 算法中 BFS 找增广路的过程，替换为用 SPFA 寻找单位费用最小的增广路即可。

复杂度 $O(F*nm)$，因为单次寻找增广路用 SPFA 最劣复杂度为 $O(nm)$。（在平均情况下，SPFA 算法的期望时间复杂度为 $O(E)$）

网络流正常不卡 SPFA，用这个复杂度不错且方便实现动态加边。

### EK：
[P3381 【模板】最小费用最大流](https://www.luogu.com.cn/problem/P3381)
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=5e3+5,M=1e5+5,ma=(1ll<<31)-1;
int n,m,st,en,x,y,W,C,tot=1,h[N],no[N],nxt[M],v[M],w[M],c[M],d[N],vis[N],p[N],e[N],f,ff,ans,val;
void add(int x,int y,int W,int C){nxt[++tot]=h[x],h[x]=tot,v[tot]=y,w[tot]=W,c[tot]=C;}
queue<int>q;
void spfa(){
	for(int i=1;i<=n;++i) d[i]=ma;
	d[st]=0,vis[st]=1,q.push(st);int u;
	while(!q.empty()){
		u=q.front(),q.pop(),vis[u]=0;
		for(int i=h[u];i;i=nxt[i])if(w[i]&&d[v[i]]>d[u]+c[i]){
			d[v[i]]=d[u]+c[i],p[v[i]]=u,e[v[i]]=i;
			if(!vis[v[i]]) vis[v[i]]=1,q.push(v[i]);
		}
	}
}
void dinic(){
	f=ma;
	while(f>0){
		spfa(),ff=ma;
		if(d[en]==ma) return ;
		for(int i=en;i!=st;i=p[i]) ff=min(ff,w[e[i]]);
		f-=ff,ans+=ff,val+=ff*d[en];
		for(int i=en;i!=st;i=p[i]) w[e[i]]-=ff,w[e[i]^1]+=ff;
	}
}
int main(){
	scanf("%d%d%d%d",&n,&m,&st,&en);
	for(int i=1;i<=m;++i) scanf("%d%d%d%d",&x,&y,&W,&C),add(x,y,W,C),add(y,x,0,-C);
	dinic();
	printf("%d %d",ans,val);
	return 0;
}
```

		int f;
		while((f=dfs(st,ma))>0) ans+=f,val+=f*d[en];
	}
}

## Primal-Dual 原始对偶算法
堆优化复杂度 $O(F*N*logM)$。

朴素 dijstra 复杂度 $O(F*N^2)$，适用稠密图。
### EK：
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=405,M=40405,ma=1e9;
int n,m,x,y,C,en,cnt=1,h[N],no[N],nx[M],v[M],w[M],c[M],d[N],H[N],e[N],l[N],f,ff,ans,val;
struct node{
	int p,d;
	bool operator <(const node &b)const{return d>b.d;}
}t;
priority_queue<node>q;
void add(int x,int y,int W,int C){nx[++cnt]=h[x],h[x]=cnt,v[cnt]=y,w[cnt]=W,c[cnt]=C;}
void adde(int x,int y,int W,int C){add(x,y,W,C),add(y,x,0,-C);}
void dij(){
	for(int i=1;i<=en;++i) d[i]=ma;
	d[1]=0,q.push((node){1,0});
	while(!q.empty()){
		t=q.top(),q.pop();
		if(d[t.p]!=t.d) continue;
		for(int i=h[t.p];i;i=nx[i])if(w[i]&&d[v[i]]>t.d+H[t.p]-H[v[i]]+c[i])
			d[v[i]]=t.d+H[t.p]-H[v[i]]+c[i],q.push((node){v[i],d[v[i]]}),e[v[i]]=i,l[v[i]]=t.p;
	}
}
void dinic(){
	f=N;
	while(f>0){
		dij(),ff=N;
		if(d[en]==ma) return ;
		for(int i=1;i<=en;++i) H[i]+=d[i];
		for(int i=en;i!=1;i=l[i]) ff=min(ff,w[e[i]]);
		f-=ff,ans+=ff,val+=H[en]*ff;
		for(int i=en;i!=1;i=l[i]) w[e[i]]-=ff,w[e[i]^1]+=ff;
	}
}
int main(){
	scanf("%d%d",&n,&m);adde(1,n+1,N,0),adde(n,n+n,N,0),en=n+n;
	for(int i=2;i<n;++i) adde(i,i+n,1,0);
	for(int i=1;i<=m;++i) scanf("%d%d%d",&x,&y,&C),adde(x+n,y,1,C);
	dinic();
	printf("%d %d",ans,val);
	return 0;
}
```

### [P2045 方格取数加强版](https://www.luogu.com.cn/problem/P2045)
**建图：**

1. **点**：拆点连容量 $1$，费用 $c$ 的边，表示一点只能贡献一次；再连容量 $inf$，费用 $0$ 的边表示一点可走无限次。（点不限流）
2. **边**：相连的两点 $(u,v)$，$u_出$ 连 $v_入$，容量 $inf$，费用 $0$。（边不限流）

跑最大费用最大流（边权取负）即可。

### [CF818G Four Melodies](https://www.luogu.com.cn/problem/CF818G)
直接暴力连边 $O(n^2)$，会寄。可以发现如需从 $i​$ 到 $j$​ ，可以表示为 $i​$ 到 $k$​ ，再从 $k$​ 到 $j$​（其中 $k$ 在子序列的选取中不会出现，仅作为一个传递点） 。

因此，对于每个点，拆点连容量 $1$，费用 $1$ 的边，再连容量 $inf$，费用 $0$ 的边。每个点的出点与下一个同余的点的入点相连，容量 $inf$，费用 $0$。

此外，还需再拆两个点，分别与下一个 $a_i+1、a_i-1$ 相连。

### [P3159 [CQOI2012]交换棋子](https://www.luogu.com.cn/problem/P3159)

对于初始矩阵 $A$ 与目标矩阵 $B$ 都是 $1$ 的变成 $0$，接下来就是把 $A$ 中的 $1$ 移动到 $B$ 中的 $1$ 上，变成多源多汇最小费用最大流。

考虑在转移过程中经过一点 $u$，移入移出各要花费一次 $u$ 的操作次数，因此可以把所有点的操作次数 $\div2$，表示可以经过的次数。然后拆点连边，容量为 $\frac C2$，费用为 $2$。

但每条路径的起点和终点无需移入/移出，在起点和终点位置的操作次数上$+1$ 再 $\div2$。与源汇点连边容量为 $1$，费用为 $-1$。

### [P4452 [国家集训队]航班安排](https://www.luogu.com.cn/problem/P4452)
补充一下题意：

每一个任务要求与原来非任务的时间、花费无关，相当于多了一条可飞一次的 $s$ 时刻从 $a$ 机场起飞，$t$ 时刻到达 $b$ 机场，可以净获利 $c$ 的航线。要求要在最大化飞过的任务航线的前提下最大化净收益（最小化花费）。

按时间分层会炸。可以观察到如果不是要飞任务航线，原地不动最优；因此只需将任务航线看作点（拆点连花费 $-c$），非任务航线看作花费，将可以连飞（$t_i+dis_{y_i,x_j}\le s_j$）的航线连起来跑最小花费最大流即可。

注意，当可以准时出发（$dis_{1,x_i}\le s_i$ ）的任务航线才与源点连边，可以准时回去的任务航线才与汇点连边。此外，还要建立一个 汇点' 与汇点连边限制流量 $k$。

### [P2050 [NOI2012]美食节](https://www.luogu.com.cn/problem/P2050)
上一题的数据范围加强版，直接连边会炸。

可以发现，有很多边点是用不到的。对于每个厨师，如果倒数第 $i+1$ 的匹配位置没有被占用，那么倒数第 $i$ 的位置绝对不会被占用。

可以每次增广之后把有新增匹配的厨师多加一个匹配位置。

总点数是 $O(n+m+p)$ 的，边数是 $O(n(n+m+p))$。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int F=45,P=105,N=1e3,M=1e5;
int n,m,x,y,p[F],t[P][F],fi[P],b[N],l[N],en,tot,cnt=1,no[N],h[N],nx[M],v[M],w[M],c[M],d[N],vis[N],val;
void add(int x,int y,int W,int C){nx[++cnt]=h[x],h[x]=cnt,v[cnt]=y,w[cnt]=W,c[cnt]=C;}
void adde(int x,int y,int W,int C){add(x,y,W,C),add(y,x,0,-C);}
queue<int>q;
void spfa(){
	for(int i=0;i<=tot;++i) d[i]=1e9,vis[i]=0;
	d[0]=0,vis[0]=1,q.push(0);int u;
	while(!q.empty()){
		u=q.front(),vis[u]=0,q.pop();
		for(int i=h[u];i;i=nx[i])if(w[i]&&d[v[i]]>d[u]+c[i]){
			d[v[i]]=d[u]+c[i];
			if(!vis[v[i]]) vis[v[i]]=1,q.push(v[i]);
		}
	}
}
int dfs(int u,int f){
	if(u==en) return f;
	vis[u]=1;//避免0环 
	for(int &i=no[u];i;i=nx[i])if(!vis[v[i]]&&w[i]&&d[u]+c[i]==d[v[i]]){
		int ff=dfs(v[i],min(w[i],f));
		if(ff) return w[i]-=ff,w[i^1]+=ff,val+=c[i]*ff,ff;
	}
	vis[u]=0;
	return 0;
}
void dinic(){
	while(1){
		spfa();
		if(d[en]==1e9) return;
		for(int i=0;i<=tot;++i) no[i]=h[i];
		//while(dfs(0,inf));以下为本题动态加边加点优化
		if(!dfs(0,n)) return;
		for(int i=h[en];i;i=nx[i])if(w[i]&&l[v[i]]==fi[b[v[i]]]){
			++fi[b[v[i]]],b[++tot]=b[v[i]],l[tot]=fi[b[v[i]]],adde(tot,en,1,0);
			for(int j=1;j<=n;++j) adde(j,tot,1,t[b[v[i]]][j]*l[tot]);
		}
	}
}
int main(){
	scanf("%d%d",&n,&m),tot=en=n+1;
	for(int i=1;i<=n;++i) scanf("%d",&p[i]),adde(0,i,p[i],0);
	for(int i=1;i<=n;++i)for(int j=1;j<=m;++j) scanf("%d",&t[j][i]);
	for(int i=1;i<=m;++i){
		fi[i]=1,b[++tot]=i,l[tot]=1,adde(tot,en,1,0);
		for(int j=1;j<=n;++j) adde(j,tot,1,t[i][j]);
	}
	dinic(),printf("%d",val);
	return 0;
}
```
