## 最小割
### 定理：最大流=最小割
必须满流的地方为瓶颈，限制了最大流的上限，也就是最小割。

最小割建模可以理解为花费最小使得 $S,T$ 不连通。每一条 $S\to T$ 的路径上至少要割一条边。

### [P3308 [SDOI2014]LIS](https://www.luogu.com.cn/problem/P3308)

接下来考虑如何让字典序最小。

#### 判断一条边可否为割边
在残量网络上，若 $u$ 无法到达 $v$，则 $E(u,v)$ 为割边。

#### 构造字典序最小的最小割
按字典序升序排序，从小到大钦定可为割的边为割边。
```cpp
#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int X=705,N=1405,M=5e5;
int T,n,en,a[X],tmp,tt,t[X],b[X],dp[X],tr[X],mx,cnt=1,h[N],no[N],nx[M],v[M],w[M],d[N],vis[N],tot,ch[X];
struct node{int c,id;}s[X];
ll ans;
void add(int k,int v){for(;k<=n;k+=k&-k) tr[k]=max(tr[k],v);}
int ask(int k){int re=0;for(;k;k-=k&-k) re=max(re,tr[k]);return re;}
void add(int x,int y,int W){nx[++cnt]=h[x],h[x]=cnt,v[cnt]=y,w[cnt]=W;}
void adde(int x,int y,int W){add(x,y,W),add(y,x,0);}
queue<int>q;
void bfs(){
	for(int i=0;i<=en;++i) d[i]=-1;
	d[0]=0,q.push(0);int u;
	while(!q.empty()){
		u=q.front(),q.pop();
		for(int i=h[u];i;i=nx[i])if(w[i]&&d[v[i]]==-1) d[v[i]]=d[u]+1,q.push(v[i]);
	}
}
ll dfs(int u,ll f){
	if(u==en) return f;
	for(int &i=no[u];i;i=nx[i])if(w[i]&&d[u]+1==d[v[i]]){
		ll ff=dfs(v[i],min((ll)w[i],f));
		if(ff) return w[i]-=ff,w[i^1]+=ff,ff;
	}
	return 0;
}
void dinic(){
	while(1){
		bfs();
		if(d[en]==-1) break;
		for(int i=0;i<=en;++i) no[i]=h[i];
		ll f;
		while((f=dfs(0,1e12))) ans+=f;
	}
}
bool cmp(node x,node y){return x.c<y.c;}
void dfs1(int u,int y){
	vis[u]=1;
	if(u==y) return ;
	for(int i=h[u];i;i=nx[i]){
		if(w[i]&&!vis[v[i]]) dfs1(v[i],y);
		if(vis[y]) return ;
	}
}
int check(int x,int y){
	for(int i=0;i<=en;++i) vis[i]=0;
	dfs1(x,y);
	return vis[y];
}
void solve(){
	scanf("%d",&n),en=n*2+1;
	for(int i=1;i<=n;++i) tr[i]=0;
	for(int i=1;i<=n;++i) ch[i]=0;
	for(int i=0;i<=en;++i) h[i]=0;
	mx=0,cnt=1,ans=0,tot=0;
	for(int i=1;i<=n;++i) scanf("%d",&a[i]),t[i]=a[i];
	for(int i=1;i<=n;++i) scanf("%d",&b[i]);
	for(int i=1;i<=n;++i) scanf("%d",&s[i].c),s[i].id=i;
	sort(t+1,t+1+n),tt=unique(t+1,t+1+n)-(t+1);
	for(int i=1;i<=n;++i){
		tmp=lower_bound(t+1,t+1+tt,a[i])-t;
		dp[i]=ask(tmp-1)+1,add(tmp,dp[i]),mx=max(mx,dp[i]);
	}
	for(int i=1;i<=n;++i) adde(i,i+n,b[i]);
	for(int i=1;i<=n;++i){
		if(dp[i]==1) adde(0,i,1e9);
		if(dp[i]==mx){adde(i+n,en,1e9);continue;}
		for(int j=i+1;j<=n;++j)if(a[i]<a[j]&&dp[j]==dp[i]+1) adde(i+n,j,1e9);
	}
	dinic(),printf("%lld ",ans);
	sort(s+1,s+1+n,cmp);
	for(int i=1;i<=n;++i){
		if(check(s[i].id,s[i].id+n)) continue;
		ch[s[i].id]=1,++tot;
		for(int j=2;j<=cnt;j+=2) w[j]+=w[j^1],w[j^1]=0;
		w[s[i].id<<1]=0,dinic();
	}
	printf("%d\n",tot);
	for(int i=1;i<=n;++i)if(ch[i]) printf("%d ",i);
	printf("\n");
}
int main(){
	scanf("%d",&T);
	while(T--) solve();
	return 0;
}
```

## 最大权闭合子图
### 概念
**闭合子图**：子图中每个点及其出边连接的点都在这幅图中，也就是说子图中的所有点的出边都是指向子图内部的。

**最大权闭合子图**：在所有的闭合子图中，它所包含的子图的点的点权之和是最大的。
### 应用与实现
经常在网络流建模中可以把问题转换为求最大权闭合子图，求解最大权闭合子图则可以通过 $总权值 -  最小割$ 实现。

**建图方式：**
-   对于所有原图中的边 $(u,v)$，连边 $u→v$，容量为 $INF$。
-   对于每个原图中的点 $u$，设 $u$ 的权值为 $val[u]$：
    1.  若 $val[u]>0$（正权点），连边 $S→u$，容量为 $val[u]$。
    2.  若 $val[u]<0$（负权点），连边 $u→T$，容量为 $−val[u]$。

### [CF103E Buying Sets](https://www.luogu.com.cn/problem/CF103E)
首先把价值取反，然后问题转换为求最大权值。

神仙最小割建模:

1. $S$ 连向集合，容量 $inf+v$ ，割掉这条边相当于不选这个集合；

2. 集合连向集合内的数字，容量 $INF$（$INF>>inf$，保证不会割掉集合连向集合内的数字的边）；

3. 数字连向 $T$，容量 $inf$，割掉这条边相当于选这个数字；

考虑为何这样建图可以满足 “选的集合个数 = 选的数字的个数” 的限制：

由于 “任意 $k$ 个子集的并的大小 $\ge k$”，根据 [[二分图#Hall 定理]]，选的集合个数 $\ge$ 选的数字的个数，割的边数 $\ge n$。

又因为边权加上了 $inf$，所以在跑最小割时能少割边就少割边，割的边数 $=n$。

由于 $不选的集合的个数 + 选的数字个数 = n$，可以得到 $选的集合个数 = 选的数字的个数$。

```cpp
		for(int i=h[u];i;i=nx[i])if(w[i]&&d[v[i]]==-1) d[v[i]]=d[u]+1,q.push(v[i]);
	}
}
int dfs(int u,int f){
	if(u==en) return f;
	for(int &i=no[u];i;i=nx[i])if(w[i]&&d[v[i]]==d[u]+1){
		int ff=dfs(v[i],min(w[i],f));
		if(ff) return w[i]-=ff,w[i^1]+=ff,ff;
	}
	return 0;
}
void dinic(){
	while(1){
		bfs();
		if(d[en]==-1) return ;
		for(int i=0;i<=en;++i) no[i]=h[i];
		int f;
		while((f=dfs(0,2e9))) ans-=f;
	}
}
int main(){
	scanf("%d",&n),en=n+n+1;
	for(int i=1;i<=n;++i){
		scanf("%d",&k);
		for(int j=1;j<=k;++j) scanf("%d",&x),adde(i,x+n,2e9);
	}
	for(int i=1;i<=n;++i)
		scanf("%d",&x),ans-=x,adde(0,i,1e9-x),adde(i+n,en,1e9);
	dinic(),printf("%lld",-(ans+1ll*inf*n));
	return 0;
}
```

### 做法
二分密度 $g$，判断 $\frac{N_e}{N_v}\ge g$，即 $N_e - g \times N_v\ge0$。

已知图中点数是 $n$，边数是 $m$，我们从源点 $S$ 向每一个点建一条容量为 $m$ 的边，原图内部的边容量为 $1$，再从每个点向汇点 $T$ 建一条容量为 $2g-d_v+m$ 的边。我们求出该流网络中的最小割，然后我们要求的 $N_e - g \times N_v$ 的最大值就是 $\frac {n\times m-f}{2}$。
