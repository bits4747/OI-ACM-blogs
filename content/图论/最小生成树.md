```cpp
#include<bits/stdc++.h>
using namespace std;
int n,m,f[50005],fx,fy,ans;
bool fl;
struct edge{
	int x,y,z;
}s[200005];
bool cmp(edge a,edge b){
	return a.z<b.z;
} 
int find(int x){
	if(f[x]==x) return x;
	return f[x]=find(f[x]);
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<=n;++i)  f[i]=i;
	for(int i=1;i<=m;++i) scanf("%d%d%d",&s[i].x,&s[i].y,&s[i].z);
	sort(s+1,s+1+m,cmp);
	for(int i=1;i<=m;++i){
		fx=find(s[i].x),fy=find(s[i].y);
		if(fx!=fy) f[fx]=f[fy],ans+=s[i].z;
	}
	fx=find(1);
	for(int i=2;i<=n;++i) if(find(i)!=fx){fl=1;break;}
	if(fl) printf("orz");
	else printf("%d",ans);
	return 0;
}
```

### [#10069. 「一本通 3.1 练习 4」Tree](https://loj.ac/p/10069)
优先加满白边 ×

影响黑边选择，有“后效性”。

kurskal按边权由小到大尝试，考虑白边数量不符：
1. 白边多了：不需要那么多，要让黑边优先一点。所以，尝试将百边边权加上$x$。
2. 白边少了：要让白边优先一点。所以，尝试将百边边权减上$x$。

综上，二分白边的增减值，求最小生成树。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=5e4+5,M=1e5+5;
int n,m,ne,f[N],fu,fv,l,r,mi,sum,wc,cnt,ans=5e6;
bool b[M];
struct edge{
	int u,v,d,c;
}s[M],t;
struct node{
	int to,dis;
}tmp;
vector<node>vec[N];
bool cmp(edge x,edge y){return x.d==y.d?x.c<y.c:x.d<y.d;}
int find(int x){return x==f[x]?x:f[x]=find(f[x]);}
bool check(){
	for(int i=1;i<=m;++i) if(!s[i].c) s[i].d+=mi;
	sort(s+1,s+m+1,cmp);
	for(int i=0;i<n;++i) f[i]=i;
	wc=cnt=sum=0;
	for(int i=1;i<=m;++i){
		fu=find(s[i].u),fv=find(s[i].v);
		if(fu!=fv){
			if(!s[i].c) ++wc;
			++cnt,f[fu]=fv,sum+=s[i].d;
		}
		if(cnt==n-1) break;
	}
	for(int i=1;i<=m;++i) if(!s[i].c) s[i].d-=mi;
	return wc>=ne;
}
int main(){
	scanf("%d%d%d",&n,&m,&ne);
	for(int i=1;i<=m;++i) scanf("%d%d%d%d",&s[i].u,&s[i].v,&s[i].d,&s[i].c);
	l=-111,r=111;
	while(l+1<r){
		mi=(l+r)>>1;
		if(check()) l=mi,ans=sum-mi*ne;//白边足够 
		else r=mi;
	}
	printf("%d",ans);
	return 0;
}
```
### [#10070. 「一本通 3.1 练习 5」最小生成树计数](https://loj.ac/p/10070)
- #### 定理：同一个图的每个最小生成树中，边权相等的边数量相等。

先跑一次最小生成树，记录各边权的所需数，再对每个边权暴力枚举选择方案，check一下是否合法，统计合法方案数 $sum$ ，用乘法原理更新 $ans$ 。

题解（转）：[click](https://hydingsy.github.io/articles/problem-JSOI-2008-MST-Count/)

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=105,M=1005,mod=31011;
int n,m,f[N],fu,fv,cnt,sum,ans=1;
bool b[M],B[M];
struct edge{
	int u,v,d;
}s[M];
bool cmp(edge x,edge y){return x.d<y.d;}
int find(int x){return x==f[x]?x:f[x]=find(f[x]);}
map<int,int>mp,h;
bool check(int l,int r){
	for(int i=1;i<=n;++i) f[i]=i;
	cnt=0;
	for(int i=1;i<=m;++i){
		if(l<=i&&i<=r&&!B[i]) continue;
		if((i<l||i>r)&&!b[i]) continue;
		fu=find(s[i].u),fv=find(s[i].v);
		if(fu!=fv) ++cnt,f[fu]=fv;
		else return 0;
		if(cnt==n-1) return 1;
	}
	return 0;
}
void dfs(int k,int su,int ne,int st,int ha){
	if(su==ne){sum+=check(st,st+ha-1);return ;}
	if(k>=st+ha) return;
	if(k+ne-su-1>=st+ha) return;
	dfs(k+1,su,ne,st,ha);
	B[k]=1;
	dfs(k+1,su+1,ne,st,ha);
	B[k]=0;
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<=n;++i) f[i]=i;
	for(int i=1;i<=m;++i) scanf("%d%d%d",&s[i].u,&s[i].v,&s[i].d),++h[s[i].d];
	sort(s+1,s+1+m,cmp);
	for(int i=1;i<=m;++i){
		fu=find(s[i].u),fv=find(s[i].v);
		if(fu!=fv) ++cnt,f[fu]=fv,++mp[s[i].d],b[i]=1;
		if(cnt==n-1) break;
	}
	for(int i=1;i<=m;i+=h[s[i].d]){
		sum=0,dfs(i,0,mp[s[i].d],i,h[s[i].d]);
		ans=ans*sum%mod;
	}
	printf("%d",ans);
	return 0;
}
```

## 次小生成树
**次小生成树和最小生成树只有一边之差**
### 非严格：
首先使用最小生成树算法将最小生成树求出来，将生成树建图，然后用维护树上的最大值（可用倍增）。 枚举所有的非树边并更新答案：`add=min(add,s[i].d-maxx[s[i].u][s[i].v]),ans+=add`

### 严格：
首先使用最小生成树算法将最小生成树求出来，将生成树建图，然后用维护树上的最大值以及严格次大值（可用倍增）。 枚举所有的非树边并更新答案。
#### [P4180 [BJWC2010] 严格次小生成树](https://www.luogu.com.cn/problem/P4180)


```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=1e5+5,M=3e5+5;
long long n,m,f[N],fu,fv,sum,fa[N][18],lg[N],dep[N],m1[N][18],m2[N][18],re,ma1,ma2,ad=1e9;
long long ans;
bool b[M];
struct edge{
	long long u,v,d;
}s[M],t;
struct node{
	long long to,dis;
}tmp;
vector<node>vec[N];
bool cmp(edge x,edge y){return x.d<y.d;}
long long find(long long x){return x==f[x]?x:f[x]=find(f[x]);}
void pdfs(long long k,long long ff,long long d){
	fa[k][0]=ff,dep[k]=d;
	for(int i=1;i<=lg[d];++i) fa[k][i]=fa[fa[k][i-1]][i-1];
	int len=vec[k].size();
	for(int i=0;i<len;++i){
		if(vec[k][i].to==ff) continue;
		m1[vec[k][i].to][0]=vec[k][i].dis;
		pdfs(vec[k][i].to,k,d+1);
	}
}
long long exc(long long x,long long a,long long b,long long c){
	re=-1;
	if(a!=x) re=a;
	if(b!=x&&b>re) re=b;
	if(c!=x&&c>re) re=c;
	return re;
}
void dfs(long long k,long long ff,long long d){
	for(int i=1;i<=lg[d];++i){
		m1[k][i]=max(m1[k][i-1],m1[fa[k][i-1]][i-1]);
		m2[k][i]=exc(m1[k][i],min(m1[k][i-1],m1[fa[k][i-1]][i-1]),m2[k][i-1],m2[fa[k][i-1]][i-1]);
	}
	int len=vec[k].size();
	for(int i=0;i<len;++i){
		if(vec[k][i].to==ff) continue;
		dfs(vec[k][i].to,k,d+1);
	}
}
void M1(long long x,long long y){
	ma1=-1;
	if(dep[x]<dep[y]) swap(x,y);
	while(dep[x]!=dep[y]) ma1=max(ma1,m1[x][lg[dep[x]-dep[y]]]),x=fa[x][lg[dep[x]-dep[y]]];
	if(x==y) return;
	for(int i=lg[dep[x]];i>=0;--i){//lg[dep[x]]!!! >=0!!!
		if(fa[x][i]!=fa[y][i])
			ma1=max(ma1,max(m1[x][i],m1[y][i])),x=fa[x][i],y=fa[y][i];
	}
	ma1=max(ma1,max(m1[x][0],m1[y][0]));
	return;
}
void M2(long long x,long long y){
	ma2=-1;
	if(dep[x]<dep[y]) swap(x,y);
	while(dep[x]!=dep[y]) ma2=exc(ma1,ma2,m1[x][lg[dep[x]-dep[y]]],m2[x][lg[dep[x]-dep[y]]]),x=fa[x][lg[dep[x]-dep[y]]];
	if(x==y) return;
	for(int i=lg[dep[x]];i>=0;--i){//lg[dep[x]]!!! >=0!!!
		if(fa[x][i]!=fa[y][i])
			ma2=exc(ma1,ma2,m1[x][i],exc(ma1,m1[y][i],m2[x][i],m2[y][i])),x=fa[x][i],y=fa[y][i];
	}
	ma2=exc(ma1,ma2,m1[x][0],exc(ma1,m1[y][0],m2[x][0],m2[y][0]));
	return;
}
int main(){
	scanf("%lld%lld",&n,&m);
	for(int i=1;i<=n;++i) f[i]=i;
	for(int i=1;i<=m;++i) scanf("%lld%lld%lld",&s[i].u,&s[i].v,&s[i].d);
	sort(s+1,s+1+m,cmp);
	for(int i=1;i<=m;++i){
		t=s[i];
		fu=find(t.u),fv=find(t.v);
		if(fu!=fv){
			f[fu]=fv,ans+=t.d,++sum,b[i]=1;
			tmp.dis=t.d,tmp.to=t.v,vec[t.u].push_back(tmp);
			tmp.to=t.u,vec[t.v].push_back(tmp);
		}
		if(sum==n-1) break;
	}
	lg[0]=-1;
	for(int i=2;i<=n;++i) lg[i]=lg[i>>1]+1;
	memset(m2,-1,sizeof(m2));
	pdfs(1,0,0);
	dfs(1,0,0);
	for(int i=1;i<=m;++i){
		if(b[i]||s[i].u==s[i].v) continue;
		M1(s[i].u,s[i].v);
		if(s[i].d>ma1) ad=min(ad,s[i].d-ma1);
		else{
			M2(s[i].u,s[i].v);
			if(s[i].d>ma2&&ma2!=-1) ad=min(ad,s[i].d-ma2);
		}
	}
	printf("%lld",ans+ad);
	return 0;
}
```

### [CF888G Xor-MST](https://www.luogu.com.cn/problem/CF888G)
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=2e5+5;
int n,a[N],f[N],rt[N],mn[N],id[N],k1,k2,c,sum,fl,re,cnt,ch[N*60][2],sz[N*60],b[N*60];
long long ans;
int F(int x){return x==f[x]?x:f[x]=F(f[x]);}
void ins(int &k,int d,int x){
	if(!k) k=++cnt,ch[k][0]=ch[k][1]=0,sz[k]=0,b[k]=0;
	++sz[k];
	if(d==-1){b[k]=x;return ;}
	ins(ch[k][(a[x]>>d)&1],d-1,x);
}
void solve(){
	cnt=re,fl=0;
	for(int i=1;i<=n;++i) rt[i]=0,mn[i]=2e9;
	for(int i=1;i<=n;++i) ins(rt[F(i)],30,i);
	for(int i=1;i<=n;++i){
		k1=rt[0],k2=rt[F(i)],sum=0;
		for(int j=30;j>=0;--j){
			c=(a[i]>>j)&1;
			if(ch[k1][c]&&sz[ch[k1][c]]-sz[ch[k2][c]]>0) k1=ch[k1][c],k2=ch[k2][c];
			else k1=ch[k1][c^1],k2=ch[k2][c^1],sum+=1<<j;
		}
		if(F(i)!=F(b[k1])&&sum<mn[F(i)]) mn[F(i)]=sum,id[F(i)]=F(b[k1]);
	}
	for(int i=1;i<=n;++i)if(mn[i]!=2e9&&F(i)!=F(id[i]))
		f[F(id[i])]=F(i),ans+=mn[i];
	for(int i=1;i<=n;++i) fl+=(F(i)==i); 
	if(fl==1) printf("%lld",ans),exit(0);
}
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i) scanf("%d",&a[i]);
	sort(a+1,a+1+n),n=unique(a+1,a+1+n)-(a+1);
	for(int i=1;i<=n;++i) f[i]=i,ins(rt[0],30,i);
	re=cnt;
	while(1) solve();
	return 0;
}
```
