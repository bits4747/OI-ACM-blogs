void dij(int st,int d[]){
	for(int i=1;i<=n;++i) d[i]=1e9;
	d[st]=0,q.push((qwq){st,0});
	while(!q.empty()){
		u=q.top(),q.pop();
		if(u.d!=d[u.x]) continue;
		for(int v:e[u.x])if(d[v]>max(a[v],u.d+1))
			d[v]=max(a[v],u.d+1),q.push((qwq){v,d[v]});
	}
}
