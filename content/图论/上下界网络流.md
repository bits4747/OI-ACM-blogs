#图论 #网络流 

## 无源汇上下界可行流
### [#115. 无源汇有上下界可行流](https://loj.ac/problem/115)
![[Pasted image 20230714114007.png]]
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=205,M=21000;
int n,m,x,y,p[M],qq,en,in[N],no[N],tot=1,h[N],nx[M],v[M],w[M],d[N],sum;
void add(int x,int y,int W){nx[++tot]=h[x],h[x]=tot,v[tot]=y,w[tot]=W;}
queue<int>q;
void bfs(){
	for(int i=0;i<=en;++i) d[i]=-1;
	d[0]=0,q.push(0);int u;
	while(!q.empty()){
		u=q.front(),q.pop();
		for(int i=h[u];i;i=nx[i])if(d[v[i]]==-1&&w[i]) d[v[i]]=d[u]+1,q.push(v[i]);
	}
}
int dfs(int u,int f){
	if(u==en) return f;
	for(int &i=no[u];i;i=nx[i])if(d[v[i]]==d[u]+1&&w[i]){
		int ff=dfs(v[i],min(w[i],f));
		if(ff) return w[i]-=ff,w[i^1]+=ff,ff;
	}
	return 0;
}
void dinic(){
	while(1){
		bfs();
		if(d[en]==-1) return ;
		for(int i=0;i<=en;++i) no[i]=h[i];
		int f;
		while((f=dfs(0,1e9))) sum-=f;
	}
}
int main(){
	scanf("%d%d",&n,&m),en=n+1;
	for(int i=1;i<=m;++i)
		scanf("%d%d%d%d",&x,&y,&p[i],&qq),in[x]-=p[i],in[y]+=p[i],add(x,y,qq-p[i]),add(y,x,0);
	for(int i=1;i<=n;++i){
		if(in[i]>0) add(0,i,in[i]),add(i,0,0),sum+=in[i];
		else add(i,en,-in[i]),add(en,i,0);
	}
	dinic();
	if(sum){printf("NO");return 0;}
	printf("YES\n");
	for(int i=1;i<=m;++i) printf("%d\n",p[i]+w[i*2+1]); 
	return 0;
}
```

## 有源汇上下界最大流
### [#116. 有源汇有上下界最大流](https://loj.ac/problem/116)
求出可行流后加上差网络上可向上浮动的流。

先在超级源超级汇上跑一次普通最大流，得到可行流。

再删去 $T$ 向 $S$ 连的流量 $inf$ 的边，再 $S,T$ 上跑普通最大流，得到可向上浮动的流。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=205,M=20500;
int n,m,x,y,s,t,pp,qq,l,r,st,en,in[N],no[N],tot=1,h[N],nx[M],v[M],w[M],d[N],sum,ans;
void add(int x,int y,int W){nx[++tot]=h[x],h[x]=tot,v[tot]=y,w[tot]=W;}
queue<int>q;
void bfs(){
	for(int i=l;i<=r;++i) d[i]=-1;
	d[st]=0,q.push(st);int u;
	while(!q.empty()){
		u=q.front(),q.pop();
		for(int i=h[u];i;i=nx[i])if(d[v[i]]==-1&&w[i]) d[v[i]]=d[u]+1,q.push(v[i]);
	}
}
int dfs(int u,int f){
	if(u==en) return f;
	for(int &i=no[u];i;i=nx[i])if(d[v[i]]==d[u]+1&&w[i]){
		int ff=dfs(v[i],min(w[i],f));
		if(ff) return w[i]-=ff,w[i^1]+=ff,ff;
	}
	return 0;
}
void dinic(){
	while(1){
		bfs();
		if(d[en]==-1) return ;
		for(int i=l;i<=r;++i) no[i]=h[i];
		int f;
		while((f=dfs(st,1e9))) ans+=f;
	}
}
int main(){
	scanf("%d%d%d%d",&n,&m,&s,&t),en=r=n+1;
	for(int i=1;i<=m;++i)
		scanf("%d%d%d%d",&x,&y,&pp,&qq),in[x]-=pp,in[y]+=pp,add(x,y,qq-pp),add(y,x,0);
	for(int i=1;i<=n;++i){
		if(in[i]>0) add(0,i,in[i]),add(i,0,0),sum+=in[i];
		else add(i,en,-in[i]),add(en,i,0);
	}
	add(t,s,1e9),add(s,t,0),dinic();
	if(ans!=sum){printf("please go home to sleep");return 0;}
	l=1,r=n,st=s,en=t,ans=w[tot],w[tot]=w[tot^1]=0,dinic();
	printf("%d",ans);
	return 0;
}
```
## 有源汇上下界最小流
### [#117. 有源汇有上下界最小流](https://loj.ac/problem/117)
