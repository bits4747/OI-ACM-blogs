#图论 #点分治

[部分转载：浅谈树分治](https://blog.csdn.net/qq_38944163/article/details/106153490)
## 简介
### 用处：用于解决某类树上路径的问题的高效算法

暴力做树上路径问题的一般就是直接枚举两个点，再求出lca来计算这条路径，但这么做复杂度太高，为 $n^2logn$ 。

而点分治可以大大优化其复杂度，通常可达到 $nlogn$ 或 $nlog^2n$ 。

- #### 算法核心思想： 
每次钦定一个点作为根，统计经过这个点的路径的贡献（两端都在子树内），然后这个点各个儿子子树之前不再有贡献，一路分治下去即可（儿子树的互相独立）。

- #### 分治
在树上钦定的点要保证删掉这个点后最大的连通块节点个数最小(使最大的子树最小)，这个所选的点就是树的重心。

- #### 答案统计
1. 每次处理当前子树和先前子树之间的贡献，再将当前子树加入到先前子树。
1. 运用容斥，计算当前点的贡献再减去其子树的贡献。

通常两种写法可以互换，但有时一种会比另一种更易实现，具体视实际情况而定 ~~（凭直觉）~~ 。

```cpp
#include<bits/stdc++.h>
using namespace std;
int n,m,k[105],be,en,siz[10005],minn,zx,ma[10005],dis[10005],jl[100005],njl[10005],sum,ns,zs;
bool b[10000005],ans[105],vis[10005];
struct node{
	int to,val;
}t;
vector<node>vec[10005];
void frt(int u,int fa){
	int len=vec[u].size();
	siz[u]=1,ma[u]=0;
	for(int i=0;i<len;++i){
		int tmp=vec[u][i].to;
		if(tmp==fa||vis[tmp]) continue;//
		frt(tmp,u);
		siz[u]+=siz[tmp];
		ma[u]=max(ma[u],siz[tmp]);
	}
	ma[u]=max(ma[u],zs-siz[u]);//是整颗子树-ma[u],而非siz[u]-ma[u]
	if(ma[zx]>ma[u]) zx=u;
}
void predis(int u,int fa){
	if(dis[u]>1e7) return ;
	njl[++ns]=dis[u];
	int len=vec[u].size();
	for(int i=0;i<len;++i){
		int tmp=vec[u][i].to;
		if(tmp==fa||vis[tmp]) continue;//
		dis[tmp]=dis[u]+vec[u][i].val;
		predis(tmp,u);
	}
}
void calc(int u){
	int len=vec[u].size();sum=0;
	for(int i=0;i<len;++i){
		int tmp=vec[u][i].to;
		if(vis[tmp]) continue;//
		dis[tmp]=vec[u][i].val,ns=0;
		predis(tmp,u);
		for(int j=1;j<=ns;++j)
			for(int o=1;o<=m;++o)
				if(njl[j]<=k[o]&&b[k[o]-njl[j]])
					ans[o]=1;
		for(int j=1;j<=ns;++j) 
			if(njl[j]<=1e7)
				jl[++sum]=njl[j],b[njl[j]]=1;
	}
	for(int i=1;i<=sum;++i) b[jl[i]]=0;
}
void solve(int u){
	b[0]=1,vis[u]=1;//
	calc(u);
	int len=vec[u].size();
	for(int i=0;i<len;++i){
		int tmp=vec[u][i].to;
		if(vis[tmp]) continue;//
		zx=0;
		zs=siz[tmp]>siz[u]?(siz[tmp]-siz[u]):siz[tmp];//!!!!!!!!!
		frt(tmp,0);
		solve(zx);
	}
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<n;++i){
		scanf("%d%d%d",&be,&en,&t.val);
		t.to=en,vec[be].push_back(t);
		t.to=be,vec[en].push_back(t);
	}
	for(int i=1;i<=m;++i) scanf("%d",&k[i]);
	ma[0]=n,zs=n;
	frt(1,0);
	solve(zx);
	for(int i=1;i<=m;++i){
		if(ans[i]) printf("AYE\n");
		else printf("NAY\n");
	}
	return 0;
}
```

```cpp
int n,x,y,SIZ,rt,siz[N],ma[N],b[3],nb[3],ans,dis[N],g;

struct node{
	int to,d;
}t;

void findrt(int u,int fa){
	int len=vec[u].size();
	siz[u]=1,ma[u]=0;
	for(int i=0;i<len;++i){
		int v=vec[u][i].to;
		if(fa==v||vis[v]) continue;
		findrt(v,u);
		siz[u]+=siz[v],ma[u]=max(ma[u],siz[v]);
	}
	ma[u]=max(ma[u],SIZ-siz[u]);
	if(ma[u]<ma[rt]) rt=u;
}
void predis(int u,int fa){
	++nb[dis[u]];
	int len=vec[u].size();
	for(int i=0;i<len;++i){
		int v=vec[u][i].to;
		if(fa==v||vis[v]) continue;
		dis[v]=(dis[u]+vec[u][i].d)%3;
		predis(v,u);
	}
}
void calc(int u){
	int len=vec[u].size();
	b[0]=b[1]=b[2]=0;
	for(int i=0;i<len;++i){
		int v=vec[u][i].to;
		if(vis[v]) continue;
		nb[0]=nb[1]=nb[2]=0,dis[v]=vec[u][i].d%3;
		predis(v,u);
		ans+=b[0]*nb[0]*2+b[1]*nb[2]*2+b[2]*nb[1]*2+nb[0]*2;
		b[0]+=nb[0],b[1]+=nb[1],b[2]+=nb[2];
	}
}
void solve(int u){
	vis[u]=1;
	calc(u);
	int len=vec[u].size();
	for(int i=0;i<len;++i){
		int v=vec[u][i].to;
		if(vis[v]) continue;
		rt=0,SIZ=siz[v];//
		findrt(v,u);
		solve(rt);
	}
}
int gcd(int a,int b){return a%b?gcd(b,a%b):b;}
int main(){
	scanf("%d",&n);
	for(int i=1;i<n;++i){
		scanf("%d%d%d",&x,&y,&t.d);
		t.to=y,vec[x].push_back(t);
		t.to=x,vec[y].push_back(t);
	}
	SIZ=ma[0]=n;
	findrt(1,0);
	solve(rt);
	ans+=n,n*=n;
	g=gcd(ans,n);
	printf("%d/%d",ans/g,n/g);
	return 0;
}
```

### [P4178 Tree](https://www.luogu.com.cn/problem/P4178)

### [P5306 [COCI2018-2019#5] Transport](https://www.luogu.com.cn/problem/P5306) <font color="red">★</font>
维护当前点 $u$ 的上行下行路径，记录下行路径所需起始油量 $dn[i]$ ，及合法的上行路径的剩余油量 $up[i]$ 。

预处理完后就变成上一题了。

但由于 $dn[i]$ 有负值，且范围较大，本题将 $up[i]$ 和 $dn[i]$ 排序后用**双指针**求出合法方案。

- 重点：如何处理出 $up[i]$ 和 $dn[i]$ 
1. $dn[i]$ ：直接 dfs 记录所需初始油量。
1. $up[i]$ ：设置一个 $fl$ ，从 $rt$ 开始，每次进入儿子的时候，更新 $fl$ 为 $min(0,-vec[u][i].d+val[v]))$。当 $fl=0$ ，记录 $up[++numu]$ 为 $rt-->u$ 的剩余油量。（感性理解一下）

本题用**容斥**写法。

核心部分：
```cpp
void predis(int u,int fa,long long dis,long long pre,long long fl){
	int len=vec[u].size();
	dis+=val[u];
	if(fl==0) up[++numu]=dis;
	for(int i=0;i<len;++i){
		int v=vec[u][i].to;
		if(fa==v||vis[v]) continue;
		dn[++numd]=min(dis-vec[u][i].d,pre);
		predis(v,u,dis-vec[u][i].d,dn[numd],min((long long)0,fl-vec[u][i].d+val[v]));
	}
}
void calc(int u){
	int len=vec[u].size();
	numu=numd=0;
	for(int i=0;i<len;++i){
		int v=vec[u][i].to;
		if(vis[v]) continue;
		pnu=numu,pnd=numd;
		dn[++numd]=val[u]-vec[u][i].d;
		predis(v,u,val[u]-vec[u][i].d,dn[numd],min((long long)0,-vec[u][i].d+val[v]));
		sort(dn+pnd+1,dn+1+numd);
		sort(up+pnu+1,up+1+numu);
		k=numu;
		for(int j=pnd+1;j<=numd;++j){
			while(k>pnu&&up[k]+dn[j]-val[u]>=0) --k;
			ans-=(numu-k);
		}
	}
	sort(dn+1,dn+1+numd);
	sort(up+1,up+1+numu);
	k=numu;
	for(int j=1;j<=numd;++j){
		while(k>=1&&up[k]+dn[j]-val[u]>=0) --k;
		ans+=(numu-k);
		if(dn[j]>=0) ++ans;
	}
	ans+=numu;
}
```
### [P5351 Ruri Loves Maschera](https://www.luogu.com.cn/problem/P5351)
定义路径魔力值为路径上最大的边权，求长度为 $[l,r]$ 的路径的魔力值和。

神似前两题，考虑用指针或树状数组。

1. ![[Pasted image 20230926192104.png]]
2. 把所有链**按照其魔力值排序**，然后从小到大枚举。因为对每条链，设其魔力值为 $w$ ，**它与其之前枚举到的链产生的贡献均为 $w$** ，所以可用**树状数组**来维护 $len$ ，然后对每条链在树状数组上查询之前有多少满足 $len$ 限制的链即可，再把这条链放进树状数组里。

本题用**容斥**写法。

核心部分：
```cpp
void predis(int u,int fa){
	if(dep[u]>r) return ;
	a[++ta].len=dep[u],a[ta].mb=dis[u];
	int len=vec[u].size();
	for(int i=0;i<len;++i){
		int v=vec[u][i].to;
		if(v==fa||vis[v]) continue;
		dis[v]=max(dis[u],vec[u][i].val),dep[v]=dep[u]+1;
		predis(v,u);
	}
}
void calc(int u){
	int len=vec[u].size();ta=0;
	for(int i=0;i<len;++i){
		int v=vec[u][i].to;
		if(vis[v]) continue;
		dis[v]=vec[u][i].val,dep[v]=1,he=ta+1;
		predis(v,u);
		sort(a+he,a+ta+1,cmp);
		for(int j=he;j<=ta;++j){
			ans-=a[j].mb*(ask(r-a[j].len)-ask(l-a[j].len-1));
			add(a[j].len,1);
		}
		for(int j=he;j<=ta;++j) add(a[j].len,-1); 
	}
	sort(a+1,a+ta+1,cmp);
	for(int i=1;i<=ta;++i){
		ans+=a[i].mb*(ask(r-a[i].len)-ask(l-a[i].len-1));
		if(l<=a[i].len&&a[i].len<=r) ans+=a[i].mb;
		add(a[i].len,1);
	}
	for(int i=1;i<=ta;++i) add(a[i].len,-1);
}
```

### [CF293E Close Vertices](https://www.luogu.com.cn/problem/CF293E)
求有多少对点，满足 $i→j$ 的长度不超过 $l$ ， 边权不超过 $w  (1<=l<=10^{5},0<=w<=10^{9})$ 。

若只有一维限制，排序后用双指针或树状数组即可。考虑再添加一维，就需要**双管齐下**了。

关注数据范围，可以按 $w$ 排序并用**双指针**，用**树状数组**处理 $l$ 。

本题用**容斥**写法。

核心部分：
```cpp
bool cmp(edge x,edge y){return x.b<y.b;} 
int lowbit(int x) {return x&-x;}
void add(int x,int num){for(;x<=l;x+=lowbit(x)) bo[x]+=num;}
int ask(int x){int sum=0;for(;x>0;x-=lowbit(x)) sum+=bo[x];return sum;}
void predis(int u,int fa){
	if(dep[u]>l||dis[u]>w) return ;
	a[++ta].len=dep[u],a[ta].b=dis[u];
	int len=vec[u].size();
	for(int i=0;i<len;++i){
		int v=vec[u][i].to;
		if(v==fa||vis[v]) continue;
		dis[v]=dis[u]+vec[u][i].val,dep[v]=dep[u]+1;
		predis(v,u);
	}
}
void calc(int u){
	int len=vec[u].size();ta=0;
	for(int i=0;i<len;++i){
		int v=vec[u][i].to;
		if(vis[v]) continue;
		dis[v]=vec[u][i].val,dep[v]=1,he=ta+1;
		predis(v,u);
		sort(a+he,a+ta+1,cmp);
		ll=he,rr=ta;
		for(int j=he;j<=ta;++j) add(a[j].len,1);
		while(ll<rr){
			if(a[ll].b+a[rr].b<=w)
				add(a[ll].len,-1),ans-=ask(l-a[ll].len),++ll;
			else add(a[rr].len,-1),--rr;
		}
		for(int j=ll;j<=rr;++j) add(a[j].len,-1);
	}
	ll=he=1,rr=ta;
	sort(a+he,a+ta+1,cmp);
	for(int j=he;j<=ta;++j) add(a[j].len,1);
	while(ll<rr){
		if(a[ll].b+a[rr].b<=w)
			add(a[ll].len,-1),ans+=ask(l-a[ll].len),++ll;
		else add(a[rr].len,-1),--rr;
	}
	for(int j=ll;j<=rr;++j) add(a[j].len,-1);
	ans+=ta;
}
```

### [P3714 [BJOI2017]树的难题](https://www.luogu.com.cn/problem/P3714)
- 代码极其难打的**毒瘤**题。

把儿子按颜色排序，就能使相同颜色的靠在一起。

维护两颗**线段树**处理相同及不同颜色的贡献。

核心部分：
```cpp
struct tree{
	struct node{int val,fl;}s[N*3];
	void del(){s[1].fl=1,s[1].val=inf;}
	void pushdown(int u){
		if(s[u].fl) s[u].fl=0,s[u<<1].fl=s[u<<1|1].fl=1,s[u<<1].val=s[u<<1|1].val=inf;
	}
	void change(int u,int ll,int rr,int p,int w){
		if(ll>p||rr<p) return ;
		if(ll==rr){
			s[u].fl=0,s[u].val=max(s[u].val,w);
			return;
		}
		int mid=(ll+rr)>>1;
		pushdown(u);
		if(mid>=p) change(u<<1,ll,mid,p,w);
		else change(u<<1|1,mid+1,rr,p,w);
		s[u].val=max(s[u].val,max(s[u<<1].val,s[u<<1|1].val));
	}
	int ask(int u,int ll,int rr,int fr,int en){
		if(ll>en||rr<fr) return inf;
		if(fr<=ll&&rr<=en) return s[u].val;
		if(s[u].fl) return inf;
		
		pushdown(u);
		int mid=(ll+rr)>>1;
		return max(ask(u<<1,ll,mid,fr,en),ask(u<<1|1,mid+1,rr,fr,en));
	}
}tr,ttr;
bool cmp(edge a,edge b){return a.ty<b.ty;}
void predis(int u,int fa,int dis,int dep,int lty){
	md[dep]=max(md[dep],dis),mnl=max(mnl,dep);
	if(dep>=r) return;
	int len=vec[u].size();
	for(int i=0;i<len;++i){
		int v=vec[u][i].to;
		if(vis[v]||v==fa) continue;
		predis(v,u,dis+(vec[u][i].ty==lty?0:c[vec[u][i].ty]),dep+1,vec[u][i].ty);
	}
}
void calc(int u){
	int len=vec[u].size();
	la=-1;
	tr.del(),ttr.del();
	sort(vec[u].begin(),vec[u].end(),cmp);
	for(int i=0;i<len;++i){
		int v=vec[u][i].to;
		if(vis[v]) continue;
		predis(v,u,c[vec[u][i].ty],1,vec[u][i].ty);
		if(la==vec[u][i].ty){
			for(int j=1;j<=mnl;++j)
				ans=max(ans,md[j]+ttr.ask(1,0,r,max(0,l-j),r-j)-c[vec[u][i].ty]);
			for(int j=1;j<=mnl;++j) ttr.change(1,0,r,j,md[j]),tmd[j]=max(tmd[j],md[j]),md[j]=inf;
			mttl=max(mttl,mnl),mnl=0;
		}
		else{
			for(int j=1;j<=mttl;++j){
				ans=max(ans,tmd[j]+tr.ask(1,0,r,max(0,l-j),r-j));
				if(j>=l) ans=max(ans,tmd[j]);
			}
			for(int j=1;j<=mttl;++j) tr.change(1,0,r,j,tmd[j]),tmd[j]=inf;
			mtl=max(mttl,mtl),mttl=0,ttr.del();
			for(int j=1;j<=mnl;++j) ttr.change(1,0,r,j,md[j]),tmd[j]=max(tmd[j],md[j]),md[j]=inf;
			mttl=max(mttl,mnl),mnl=0;
		}
		la=vec[u][i].ty;
	}
	for(int j=1;j<=mttl;++j){
		ans=max(ans,tmd[j]+tr.ask(1,0,r,max(0,l-j),r-j));
		if(j>=l) ans=max(ans,tmd[j]);
	}
	for(int j=1;j<=mttl;++j) tmd[j]=inf;
```

### [P2664 树上游戏](https://www.luogu.com.cn/problem/P2664)
与上题相反，本题思维难度大，但代码难度还成。

记 $cnt[i]$ 为它的所有子树内到他的路径中包含颜色 $i$ 的路径条数，那么当处理到一颗子树内的时候，我们统计得到其他子树贡献的所有 $cnt$ 数组。对于该子树内的一个点 $x$ ，先不考虑它到根节点的路径上的颜色，则其它子树内的颜色对当前节点 $x$ 的贡献为 $∑cnt[i]$ 。

考虑如何处理到根节点的路径上的颜色对答案统计的影响（不好描述，见代码，看不懂点题解）。

```cpp
//cnt[i]为出现颜色i的链数，siz[i]为子树i的大小 
void getsize(int u,int fa){
	int len=vec[u].size();
	siz[u]=1;
	for(int i=0;i<len;++i){
		int v=vec[u][i];
		if(fa==v||vis[v]) continue;
		getsize(v,u),siz[u]+=siz[v];
	}
}
void add(int u,int fa){
	if(!b[c[u]]) cnt[c[u]]+=siz[u],sum+=siz[u];//颜色c[u]未出现，更新 
	if(!B[c[u]]) B[c[u]]=1,ap[++num]=c[u];//记录出现了什么颜色，方便清零 
	int len=vec[u].size();
	++b[c[u]];//标记颜色c[u]已出现 
	for(int i=len-1;i>=0;--i){
		int v=vec[u][i];
		if(fa==v||vis[v]) continue;
		add(v,u);
	}
	--b[c[u]];//清除 
}
void del(int u,int fa){//类似add 
	if(!b[c[u]]) cnt[c[u]]-=siz[u],sum-=siz[u];
	int len=vec[u].size();
	++b[c[u]];
	for(int i=len-1;i>=0;--i){
		int v=vec[u][i];
		if(fa==v||vis[v]) continue;
		del(v,u);
	}
	--b[c[u]];
}
void upd(int u,int fa,int fir,int ins){
	//fir为当前子树到当前点为止出现的颜色数
	//ins为目当前子树到当前点为止出现的颜色在其余子树中出现的路径数 
	if(!b[c[u]]) ins+=cnt[c[u]],++fir;
	int len=vec[u].size();
	ans[u]+=sum-ins+oth*fir;//统计已当前点为起点的贡献 
	++b[c[u]]; 
	for(int i=len-1;i>=0;--i){
		int v=vec[u][i];
		if(fa==v||vis[v]) continue;
		upd(v,u,fir,ins);
	}
	--b[c[u]];
}
void calc(int u){
	num=sum=0;
	getsize(u,0);
	add(u,0);
	ans[u]+=sum-cnt[c[u]]+siz[u];//统计c[u]对答案的贡献 
	int len=vec[u].size();
	for(int i=len-1;i>=0;--i){//一棵一棵子树地处理 
		int v=vec[u][i];
		if(vis[v]) continue;
		++b[c[u]],del(v,u);				//删去当期子树出现颜色的贡献，
		sum-=siz[v],cnt[c[u]]-=siz[v];	//但不删去c[u]的贡献，因为已统一处理 
		oth=siz[u]-siz[v],upd(v,u,0,0);//计算当前子树与其余子树之间的贡献 
		add(v,u),--b[c[u]],sum+=siz[v],cnt[c[u]]+=siz[v];//加回来 
	}
	for(int i=1;i<=num;++i) B[ap[i]]=0,cnt[ap[i]]=0;
}
```

### [P4183 [USACO18JAN] Cow at Large P](https://www.luogu.com.cn/problem/P4183)
考虑一次询问，把贝西的位置当作根。

一个叶子节点 $x$ 的农民最多可控制 $x$ 的 $\lceil \frac {dis(rt,x)}2 \rceil$ 级祖先的子树。要使农民最少，则当一个叶子节点可控制的子树的父亲可被其它叶子节点控制时，不选此节点。

记 $u$ 与最近的叶子节点的距离为 $l_u$，若要选择一个子树 $u$，当且仅当 $l_u\le dis(u,rt)$ 且边数等于点数（一条边连接 $u$ 和原树），即 $\sum 2-deg_i=1$。

点分治用树状数组容斥统计 $ans_u=\sum 2-deg_i,l_i\le dis(u,i)$。
```cpp
void dfs(int u,int fa){
	q[++t]=u;
	for(int v:e[u])if(!vis[v]&&v!=fa) d[v]=d[u]+1,dfs(v,u);
}
void add(int x,int v){for(;x<=n+n;x+=x&-x) s[x]+=v;}
int ask(int x){int re=0;for(;x;x-=x&-x) re+=s[x];return re;}
void calc(int ty){
	for(int i=1;i<=t;++i) add(l[q[i]]-d[q[i]]+n,de[q[i]]);
	for(int i=1;i<=t;++i) ans[q[i]]+=ty*ask(d[q[i]]+n);
	for(int i=1;i<=t;++i) add(l[q[i]]-d[q[i]]+n,-de[q[i]]);
}
void solve(int u){
	vis[u]=1,d[u]=0,t=0,dfs(u,u),calc(1);
	for(int v:e[u])if(!vis[v]) t=0,dfs(v,u),calc(-1);
	int tot=SZ;
	for(int v:e[u])if(!vis[v])
		SZ=mx=(sz[v]>sz[u]?tot-sz[u]:sz[v]),grt(v,v),solve(rt);
}
```
