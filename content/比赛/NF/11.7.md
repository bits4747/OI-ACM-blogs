## [A. 区间](http://www.nfls.com.cn:10611/contest/1054/problem/1)
**题意：**

给定 $n(n\le4e5)$ 个区间，选出尽可能多对区间，满足每对区间都不交，一个区间只能被选一次。 

**题解：**

将所有区间按 $l$ 从小到大排序依次加入，当前加入区间 $[l_i,r_i]$：
1. 存在未匹配的区间 $j$ 且 $r_j<l_i)$，由于之后的区间的 $l$ 都不会小于 $l_i$ ，我们随便选择一个区间跟当前区间匹配即可。
2. 否则，我们在已经匹配的区间对 $(x,y)$ 中找到 $r_y$ 最小的区间，若 $r_y<r_i$，则交换区间 $y,i$。

维护两个 set 即可。

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=4e5+5;
int n,ans;
struct qwq{int l,r;}s[N];
multiset<int>q1,q2;
multiset<int>::iterator it; 
int main(){
	freopen("a.in","r",stdin);
	freopen("a.out","w",stdout);
	scanf("%d",&n);
	for(int i=1;i<=n;++i) scanf("%d%d",&s[i].l,&s[i].r);
	sort(s+1,s+1+n,[](qwq a,qwq b){return a.l<b.l;});
	for(int i=1;i<=n;++i){
		if((it=q1.lower_bound(s[i].l))!=q1.begin())
			++ans,q1.erase(--it),q2.insert(s[i].r);
		else
			if(!q2.empty()&&*(it=q2.begin())<s[i].r)
				q1.insert(*it),q2.erase(it),q2.insert(s[i].r);
			else q1.insert(s[i].r);
	}
	printf("%d",ans);
	return 0;
}
```
## [B. 排序](http://www.nfls.com.cn:10611/contest/1054/problem/2)
原题：[P9291 [ROI 2018] Quick sort](https://www.luogu.com.cn/problem/P9291)

**题意：**

给一个 $n\le3000$ 的排列 $a_n$，定义操作 $S(l,r)$ 表示 $[a_l,a_{l+1},\ldots,a_r]$ 重排成 $[a_{l+1},a_{l+3},\ldots,a_l,a_{l+2},\ldots]$。

求经过多少次操作能使排列递增，并输出任意一组方案，不要求方案的操作次数最少，但要求操作次数 $\leq 6000$。

**题解：**

首先，最基本的冒泡排序操作 $n^2$ 次肯定行。

进一步，$i\to j$，设 $D=dis(i,j)$，每次可移动 $\log D$，可以做到 $n\log n$。

我们考虑倒逆操作，指将 $[a_{l+1},a_{l+3},\ldots,a_l,a_{l+2},\ldots]$ 重排成 $[a_l,a_{l+1},\ldots,a_r]$。我们初始另 $a_i=i$，要将 $i$ 挪到 $a_i$ 处。

即对原排列的逆排列排序，且使用原操作的逆操作，最后将操作序列倒序输出。

![](Pasted%20image%2020231108090002.png)

```cpp
#include<bits/stdc++.h>
using namespace std;
#define pii pair<int,int>
const int N=3005;
int n,a[N],x,p[N],t[N],l,r;
vector<pii>s;
mt19937 R(time(0));
void rev(int l,int r){
	s.push_back(make_pair(l,r));int j=l;
	for(int i=l+1;i<=r;i+=2,++j) t[i]=p[j];
	for(int i=l;i<=r;i+=2,++j) t[i]=p[j];
	for(int i=l;i<=r;++i) p[i]=t[i];
}
void work(){
	s.clear();
	for(int i=1;i<=n;++i) p[a[i]]=i;
	for(int i=1;i<=47;++i){
		l=R()%n+1,r=R()%n+1;
		if(l>r) swap(l,r);
		rev(l,r);
	}
	for(int i=n;i;--i){
		x=i;
		while(p[x]^i) --x;
		while(x*2<=i) rev(1,2*x),x*=2;
		if(x^i) rev(x*2-i+1,i);
	}
	if(s.size()>6000) return ;
	printf("%d\n",(int)s.size());
	for(int i=(int)s.size()-1;~i;--i) printf("%d %d\n",s[i].first,s[i].second);
	exit(0);
}
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i) scanf("%d",&a[i]);
	while(1) work();
	return 0;
}
```
## [div-2 C. 分组](http://www.nfls.com.cn:10611/contest/1055/problem/3)
原题：[P5861 [IOI2015] teams](https://www.luogu.com.cn/problem/P5861)

**题意：**

$N$ 个学生，第 $i$ 个学生只愿意在小组规模为 $[A_i,B_i]$ 的小组工作。

$Q$ 次询问，每次给定组数 $M$，和每组人数 $K_i$，问能否将分成 $M$ 组。

$1\le N\le 5 \times 10^5$，$1\le Q\le 2 \times 10^5$， $\sum M \leq 2\times 10^5$。

**题解：**

贪心地想，对于每组询问， 我们从小到大考虑 $K$ ，那么每次在 $A_i\le K$ 的 $B$ 中选取最小的 $K$ 个 $B$。

我们把 $(A_i,B_i)$ 转化为二维平面上的点，对于每个 $K$，可以选择 $(K,K)$ 左上角的所有点。

![](Pasted%20image%2020231108112508.png)

我们按 $K$ 把原图划分为一个个横坐标为 $(K_{i-1},K_i]$ 的矩形。

![](Pasted%20image%2020231108113435.png)

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=5e5+5,M=2e5+5;
int n,x,y,cnt,rt[N],Q,m,a[M],t,q[M],H,h[M],sum,res[M];
vector<int>e[N];
struct seg{int l,r,x;}s[N*20];
void add(int &k,int l,int r,int x){
	s[++cnt]=s[k],k=cnt,++s[k].x;
	if(l==r) return;
	int mi=(l+r)>>1;
	if(x<=mi) add(s[k].l,l,mi,x);
	else add(s[k].r,mi+1,r,x);
}
int ask(int k1,int k2,int l,int r,int x){
	if(l==r) return s[k2].x-s[k1].x;
	int mi=(l+r)>>1;
	if(x<=mi) return s[s[k2].r].x-s[s[k1].r].x+ask(s[k1].l,s[k2].l,l,mi,x);
	return ask(s[k1].r,s[k2].r,mi+1,r,x);
}
int kth(int k1,int k2,int l,int r,int x){
	if(l==r) return l;
	int mi=(l+r)>>1;
	if(s[s[k2].r].x-s[s[k1].r].x>=x) return kth(s[k1].r,s[k2].r,mi+1,r,x);
	return kth(s[k1].l,s[k2].l,l,mi,x-(s[s[k2].r].x-s[s[k1].r].x));
}
int work(){
	scanf("%d",&m),t=0;
	for(int i=1;i<=m;++i) scanf("%d",&a[i]);
	sort(a+1,a+1+m);
	for(int i=1;i<=m;++i){
		while(t&&h[t]<a[i]) --t;
		sum=res[t]+ask(rt[q[t]],rt[a[i]],1,n,a[i])-a[i];
		if(sum<0) return 0;
		while((H=kth(rt[q[t]],rt[a[i]],1,n,sum-res[t]))>h[t]&&t) --t;
		q[++t]=a[i],h[t]=H,res[t]=sum;
	}
	return 1;
}
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i) scanf("%d%d",&x,&y),e[x].push_back(y);
	for(int i=1;i<=n;++i){
		rt[i]=rt[i-1];
		for(int j:e[i]) add(rt[i],1,n,j);
	}
	scanf("%d",&Q);
	while(Q--) printf("%d\n",work());
	return 0;
}
```
## [C. 路过中丹](http://www.nfls.com.cn:10611/contest/1054/problem/3)
**题意：**

定义一次行走为，从 $x$ 出发，每次向左或向右走一步，走到 $y,x\ne y$ 停止，一个位置可以重复经过。

一个字符串是合法点要求可以通过若干次行走，使得所有位置至少经过一次。

给定字符串 $S,|S|\le1e6$，有 $Q\le1e6$ 次询问，每次询问判断子串 $[L,R]$ 是否合法。

**题解：**

- 若一个子串可以被若干个偶回文串恰好覆盖（边边的偶回文没有越出 $L,R$），则合法。
- 若存在奇回文（判断是否存在距离为 $2$ 的相同字母，如 `aba` 即可）也合法。
  ![](Pasted%20image%2020231108192709.png)

通过马拉车求出偶回文的长度，用**并查集跑区间覆盖**（详见代码）算出每个位置前后一个偶回文的中心（记靠左的为中心）$l_i,r_i$。

对于位置 $i$，当 $L\le l_i*2-i+1$ 或 $r_i*2-i+1\le R$ 时，$i$ 再区间 $[L<R]$ 内被偶回文覆盖。

把询问按 $L$ 离线，在 $l_i*2-i+1$ 处 $i$ 插入线段 $[i+1,r_i*2-i+1]$（$L>l_i*2-i+1$，左边的偶回文越界，必须用右边的偶回文覆盖），可用树状数组维护。

若 $R$ 被覆盖说明部分位置右边的覆盖到它的最小偶回文越界，子串 $[L,R]$ 不合法；反之。

```cpp
#include<bits/stdc++.h>
using namespace std;
#define pii pair<int,int>
const int N=1e6+5;
char c[N];
int n,p,R,a[N],z[N],f[N],l[N],r[N],q,x,y,tr[N],ans[N];
vector<pii>e[N];
vector<int>sl[N];
int F(int x){return x==f[x]?x:f[x]=F(f[x]);}
void add(int k,int v){for(;k<=n;k+=k&-k) tr[k]+=v;}
int ask(int k){int re=0;for(;k;k-=k&-k) re+=tr[k];return re;} 
int main(){
	freopen("pass.in","r",stdin);
	freopen("pass.out","w",stdout);
	scanf("%d%s",&n,c+1),c[0]='#',c[n+1]='$';
	for(int i=2;i<=n;++i) a[i]=a[i-1]+(c[i-1]==c[i+1]);
	for(int i=1;i<n;++i){
		if(i<=R) z[i]=min(R-i,z[p*2-i]);
		while(c[i-z[i]]==c[i+z[i]+1]) ++z[i];
		if(i+z[i]>R) R=i+z[i],p=i;
	}
	for(int i=1;i<=n+1;++i) f[i]=i,r[i]=n+1;
	for(int i=1;i<n;++i)for(int j=F(i-z[i]+1);j<=i;j=F(j)) r[j]=i,f[j]=j+1;	
	for(int i=1;i<=n+1;++i) f[i]=i;
	for(int i=n-1;i;--i)for(int j=F(i+1);j<=i+z[i];j=F(j)) l[j]=i,f[j]=j+1;
	scanf("%d",&q);
	for(int i=1;i<=q;++i){
		scanf("%d%d",&x,&y);
		if(a[y-1]>a[x]) ans[i]=1;
		else e[x].push_back(make_pair(y,i));
	}
	for(int i=1;i<=n;++i)
		if(l[i]) sl[l[i]*2-i+1].push_back(i);
		else add(i,1),add(r[i]*2-i+1,-1);
	for(int i=1;i<=n;++i){
		for(pii j:e[i]) ans[j.second]=!ask(j.first);
		for(int j:sl[i]) add(j,1),add(r[j]*2-j+1,-1);
		add(i,-1),add(r[i]*2-i+1,1);
	}
	for(int i=1;i<=q;++i) printf("%d",ans[i]);
	return 0;
}
```
## [D. LCM序列](http://www.nfls.com.cn:10611/contest/1054/problem/4)
**题意：**

要构造长度为 $N(N\le38)$ 的**正整数序列** $A$ 满足对于给定的 $M(M<\frac{N(N+1)}2)$ 个限制 $(X_i,Y_i,B_i)$，其中 $A_{X_i},A_{Y_i}$ 的最小公倍数为 $B_i$，其中 $B_i$ 仅有前 $K(K\le8)$ 的质因子。

求序列的个数，若有无穷个序列输出 `-1`。

**题解：**

容易发现每个质因子是独立的。

将问题归纳为一张图，边为 $(X_i,Y_i,B_i)$，要求 $X_i$ 或 $Y_i$ 有一个点达到 $B_i$，图上每个点的的最大幂次为相连的边中最小的 $B$。

若一条边的两个点都达不到 $B$，则无解；只有一个点可达到则确定两点状态。

最终，只剩下若干个边权相同的点状态不确定（点状态指是否**未**达到最大幂次）的连通块。

若一点未达到最大幂次，则相邻点必须达到，可以状压 DP。

由于 $N\le38$，我们分成两半，先处理处前 $\frac n2$ 个点，再高维前缀和与后一半合并。

```cpp
#include<bits/stdc++.h>
using namespace std;
const int p=1e9+7,M=750,N=40,X=1<<19;
int n,m,k,fl,x[M],y[M],a[M][8],mx[N],lg[X],nn,MX1,MX2,g[X],f[X],sum,ans=1;
long long b,e[N],ne[X];
void work(int o){
	b=sum=0;
	for(int i=0;i<n;++i) mx[i]=2e9,e[i]=0;
	for(int i=1;i<=m;++i) mx[x[i]]=min(mx[x[i]],a[i][o]),mx[y[i]]=min(mx[y[i]],a[i][o]);
	for(int i=0;i<n;++i)if(mx[i]==2e9) fl=1;
	for(int i=1;i<=m;++i){
		if(mx[x[i]]<a[i][o]&&mx[y[i]]<a[i][o]) printf("0"),exit(0);
		if(mx[x[i]]>=a[i][o]&&mx[y[i]]>=a[i][o]) e[x[i]]|=1ll<<y[i],e[y[i]]|=1ll<<x[i];
		if(mx[x[i]]<a[i][o]) b|=1ll<<y[i];
		if(mx[y[i]]<a[i][o]) b|=1ll<<x[i];
	}
	memset(f,0,sizeof(f)),ne[0]=0,g[0]=1;
	for(int i=0,j;i<MX1;++i)if(!(i&b)){
		if(i) j=i&-i,ne[i]=ne[i^j]|e[lg[j]],g[i]=1ll*g[i^j]*mx[lg[j]]%p;
		if(!(i&ne[i])) f[i]=g[i];
	}
	for(int i=0;i<nn;++i)for(int j=0;j<MX1;++j)if(j&(1<<i)) (f[j]+=f[j^(1<<i)])%=p;
	ne[0]=0,g[0]=1;
	for(int i=0,j;i<MX2;++i)if(!(i&(b>>nn))){
		if(i) j=i&-i,ne[i]=ne[i^j]|e[lg[j]+nn],g[i]=1ll*g[i^j]*mx[lg[j]+nn]%p;
		if(!(i&(ne[i]>>nn))) (sum+=1ll*g[i]*f[(MX1-1)^(ne[i]&(MX1-1))]%p)%=p;
	}
	ans=1ll*ans*sum%p;
}
int main(){
	freopen("lcm.in","r",stdin);
	freopen("lcm.out","w",stdout);
	scanf("%d%d%d",&n,&m,&k),nn=n>>1,MX1=1<<nn,MX2=1<<(n-nn);
	for(int i=2;i<MX2;i<<=1) lg[i]=lg[i>>1]+1;
	for(int i=1;i<=m;++i){
		scanf("%d%d",&x[i],&y[i]),--x[i],--y[i];
		for(int j=0;j<k;++j) scanf("%d",&a[i][j]);
	}
	for(int i=0;i<k;++i) work(i);
	if(fl) printf("-1");
	else printf("%d",ans);
	return 0;
}
```
