## [B. 失落的世界](http://www.nfls.com.cn:10611/contest/1059/problem/2)
**题意：**

交互题。

$T\le1200$ 局游戏，选定两个正整数 $n,c,1\le c\le n\le1e18$，并将 $n$ 告诉你。

可以询问至多 $64$ 次，每次询问时你需要给出一个数 $x$ 满足 $1\le x\le n$。如果 $x$ 与你上次询问的数 $y$ 有 $|x − y|\ge c$，则返回 $1$，否则返回 $0$（第一次询问随机返回）。

**题解：**

首先想到二分，若 $n\le 1e9$ 就每两个询问 $(1,x)$ 一组，显然可以求出 $c$。

拓展到 $1e18$，考虑怎么把常数 $2$ 去掉。

观察到，当 $c>\frac{l+r}2$ 时要重置起点，常数为 $2$。因此，我们不妨平衡一下，将 $l,r$ 四等分，每次询问最右侧的四等分点。

![](Pasted%20image%2020231109140240.png)

直到返回第一个 $1$，我们确认了一个 $c$ 的范围 $l,r$。

定义询问点到边界的距离为 $D$，我们先对称地询问一次 $D$（将 $l,r$ 确定为下图红色或橙色，$r-l$ 为 $2*D$），再开始左右横跳询问。

可以发现，即使每次往左跳都小，往右跳都大，右侧询问也只是不断靠近 $n$，不会超过 $n$，因为边界距离 $D$ 与 $r-l$ 同步缩小。

![](Pasted%20image%2020231109143217.png)

```cpp
ll Guess(ll n){
	ll l=1,r=n,la,mi;int fl=0;
	Query(1);
	while(l^r){
		mi=(r-l)/4+1;
		if(Query(la=r-mi+1)){r=la-1;break;}
		else l=la;
		Query(1);
	}
	if(l^r){
		if(Query(mi)) r=la-mi;
		else l=la-mi+1;
		la=mi;
	}
	while(l^r){
		fl^=1,mi=(l+r)>>1;
		if(fl){
			if(Query(la=la+mi)) r=mi;
			else l=mi+1;
		}
		else{
			if(Query(la=la-mi)) r=mi;
			else l=mi+1;
		}
	}
	return l;
}
```
## [C. 最短路](http://www.nfls.com.cn:10611/contest/1059/problem/3)
**题意：**

![](Pasted%20image%2020231109192525.png)
![](Pasted%20image%2020231109192545.png)

**题解：**

我们把边当作为点跑最短路，但个原图的点出入边都较多时单点转移是 $O(m^2)$ 的。

![](Pasted%20image%2020231109193346.png)

如图，$j$ 转移比 $i$ 优的为出边的一段后缀。（$w$ 越大 $cost$ 增速越快）

因此，对于每个点 $x$，每从优先队列中取出一条入边 $i$，用栈和二分维护用 $i$ 来松弛的最右区间（一开始是后缀）。

注意，我们找到可松弛的区间后先不急着松弛入队，因为后面可能还有入边能进一步松弛，使得每个出边多次入队。

观察到，点 $x$ 的出边中前面的一定比后面的更早结束松弛（因为每新加入 $dis$ 更大入边可能影响的都是后缀）。因此，当我们把出边 $(x,y)$ 操作完（当作 $y$ 的入边松弛 $y$ 的出边）后，再把 $x$ 的下一条出边入队。

注意，此时下一条出边可能还能被松弛，因此每加入一条入边 $(x,y)$，也要把 $y$ 当前第一条还未操作的出边松弛入队。

```cpp
#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define pb push_back
#define pii pair<int,int>
#define pli pair<ll,int>
#define mp make_pair
#define fi first
#define se second
const int N=2e5+5;
const ll inf=1e18+5;
int n,m,k,a[6],b[6],c[6],x[N],y[N],w[N],oh[N],ih[N];
ll pw[N][6],dis,d[N],ans[N];
vector<int>ou[N];
vector<pii>in[N];
priority_queue<pli>q;
pli u;
ll D(int u,int v){
	ll re=d[u];
	for(int i=1;i<=k;++i) re+=pw[u][a[i]]*pw[v][b[i]]*c[i];
	return re;
}
void add(int u){
	while(ih[u]<(int)in[u].size()&&in[u][ih[u]].se<oh[u]) ++ih[u];
	if(oh[u]<(int)ou[u].size()&&d[ou[u][oh[u]]]>(dis=(u==1?0:D(in[u][ih[u]].fi,ou[u][oh[u]]))))
		q.push(mp(-(d[ou[u][oh[u]]]=dis),ou[u][oh[u]]));
}
void upd(int v,int i){
	int j,l,r,mi;
	while((int)in[v].size()>ih[v]){
		j=in[v].back().fi,r=in[v].back().se,in[v].pop_back();
		l=(int)in[v].size()>ih[v]?in[v].back().se+1:oh[v];
		if(l==(int)ou[v].size()) return;
		if(D(i,ou[v][l])<=D(j,ou[v][l])) continue;
		++r;
		while(l+1<r){
			mi=(l+r)>>1;
			if(D(j,ou[v][mi])<D(i,ou[v][mi])) l=mi;
			else r=mi;
		}
		in[v].pb(mp(j,l));
		if(l==(int)ou[v].size()-1) return;
		break;
	}
	in[v].pb(mp(i,(int)ou[v].size()-1));
}
int main(){
	freopen("path.in","r",stdin);
	freopen("path.out","w",stdout);
	scanf("%d%d%d",&n,&m,&k);
	for(int i=1;i<=k;++i) scanf("%d%d%d",&a[i],&b[i],&c[i]);
	for(int i=1;i<=m;++i){
		scanf("%d%d%d",&x[i],&y[i],&w[i]),ou[x[i]].pb(i),d[i]=inf,pw[i][0]=1;
		for(int j=1;j<=5;++j) pw[i][j]=pw[i][j-1]*w[i];
	}
	for(int i=1;i<=n;++i) ans[i]=inf,sort(ou[i].begin(),ou[i].end(),[](int a,int b){return w[a]<w[b];});
	add(1);
	while(!q.empty()){
		u=q.top(),q.pop();
		if(d[u.se]!=-u.fi) continue;
		upd(y[u.se],u.se),add(y[u.se]),++oh[x[u.se]],add(x[u.se]);
	}
	for(int i=1;i<=m;++i) ans[y[i]]=min(ans[y[i]],d[i]);
	for(int i=2;i<=n;++i) printf("%lld\n",ans[i]==inf?-1ll:ans[i]);
	return 0;
}
```
## [D. 聚合的塔尖](http://www.nfls.com.cn:10611/contest/1059/problem/4)
原题：[#2838. 「JOISC 2018 Day 3」比太郎的聚会](https://loj.ac/p/2838)

**题意：**

给定 $n\le1e5$ 个点 $m\le2e5$ 条边的有向图，其中点按拓扑排序。有 $Q\le1e5$ 次询问 $s,k,\sum k\le1e5$，除了 $k$ 个给定点外，其他点到 $s$ 的最长路。

**题解：**

根号分治。

- 如果 $k>\sqrt n$，这样的询问只会有至多 $O(\sqrt n)$ 个，直接建反图求最长路即可。

- 如果 $k\le\sqrt n$，我们可以对每个点预处理出距离它最远的 $\sqrt n$ 个点。具体的，每个点记录距离它前 $\sqrt n$ 大的点集 $q$，建反图后从小到大枚举 $i$，把 $v_i$ 的点集和 $\{i,0\}$ 加入 $q_i$，多的删去。

```cpp
#include<bits/stdc++.h>
using namespace std;
#define pb push_back
#define mp make_pair
#define pii pair<int,int>
#define fi first
#define se second
const int N=1e5+5;
int n,m,Q,x,y,s,k,fk,mx[N],d[N];
vector<int>e[N];
vector<pii>q[N];
map<int,int>b;
void work(){
	scanf("%d%d",&s,&k),b.clear();
	for(int i=1;i<=k;++i) scanf("%d",&x),b[x]=1;
	if(k<fk){
		for(pii i:q[s])if(!b[i.se]){printf("%d\n",i.fi);return;}
		printf("-1\n");return;
	}
	for(int i=1;i<=s;++i) d[i]=(b[i]?-n:0);
	for(int i=1;i<=s;++i)for(int v:e[i]) d[i]=max(d[i],d[v]+1);
	printf("%d\n",max(-1,d[s]));
}
int main(){
	scanf("%d%d%d",&n,&m,&Q),fk=sqrt(n);
	for(int i=1;i<=m;++i) scanf("%d%d",&x,&y),e[y].pb(x);
	memset(mx,-1,sizeof(mx));
	for(int i=1;i<=n;++i){
		q[i].pb(mp(0,i));
		for(int v:e[i])for(pii j:q[v]) mx[j.se]=max(mx[j.se],j.fi+1);
		for(int v:e[i])for(pii j:q[v])if(~mx[j.se]) q[i].pb(mp(mx[j.se],j.se)),mx[j.se]=-1;
		sort(q[i].begin(),q[i].end(),greater<pii>());
		if((int)q[i].size()>fk) q[i].erase(q[i].begin()+fk+1,q[i].end()); 
	}
	while(Q--) work();
	return 0;
}
```
