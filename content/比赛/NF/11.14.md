## [A. 平方](http://www.nfls.com.cn:10611/contest/1067/problem/1)
**题意：**

给定一长度为 $n$ 的序列 $a(1\le a_i\le1e6)$，求长度为 $n$ 的正整数序列 $b$，使得 $a_i\times a_{i+1}\times b_i\times b_{i+1}$ 为完全平方数，且 $\prod b_i$ 最小，输出 $\prod b_i\pmod{1e9+7}$。

**题解：**

![](Pasted%20image%2020231116161753.png)

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=1e6+5,p=1e9+7;
int n,x,a[N],ans=1;
void calc(int x){
	for(int i=2;i*i<=x;++i)if(x%i==0){
		int t=0;
		while(x%i==0) x/=i,t^=1;
		a[i]+=t;
	}
	if(x) ++a[x];
}
int main(){
	freopen("square.in","r",stdin);
	freopen("square.out","w",stdout);
	scanf("%d",&n);
	for(int i=1;i<=n;++i) scanf("%d",&x),calc(x);
	for(int i=2;i<=1e6;++i){
		a[i]=min(a[i],n-a[i]);
		for(int j=1;j<=a[i];++j) ans=1ll*ans*i%p;
	}
	printf("%d",ans);
	return 0;
}
```
## [B. 路径覆盖](http://www.nfls.com.cn:10611/contest/1067/problem/2)
**题意：**

给 $n\le1e5$ 个点的无根树，可以使用下面两种操作来覆盖所有边，且保证任意一条边恰被覆盖一次： 
1. 覆盖路径 $(u,v)$；
2. 选择一个点 $u$，覆盖与 $u$ 相连的所有边。

有 $q\le1e5$ 次独立询问，每次首先指定一个点 $x$ 进行操作 2，求最小化操作次数（即输出覆盖剩下的森林所需的最小操作次数 $+1$）。

**题解：**

定义 $f_{i,0/1/2}$ 分别为 $i$ 子树内所有点两两匹配、有一个点没匹配（可以与子树外的点连成路径）、$i$ 用操作 2 的方案数，则有：
- $f_{u,2}=\sum f_{v,0}+1$；
- $f_{v,1}$ 可以和 $f_{u,0}$ 合并为 $f_{u,1}$，可以和 $f_{u,1}-1$ 合并为 $f_{u,0}$；
- $f_{v,0}$ 可以和 $f_{u,1}$ 合并为 $f_{u,0}$，可以和 $f_{u,0}+1$ 合并为 $f_{u,1}$；
- $f_{v,2}$ 可以和 $f_{u,0/1}$ 合并为 $f_{u,0/1}$。

为了转移方便，我们另 $f_{i,0}*2+1,f_{i,1}*2-1,f_{i,2}*2$，即未匹配的路算半条，路径数统一翻倍。

同理，我们定义 $g_{i,0/1/2}$ 为 $i$ 子树外的答案。

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=1e5+5;
int n,q,x,y,f[N][3],g[N][3],ans[N];
vector<int>e[N];
void dfs1(int u,int fa){
	f[u][1]=f[u][2]=1;int t;
	for(int v:e[u])if(v^fa){
		dfs1(v,u),f[u][2]+=f[v][0];
		f[u][0]+=(t=min(f[v][2]*2,min(f[v][0]*2+1,f[v][1]*2-1))),f[u][1]+=t;
	}
	f[u][0]=(f[u][0]+1)/2,f[u][1]=(f[u][1]+1)/2;
}
void dfs2(int u,int fa){
	int t=max(0,min(g[u][2]*2,min(g[u][0]*2+1,g[u][1]*2-1)));int s[3]={t,t+1,g[u][0]+1};
	for(int v:e[u])if(v^fa)
		s[2]+=f[v][0],s[0]+=(t=min(f[v][2]*2,min(f[v][0]*2+1,f[v][1]*2-1))),s[1]+=t;
	ans[u]=s[2];
	for(int v:e[u])if(v^fa){
		g[v][0]=(s[0]-(t=min(f[v][2]*2,min(f[v][0]*2+1,f[v][1]*2-1)))+1)/2;
		g[v][1]=(s[1]-t+1)/2,g[v][2]=s[2]-f[v][0],dfs2(v,u);
	}
}
int main(){
	freopen("cover.in","r",stdin);
	freopen("cover.out","w",stdout);
	scanf("%d%d",&n,&q);
	for(int i=1;i<n;++i) scanf("%d%d",&x,&y),e[x].push_back(y),e[y].push_back(x);
	dfs1(1,0),dfs2(1,0);
	for(int i=1;i<=q;++i)scanf("%d",&x),printf("%d\n",ans[x]);
	return 0;
}
```

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=4e6+5;
int n,m,ans=1e9;
int a[N],b[N],u[N],v[N],c[N],fa[N];
int R()
{
	int x=0;char c=getchar();
	while(!isdigit(c)) c=getchar();
	while(isdigit(c)) x=x*10+c-48,c=getchar();
	return x;
}
int get(int x){return fa[x]==x?x:fa[x]=get(fa[x]);}
void merge(int x,int w)
{
	b[0]+=w,b[get(x)]-=w,x=fa[x];
	while(b[x]<0) b[get(x-1)]+=b[x],x=fa[x]=fa[x-1];
}
int main()
{
	freopen("challenge.in","r",stdin);
	freopen("challenge.out","w",stdout);
	n=R(),m=R();
	for(int i=1;i<=n;i++) a[i]=a[i-1]+R();
	for(int i=1;i<=n;i++) b[i]=R(),fa[i]=i;
	for(int i=1;i<=m;i++) u[i]=u[i-1]+R(),v[i]=R(),c[i]=R();
	for(int i=n,j=m;i>=0;i--)
	{
		ans=min(ans,a[i]+b[0]);
		while(j&&u[j]==i) merge(v[j],c[j]),j--;
	}
	printf("%d\n",ans);
	return 0;
}
```
