## [B. 艺术家](http://www.nfls.com.cn:10611/contest/1069/problem/2)
**题意：**

![](Pasted%20image%2020231117084459.png)
![](Pasted%20image%2020231117084538.png)

**题解：**

区间不相交或包含，可以将小区间连向大区间，按拓扑序更新。

用线段树维护每个位置颜色上一次出现的位置，判断区间最大值是否小于 $l$ 即可。

```cpp
#include<bits/stdc++.h>
using namespace std;
#define ls k<<1
#define rs k<<1|1
#define mp make_pair
#define pii pair<int,int>
#define fi first
#define se second
#define bg begin
const int N=5e5+5;
int n,m,Q,x,y,la,l[N],r[N],sb,a[N],nx[N],in[N],ff[N],bl[N],mx[N<<2],f[N],ans[N],ANS;
set<int>b[N];set<int>::iterator it;
vector<pii>e[N];
set<pair<pii,int>>c;set<pair<pii,int>>::iterator ii;
queue<int>q;
int F(int x){return x==f[x]?x:f[x]=F(f[x]);}
int FF(int x){return x==ff[x]?x:ff[x]=FF(ff[x]);}
void upd(int k,int l,int r,int x,int v){
	if(l==r){mx[k]=v;return;}
	int mi=(l+r)>>1;
	if(x<=mi) upd(ls,l,mi,x,v);
	else upd(rs,mi+1,r,x,v);
	mx[k]=max(mx[ls],mx[rs]);
}
int ask(int k,int l,int r,int x,int y){
	if(x<=l&&r<=y) return mx[k];
	int mi=(l+r)>>1,MX=0;
	if(x<=mi) MX=max(MX,ask(ls,l,mi,x,y));
	if(y>mi) MX=max(MX,ask(rs,mi+1,r,x,y));
	return MX;
}
int main(){
	freopen("artist.in","r",stdin);
	freopen("artist.out","w",stdout);
	scanf("%d%d%d",&n,&m,&Q);
	for(int i=1;i<=n;++i)
		scanf("%d",&a[i]),upd(1,1,n,i,(b[a[i]].size()?*b[a[i]].rbegin():0)),b[a[i]].insert(i),ff[i]=i;
	for(int i=1;i<=m;++i)
		scanf("%d%d",&l[i],&r[i]),e[l[i]].push_back(mp(r[i],i)),f[i]=i,ans[i]=m+i;
	for(int i=1;i<=n;++i)if(e[i].size()){
		sort(e[i].bg(),e[i].end()),la=0;
		for(pii j:e[i]){
			if(la) nx[la]=j.se,++in[j.se];
			la=j.se;
		}
		ii=c.lower_bound(mp(mp(r[la],-n),0));
		if(ii!=c.end()) nx[la]=(*ii).se,++in[nx[la]];
		for(pii j:e[i]) c.insert(mp(mp(j.fi,-i),j.se));
	}
	for(int i=1;i<=m;++i)if(!in[i]) q.push(i);
	while(!q.empty()){
		x=q.front(),q.pop();
		if(!(--in[nx[x]])) q.push(nx[x]);
		if(ask(1,1,n,l[x],r[x])<l[x]){ans[x]=0;continue;}
		for(int i=FF(l[x]);i<=r[x]&&i;i=FF(i)) bl[i]=x,ff[i]=i+1;
	}
	for(int i=1;i<=Q;++i){
		scanf("%d%d",&x,&y),b[a[x]].erase(x),it=b[a[x]].upper_bound(x);
		if(it!=b[a[x]].end()) sb=*it,upd(1,1,n,sb,(it==b[a[x]].bg()?0:*(--it)));
		it=b[y].upper_bound(x);
		if(it!=b[y].end()) upd(1,1,n,*it,x);
		upd(1,1,n,x,(it==b[y].bg()?0:*(--it))),b[y].insert(x),a[x]=y;
		if(!(y=F(bl[x]))) continue;
		while(y&&ask(1,1,n,l[y],r[y])<l[y]) ans[y]=i,f[y]=nx[y],y=nx[y];
	}
	for(int i=1;i<=m;++i) ANS^=ans[i];
	printf("%d",ANS);
	return 0;
}
```
