## [A. 棋局](http://www.nfls.com.cn:10611/contest/1068/problem/1)
**题意：**

![](Pasted%20image%2020231116200851.png)
![](Pasted%20image%2020231116200904.png)

**题解：**

![](Pasted%20image%2020231116201843.png)

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=2e5+5;
#define int long long
int n,m,p,a[N],l,r,mi,la,cnt[N];
long long s[N];
signed main(){
	freopen("chess.in","r",stdin);
	freopen("chess.out","w",stdout);
	scanf("%lld%lld",&n,&m),la=m;
	for(int i=1;i<=n;++i) scanf("%lld",&a[i]);
	sort(a+1,a+1+n);
	for(int i=1;i<=n;++i) s[i]=s[i-1]+a[i];
	for(int i=1;i<=n;++i){
		if(s[n]<=1ll*(i-1)*m){printf("0 ");continue;}
		l=(s[n]-1ll*(i-1)*m-1)/(n-i+1),r=la;
		while(l+1<r){
			mi=(l+r)>>1,p=lower_bound(a+1,a+1+n,mi)-a;
			if(s[n]-s[p-1]-1ll*(n-p+1)*mi<=1ll*(m-mi)*(i-1)) r=mi;
			else l=mi;
		}
		la=r,printf("%lld ",r);
	}
	return 0;
}
```
## [B. 难题走三方](http://www.nfls.com.cn:10611/contest/1068/problem/2)
**题意：**

![](Pasted%20image%2020231116202000.png)
![](Pasted%20image%2020231116202011.png)

**题解：**

找到 $a_i$ 左右两边第一个大于 $a_i$ 的位置 $l_i,r_i$，定义 $f_{i,j,k}$ 表示 $a$ 匹配到第 $i$ 位，用了 $j$ 次操作，$b$ 填到第 $k$ 位的答案，则 $f_{i,j,k}=\sum_{p\ne i-1,l_i<p<r_i} f_{i-1,j-1,p}+f_{i-1,j,i-1}$，前缀和优化一下。

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=405,p=998244353;
int n,m,a[N],t,q[N],l[N],r[N],f[N][N],g[N][N],ans;
int main(){
	freopen("cube.in","r",stdin);
	freopen("cube.out","w",stdout);
	scanf("%d%d",&n,&m);
	for(int i=1;i<=n;++i) scanf("%d",&a[i]);
	for(int i=1;i<=n;++i){
		while(t&&a[q[t]]<a[i]) --t;
		l[i]=q[t],q[++t]=i;
	}
	q[t=0]=n+1;
	for(int i=n;i;--i){
		while(t&&a[q[t]]<a[i]) --t;
		r[i]=q[t],q[++t]=i;
	}
	f[0][0]=1;
	for(int i=1;i<=n;++i){
		memcpy(g,f,sizeof(g));
		for(int j=1;j<=i;++j)for(int k=l[i]+1,s=0;k<r[i];++k)
			(s+=g[j-1][k-1])%=p,(f[j][k]+=s)%=p;
		for(int j=1;j<=i;++j) (f[j][i]+=p-g[j-1][i-1])%=p;
		for(int j=0;j<i;++j) (f[j][i]+=g[j][i-1])%=p;
	}
	for(int i=0;i<=m;++i) (ans+=f[i][n])%=p;
	printf("%d",ans);
	return 0;
}
```
## [C. 动态半平面交](http://www.nfls.com.cn:10611/contest/1068/problem/3)
**题意：**

![](Pasted%20image%2020231116203554.png)
![](Pasted%20image%2020231116203606.png)

**题解：**

我们可以把一个数的质因子 $p^k$ 拆成 $p^1,p^2\dots p^k$，每个贡献为 $p$，在 $u$ 子树内距离 $u$ 不超过 $d$ 的点中，若存在 $p^i$ 则 lcm 乘上 $p$。

![](Pasted%20image%2020231117082256.png)

```cpp
#include<bits/stdc++.h>
using namespace std;
#define pb push_back
const int N=1e5+5,p=998244353;
int n,k,x,y,lg[N],d[N],tim,l[N],id[N],r[N],f[N][20],o[N],tr[N],ans[N],q;
struct que{int x,d,id;}s[N];
vector<int>a[N],e[N];
void dfs(int u,int fa){
	d[u]=d[fa]+1,l[u]=++tim,id[tim]=u,f[u][0]=fa;
	for(int i=1;i<=lg[d[u]];++i) f[u][i]=f[f[u][i-1]][i-1];
	for(int v:e[u])if(v^fa) dfs(v,u);
	r[u]=tim;
}
int LCA(int x,int y){
	if(d[x]<d[y]) swap(x,y);
	while(d[x]^d[y]) x=f[x][lg[d[x]-d[y]]];
	if(x==y) return x;
	for(int i=lg[d[x]];~i;--i)if(f[x][i]^f[y][i]) x=f[x][i],y=f[y][i];
	return f[x][0];
}
int poww(int x,int y){
	int e=1;
	while(y){
		if(y&1) e=1ll*e*x%p;
		x=1ll*x*x%p,y>>=1;
	}
	return e;
}
void add(int k,int v){for(;k<=n;k+=k&-k) tr[k]=1ll*tr[k]*v%p;}
int ask(int k){int re=1;for(;k;k-=k&-k) re=1ll*re*tr[k]%p;return re;}
map<int,set<int>>c;
set<int>::iterator it;
void upd(int x){
	int sv=1,la=0,inv;
	for(int i:a[x]){
		if(la^i) sv=1,inv=poww(i,p-2);
		sv*=i,it=c[sv].lower_bound(l[x]);int ll=0,rr=0;
		if(it!=c[sv].end()) rr=id[*it],add(l[LCA(rr,x)],inv);
		if(it!=c[sv].begin()) --it,ll=id[*it],add(l[LCA(ll,x)],inv);
		if(ll&&rr) add(l[LCA(ll,rr)],i);
		add(l[x],i),c[sv].insert(l[x]),la=i;
	}
}
int main(){
	freopen("half.in","r",stdin);
	freopen("half.out","w",stdout);
	scanf("%d%d",&k,&n);
	for(int i=1;i<=n;++i){
		scanf("%d",&x);
		for(int j=2;j*j<=x;++j)while(x%j==0) a[i].pb(j),x/=j;
		if(x>1) a[i].push_back(x);
	}
	for(int i=1;i<=n;++i) o[i]=i,tr[i]=1;
	for(int i=2;i<=n;++i) scanf("%d%d",&x,&y),e[x].pb(y),e[y].pb(x),lg[i]=lg[i>>1]+1;
	dfs(1,0),sort(o+1,o+1+n,[](int a,int b){return d[a]<d[b];}),scanf("%d",&q);
	for(int i=1;i<=q;++i) scanf("%d%d",&s[i].x,&s[i].d),s[i].id=i;
	sort(s+1,s+1+q,[](que a,que b){return d[a.x]+a.d<d[b.x]+b.d;});
	for(int i=1,j=1;i<=q;++i){
		while(j<=n&&d[o[j]]<=d[s[i].x]+s[i].d) upd(o[j++]);
		ans[s[i].id]=1ll*ask(r[s[i].x])*poww(ask(l[s[i].x]-1),p-2)%p;
	}
	for(int i=1;i<=q;++i) printf("%d\n",ans[i]);
	return 0;
}
```
