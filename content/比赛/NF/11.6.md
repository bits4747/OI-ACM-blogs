## [A. 缩略](http://www.nfls.com.cn:10611/contest/1052/problem/1)
原题：[[AGC027E] ABBreviate](https://www.luogu.com.cn/problem/AT_agc027_e)

**题意：**

![](Pasted%20image%2020231107115640.png)

要求时间复杂度 $O(|S|)$。

**题解：**

令 `x` 为 $1$，`y` 为 $2$，那么操作完以后所有数对 $3$ 取模的和不变。

只要权值和取模后不为 $0$ 且有两个相邻的字符相同的字符串都可以缩成一个字符。

特判答案为 $1$ 的情况（`xyxyx`）后 dp，定义 $f_i$ 为前 $i$ 个字符的答案。

若 $s_i\ne0$，初始化 $f_i=1$ 表示 $[1,i]$ 缩成一个字符；再找到上一个 $s_j\ne s_i$ 的位置，将 $[j+1,i]$ 缩成一个字符，另 $f_i+=f_j$。

转移中一段 $[l,r]\ne0$ 可能会出现形如 `xyxyx` 的，但经过调整能缩成一个字符，具体我不会证。

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=5e4+5,p=998244353;
char c[N];
int n,b,s[N],f[N],la[3];
void work(){
	n=strlen(c+1),b=0;
	for(int i=1;i<=n;++i) s[i]=(s[i-1]+c[i]-'w')%3,b|=(c[i]==c[i+1]);
	if(!b){printf("1\n");return;}
	la[0]=la[1]=la[2]=0;
	for(int i=1;i<=n;++i){
		f[i]=(s[i]!=0);
		for(int j=0;j<3;++j)if(j^s[i]) (f[i]+=f[la[j]])%=p;
		la[s[i]]=i;
	}
	printf("%d\n",f[n]);
}
int main(){
	freopen("abbreviate.in","r",stdin);
	freopen("abbreviate.out","w",stdout);
	while(scanf("%s",c+1)!=EOF) work();
	return 0;
}
```
## [B. 最小生成树](http://www.nfls.com.cn:10611/contest/1052/problem/2)
**题意：**

给定 $N(N\le3e5)$ 个顶的带权简单无向连通图，保证点 $1$ 不是割点，且点 $1$ 到其他点都有连边。

求点 $1$ 度数为 $[1,N-1]$ 的最小生成树。

**题解：**

首先，度数为 $1$ 的好求。定义 $v_i$ 表示 $val(1,i)$，先把点 $1$ 删去，求出剩下点的最小生成树再加上 $\min\{v_i\}$ 即可。

接下来要不断加入 $(1,i)$。

我假设们当前要加入边 $(1,x)$，记 $x$ 所在连通块（连通块不包括 $1$ 及 $1$ 的边）与 $1$ 相连的点为 $y$，则变化为 $v_x-\max\{w_i,i\in path(x,y)\}$。

等同于删去 $1$ 后求 kruskal，另 $v$ 最小的为连通块的根，合并 $x_i,y_i$ 所在连通块时，会贡献 $\max(rt_{v_x},rt_{v_y})+w_i$（取 $\max$ 是因为小的会先加入），记录下来排序加入即可。

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=3e5+5,M=6e5+5;
int n,m,x,y,w,t,v[N],f[N],tt,del[N];
long long ans;
struct qwq{int x,y,w;}s[M];
int F(int x){return x==f[x]?x:f[x]=F(f[x]);}
int main(){
	freopen("rootedmst.in","r",stdin);
	freopen("rootedmst.out","w",stdout);
	scanf("%d%d",&n,&m);
	for(int i=2;i<=n;++i) v[i]=1e6,f[i]=i;
	for(int i=1;i<=m;++i){
		scanf("%d%d%d",&x,&y,&w);
		if(x==1) v[y]=min(v[y],w);
		else s[++t]=(qwq){x,y,w};
	}
	sort(s+1,s+1+t,[](qwq a,qwq b){return a.w<b.w;});
	for(int i=1;i<=t;++i)if((x=F(s[i].x))^(y=F(s[i].y))){
		if(v[x]>v[y]) swap(x,y);
		del[++tt]=v[y]-s[i].w,ans+=s[i].w,f[y]=x;
	}
	sort(del+1,del+1+tt);
	for(int i=2;i<=n;++i)if(F(i)==i){ans+=v[i];break;}
	printf("%lld",ans);
	for(int i=1;i<=n-2;++i) ans+=del[i],printf(" %lld",ans);
	return 0;
}
```

```cpp
#include<bits/stdc++.h>
using namespace std;
#define pb push_back
#define ui unsigned int
const int N=2e5+5,M=N*40;
int n,m,x,y,tp[N],lg[N],d[N],f[N][20],cnt,rt1[N],rt2[N],ls[M],rs[M],mn[M];
ui s[M],p[N],c[N],ans[N];
vector<int>e[N],ad[N],de[N],qe[N],t;
void dfs(int u,int fa){
	d[u]=d[fa]+1,f[u][0]=fa;
	for(int i=1;i<=lg[d[u]];++i) f[u][i]=f[f[u][i-1]][i-1];
	for(int v:e[u])if(v^fa) dfs(v,u);
}
int LCA(int x,int y){
	if(d[x]<d[y]) swap(x,y);
	while(d[x]^d[y]) x=f[x][lg[d[x]-d[y]]];
	if(x==y) return x;
	for(int i=lg[d[x]];~i;--i)if(f[x][i]^f[y][i]) x=f[x][i],y=f[y][i];
	return f[x][0];
}
ui F(int k,int v){
	if(mn[k]>v) return 0;
	if(!ls[k]&&!rs[k]) return s[k];
	if(!ls[k]) return F(rs[k],v);
	if(!rs[k]) return F(ls[k],v);
	if(mn[rs[k]]<v) return s[k]-s[rs[k]]+F(rs[k],v);
	return F(ls[k],v);
}
void pu(int k){
	if(!ls[k]) mn[k]=mn[rs[k]],s[k]=s[rs[k]];
	else if(!rs[k]) mn[k]=mn[ls[k]],s[k]=s[ls[k]];
	else mn[k]=min(mn[ls[k]],mn[rs[k]]),s[k]=s[rs[k]]+F(ls[k],mn[rs[k]]);
}
void mrg(int &k1,int &k2,int l,int r){
	if(!k1||!k2){k1+=k2;return;}
	if(l==r) return;
	int mi=(l+r)>>1;mrg(ls[k1],ls[k2],l,mi),mrg(rs[k1],rs[k2],mi+1,r),pu(k1);
}
void add(int &k,int l,int r,int v,int ti){
	if(!k) k=++cnt;
	if(l==r){mn[k]=ti,s[k]=c[p[ti]];return;}
	int mi=(l+r)>>1;
	if(v<=mi) add(ls[k],l,mi,v,ti);
	else add(rs[k],mi+1,r,v,ti);
	pu(k);
}
void que(int k,int l,int r,int v){
	if(!k) return;
	if(r<=v){t.pb(k);return;}
	int mi=(l+r)>>1;que(ls[k],l,mi,v);
	if(v>mi) que(rs[k],mi+1,r,v);
}
ui ask(int k,int v,int ti){
	t.clear(),que(k,1,m,v);ui re=0;
	for(int i=t.size()-1;~i;--i) re+=F(t[i],ti),ti=min(ti,mn[t[i]]);
	return re;
}
void dfs(int u){
	for(int v:e[u])if(v^f[u][0]) dfs(v),mrg(rt1[u],rt1[v],1,m),mrg(rt2[u],rt2[v],1,m);
	for(int i:ad[u]) add(rt1[u],1,m,p[i],i),add(rt2[u],1,m,m-p[i]+1,i);
	for(int i:qe[u]) ans[i]=ask(rt1[u],p[i],i)+ask(rt2[u],m-p[i]+1,i);
	for(int i:de[u]) add(rt1[u],1,m,p[i],m+1),add(rt2[u],1,m,m-p[i]+1,m+1);
}
int main(){
	freopen("tree.in","r",stdin);
	freopen("tree.out","w",stdout);
	scanf("%d%d",&n,&m);
	for(int i=1;i<n;++i) scanf("%d%d",&x,&y),e[x].pb(y),e[y].pb(x);
	for(int i=2;i<=n;++i) lg[i]=lg[i>>1]+1;
	dfs(1,0);
	for(int i=1;i<=m;++i){
		scanf("%d%d",&tp[i],&x);
		if(tp[i]==1) scanf("%d",&y),ad[x].pb(i),ad[y].pb(i),de[LCA(x,y)].pb(i);
		else qe[x].pb(i);
		scanf("%u",&p[i]),c[i]=p[i];
	}
	sort(c+1,c+1+m);
	for(int i=1;i<=m;++i) p[i]=lower_bound(c+1,c+1+m,p[i])-c;
	dfs(1);
	for(int i=1;i<=m;++i)if(tp[i]==2) printf("%u\n",ans[i]);
	return 0;
}
```
## [D. 岛屿](http://www.nfls.com.cn:10611/contest/1052/problem/4)
原题：[【UR # 6】懒癌](https://uoj.ac/problem/76)

**题意：**

往岛上放 $n\le3000$ 个人，每个人都不知道自己的眼睛是红色还是蓝色，已知至少有一个人是红眼睛。

每个人会留意其他哪些人的眼睛颜色，每个人都知道彼此留意的对象，但只有当事人知道具体颜色。

如果有人确认自己眼睛的颜色，他就会在中午当着所有人的面自杀，多个人确定就同时自杀。

请问在所有眼睛颜色情况中：

1. 首次自杀时间之和。
2. 首次自杀人数之和。

**题解：**

考虑完全图，即每个人都知道其他所有人的眼睛颜色时，若总共有 $K$ 个红眼睛，则所有红眼睛的人都会在第 $K$ 天一起自杀。

> 证明：
> 
> 一开始大家都假设自己是蓝眼睛。
> 
> 当只有 $1$ 个红眼睛时红眼睛的人会发现自己是红眼睛，第一天自杀。
> 
> 若第一天无人自杀，则说明至少有 $2$ 个红眼睛，若此时有人只看到 $1$ 个红眼睛，则知道自己是红眼睛，第二天自杀，以此类推。

推广到原题，每个人都会假设自己看不到的人是红眼睛，自己是蓝眼睛，直到这种假设都无法成立时才自杀。

若 $i$ 看不到 $j$，我们连边 $i\to j$。当 $j$ 确定无法得知自己眼睛颜色后（确定不自杀），$i$ 可以根据 $j$ 来尝试推理自己的眼睛颜色。

记红色为 $1$，蓝色为 $0$。一开始 $i$ 把自己看作 $0$，把 $j(v_i)$ 看作 $1$。为了得知 $j$ 的状态，带入 $j$ 视角，这时 $j$ 把自己看作 $0$，$v_j$ 看作 $1$……一直到走到叶子。

因此，若推理链上能到达环则无人自杀，我们把能到达环的点删去，剩下的图为 DAG。

可以发现，每一次 $1$ 的传递都是一次状态转移，设转移前状态为 $U$，转移后状态为 $V$，则有 $f_U=f_V+1$。因此，每一个点自杀的时间为它能到达的点数。

- 求所有首次自杀时间之和：

> 若有 $1$ 能到达 $i$，则 $i$ 会产生贡献。
> 
> 记能到达 $i$ 的点数为 $x$，DAG 点数为 $t$，则 $i$ 有贡献 $(2^x-1)\times 2^{t-x}$，即能到达 $i$ 的点（包括 $i$）至少有一个 $1$，无观点随意。
- 求所有首次自杀的人数之和：

> 一个状态的自杀时间就是 $1$ 能到达的点数。假如一个$1$ 不能被其它 $1$ 到达，那往后推一步得到的新状态中 $1$ 能到达的点数就会 $-1$；反之，则不变。
> 
> 只有最早自杀的人才会被记入答案，因此只有 $-1$ 的人才会自杀，而 $-1$ 的条件是没有其它 $1$ 能到达自己。
> 
> 对每个点考虑贡献，答案为 $2^{t-x}$（自己必须为 $1$，其他能到 $i$ 的点不能为 $1$，剩下的随便选）。

我们建反图，便于去掉环及环能到的点（原图中能到环的点）。

拓扑排序后，对于每个 $e(i,j)=1,i<j$，用 bitset 优化$e_i|=e_j$，求出每个点能到的点（及原图中能到 $i$ 的点），复杂度 $O(\frac{n^3}w)。$

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=3005,p=998244353;
char c[N][N];
int n,in[N],t,q[N],x,tim,num;
bitset<N>g[N];
int poww(int x,int y){
	int e=1;
	while(y){
		if(y&1) e=1ll*e*x%p;
		x=1ll*x*x%p,y>>=1;
	}
	return e;
}
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;++i)for(int j=1;j<=n;++j){
		scanf(" %c",&c[i][j]);
		if(c[i][j]=='0'&&i!=j) ++in[i];
	}
	for(int i=1;i<=n;++i)if(!in[i]) q[++t]=i;
	for(int i=1;i<=t;++i)for(int j=1;j<=n;++j)if(j!=q[i]&&c[j][q[i]]=='0'&&!(--in[j])) q[++t]=j;
	for(int i=1;i<=t;++i)for(int j=1;j<=t;++j)if(c[q[i]][q[j]]=='0') g[j][i]=1;
	for(int i=1;i<=t;++i)for(int j=1;j<=t;++j)if(g[i][j]) g[i]|=g[j];
	for(int i=1;i<=t;++i)
		x=g[i].count(),(tim+=1ll*(poww(2,x)+p-1)*poww(2,t-x)%p)%=p,(num+=poww(2,t-x))%=p;
	printf("%d %d",tim,num);
	return 0;
}
```
