## [A. 整数分解](http://www.nfls.com.cn:10611/contest/1065/problem/1)
**题意：**

![](Pasted%20image%2020231113164934.png)
![](Pasted%20image%2020231113164946.png)

**题解：**

找出所有约数排序，暴搜剪枝，复杂度 O(能过)。

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=64000;
int cnt,a[N],b[N],sum,q,n,m;
int dfs(int k,int num,int st){
	if(num==m&&st==n) return 1;
	if(k>cnt||num>=m||cnt-k+1+num<m) return 0;
	if(~b[k]&&1ll*b[k]*st<n) return 0;
	if(b[cnt-(m-num)+1]==-1||1ll*b[cnt-(m-num)+1]*st>n) return 0;
	if(1ll*st*a[k]<=n&&dfs(k+1,num+1,st*a[k])) return 1;
	return dfs(k+1,num,st);
}
void work(){
	scanf("%d%d",&n,&m),cnt=0,sum=1;
	for(int i=1;i*i<=n;++i)if(n%i==0) a[++cnt]=i,a[++cnt]=n/i;
	if(a[cnt]==a[cnt-1]) --cnt;
	sort(a+1,a+1+cnt,[](int x,int y){return x>y;});
	for(int i=cnt;i;--i){
		if(~sum){
			if(1ll*sum*a[i]>n) sum=-1,b[i]=-1;
			else sum*=a[i],b[i]=sum;
		}
		else b[i]=-1;
	}
	if(dfs(1,0,1)) printf("TAK\n");
	else printf("NIE\n");
}
int main(){
	freopen("num.in","r",stdin);
	freopen("num.out","w",stdout);
	scanf("%d",&q);
	while(q--) work();
	return 0;
}
```
## [B. 闪烁之光](http://www.nfls.com.cn:10611/contest/1065/problem/2)
**题意：**

有一张 $n\le2000$ 个点 $m\le4000$ 条边的图，已知 $n$，最多询问 $75000\le Q\le100000$ 次，每次询问一个点集，交互库返回该子图的边数，要还原出此图。

**题解：**

首先 $n^2$ 必定能做。

观察询问次数，发现是 $\log$ 级别的。

我们从 $1$ 到 $n$ 依次确定每个点 $i$ 和 $<i$ 的点的连边。我们可以用类似线段树的形式来询问，先问区间 $[1,i-1]$ 和 $i$ 点集得到 $i$ 和前面所有点的边数（小于 $i$ 的点的边数之前已知），再询问 $[1,mid]$，若左侧有边则递归，右侧同理。

```cpp
#include<bits/stdc++.h>
#include"YuukiAsuna.h"
#define vt vector
#define pii pair<int,int>
#define pb push_back
#define fi first
#define se second
using namespace std;
vt<pii>ans;vt<int>s;
int t;
int ask(int k,int l,int r){
	s.clear(),s.pb(k);
	for(int i=l;i<=r;++i) s.pb(i);
	int re=Query(s);
	for(pii i:ans)if(l<=i.fi&&i.fi<=r&&l<=i.se&&i.se<=r) --re;
	return re;
}
void solve(int k,int l,int r,int sum){
	if(l==r){ans.pb(make_pair(k,l));return;}
	int mi=(l+r)>>1,tt=ask(k,l,mi);
	if(tt) solve(k,l,mi,tt);
	if(tt<sum) solve(k,mi+1,r,sum-tt);
}
vt<pii>Asuna(int n,int Q){
	for(int i=2;i<=n;++i)if((t=ask(i,1,i-1))) solve(i,1,i-1,t);
	return ans;
}
```
## [C. 反射](http://www.nfls.com.cn:10611/contest/1065/problem/3)
**题意：**

![](Pasted%20image%2020231113170539.png)
$1\le n\le 3000$

**题解：**

由于父亲编号大于儿子，删掉边 $(u,v)$，则 $u$ 最浅的还连通的祖先会与 $v$ 连边。因此，每个点只与其祖先有关。

定义 $f_{i,j}$ 表示 $i$ 子树只能和 $i$ 的前 $j$ 级祖先连边的合法方案数，则 $f_{u,i}=\prod f_{v,i+1}$，答案为 $f_{n,0}$。

注意，由于只有 $n$ 不用往上连边，前缀和时不能把 $f_{i,0}$ 算上。

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=3005,p=998244353;
int n,x,y,f[N][N];
vector<int>e[N];
void dfs(int u,int d){
	for(int i=0;i<=d;++i) f[u][i]=1;
	for(int v:e[u]){
		dfs(v,d+1);
		for(int i=0;i<=d;++i) f[u][i]=1ll*f[u][i]*f[v][i+1]%p;
	}
	for(int i=2;i<=d;++i) (f[u][i]+=f[u][i-1])%=p;
}
int main(){
	freopen("reflection.in","r",stdin);
	freopen("reflection.out","w",stdout);
	scanf("%d",&n);
	for(int i=1;i<n;++i) scanf("%d%d",&x,&y),e[max(x,y)].push_back(min(x,y));
	dfs(n,1),printf("%d",f[n][0]);
	return 0;
}
```
## [D. 人赢](http://www.nfls.com.cn:10611/contest/1065/problem/4)
**题意：**

给定 $n\le5e5$ 个点，每个点有 $a_i<2^{30}$。你需要连 $n$ 条边（可以有重边或自环）来组成若干个不相交的环，使得每个点都在环上。边的边权是其两个端点点权的异或值，你需要最大化所有边权的最小值。

**题解：**

二分答案 $x$。

我们把一点拆成两点，相邻的点连边跑二分图匹配。

我们把问题弱化为一般图匹配，即两边的点内部也可以连边，可以发现因为每个点度为 $2$，一般图匹配都可以转化为二分图匹配。

因此，我们要求边权不超过 $x$ 的二分图完美匹配。

![](Pasted%20image%2020231113184029.png)

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=15e6+5;
int n,x,mx,u,cnt=1,ch[N][2],sz[N],l,r=1<<30,mi;
void add(){
	int c;u=1;
	for(int i=29;~i;--i){
		c=(x>>i)&1;
		if(!ch[u][c]) ch[u][c]=++cnt;
		u=ch[u][c],++sz[u];
	}
}
int ask(int u,int v,int d){
	int lim=min(sz[v]-sz[u],0);
	if(!u||!v||d==-1) return lim;
	if((mi>>d)&1) return ask(ch[u][0],ch[v][1],d-1)+ask(ch[u][1],ch[v][0],d-1);
	return min(lim,min(ask(ch[u][0],ch[v][0],d-1)+sz[ch[v][1]],ask(ch[u][1],ch[v][1],d-1)+sz[ch[v][0]]));
}
int main(){
	freopen("win.in","r",stdin);
	freopen("win.out","w",stdout);
	scanf("%d",&n),sz[1]=n;
	for(int i=1;i<=n;++i) scanf("%d",&x),mx=max(x,mx),add();
	if(n>1e5&&mx>(1<<29)) l=(1<<29)-100,r=(1<<29)+100;
	while(l+1<r){
		mi=(l+r)>>1;
		if(ask(1,1,29)>=0) l=mi;
		else r=mi;
	}
	printf("%d",l);
	return 0;
}
```
