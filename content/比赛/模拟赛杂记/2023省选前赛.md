## [树上路径（tree）](http://xsy.gdgzez.com.cn/JudgeOnline/problem.php?cid=1993&pid=0)

### 快速求出每棵子树内的重心
根据树的重心的性质:两颗子树合并，新树重心一定在两个子树重心的连线上。

所以合并子树的时候，让子树们的重心往祖先跳。
```cpp
f[u][0]=fa,d[u]=d[fa]+1,sz[u]=1,rt[u]=u;
for(int v:e[u]){
	if(v==fa) continue;
	dfs1(v,u),sz[u]+=sz[v],dp1[u]+=dp1[v];
	if(sz[v]>sz[son[u]]) son[u]=v;
	if(sz[v]*2>sz[u]) rt[u]=rt[v];//重心在v内
	while((sz[u]-sz[rt[u]])*2>sz[u]) rt[u]=f[rt[u]][0];//v的重心往上跳，直到整棵子树的重心
}
```
### 把一堆数分成尽量平均的两堆
#### 随机化

#### 背包（bitset优化）
```cpp
bitset<50001>s,tmp,tt;
void dfs(int u){
	for(int v:e[u]) if(v!=f[u][0]&&v!=son[u]) dfs(v);
	if(son[u]) dfs(son[u]);//利用树链剖分的思想，重儿子子树的重心有可能和u子树的重心相同
	if(rt[u]!=rt[son[u]]){//先把重心的子树放入背包（重心上面那一块的大小会变）
		s.reset(),s[0]=1,sum=0;
		for(int v:e[rt[u]])
			if(v!=f[rt[u]][0]){
				if(sz[v]<=50000) tt=s,tt<<=sz[v],s|=tt;//核心
				//注意：不能写成s|=s<<sz[v]。这么写每次运算系统都会多开几个bitset，会MLE
				sum+=dp1[v];
			}
	}
	tmp=s;
	if(sz[u]-sz[rt[u]]<=50000) tt=tmp,tt<<=sz[u]-sz[rt[u]],tmp|=tt;//把重心上面的放入背包
	ans[u]+=sum+get0(rt[u],u)+sz[u]/*统计子树内和所有节点到u的贡献*/,mx=(sz[u]-1)>>1;
	for(int i=mx;i>=0;--i)if(tmp[i]){ans[u]+=1ll*i*(sz[u]-1-i);break;}//分成最接近sz[u]/2的两堆，一堆a正向，一堆b逆向，统计a-->b的路径
}
```
### 预处理重心的每棵子树内部的贡献
记 $dp1_i$ 表示原树上以 $i$ 为根的子树内同向的简单路径数。

记 $dp0_i$ 表示原树上以 $i$ 的父亲为根的子树除去以 $i$ 为根的子树后同向的简单路径数。

对 $dp0_{i,j}$ 倍增处理，表示原树上以 $i$ 的 $i$ 的 $2^j$ 级祖先为根的子树除去以 $i$ 为根的子树后同向的简单路径数。
```cpp
dp0[v][0]=dp1[u]-dp1[v]-sz[v];
void dfs2(int u){
	for(int i=1;i<18;++i){
		f[u][i]=f[f[u][i-1]][i-1];
		if(!f[u][i]) break;
		dp0[u][i]=dp0[u][i-1]+dp0[f[u][i-1]][i-1]+(sz[f[u][i]]-sz[f[u][i-1]])*(1ll<<(i-1));//中间这块要计算1ll<<(i-1)次

		re+=dp0[u][i]+(sz[f[u][i]]-sz[u])*dis,dis+=1<<i,u=f[u][i];//中间这块要计算dis次
	return re;
}
```

## [Problem A: A](http://xsy.gdgzez.com.cn/JudgeOnline/problem.php?cid=1996&pid=0)

![[Pasted image 20230322075826.png]]

可以发现，将所有偶数操作完后就会剩下一个奇数集合，接下来的操作为：取出最大的奇数 $x$，$x=\frac{x−1}2$ 后一直除以 $2$ 直到变成奇数，后重新放入奇数集合。

前缀和维护区间内所有偶数变成奇数所需的操作次数，当操作次数小于等于把原集合的所有偶数变成奇数所需的操作次数时，答案为 $max(最大的奇数,操作后剩下的最大偶数)$。我们对原来的偶数维护一棵值域主席树，在主席树上二分可以找出操作后剩下的最大偶数。

否则，我们把目光放在奇数集合里（包括偶数操作后新增的奇数），再维护一棵值域主席树，在主席树上二分可以找出操作需要到奇数， $max(下一个的奇数,这个奇数操作后变成的偶数)$。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=1e4+5,M=1e9;
int n,q,x,y,kk,z,ta,e[N],b[N],o[N],mx,cnt,rt[N],cnt2,rt2[N];
struct node{int ls,rs,v,sz;}s[N<<10];
struct node2{int ls,rs,sz;}s2[N<<10];
struct re{int p,sz,v;}t;
void add(int &k,int l,int r){
	s[++cnt]=s[k],k=cnt;
	if(l==r){s[k].v+=z,++s[k].sz;return ;}
	int mi=(l+r)>>1;
	if(y<=mi) add(s[k].ls,l,mi);
	else add(s[k].rs,mi+1,r);
	s[k].v=s[s[k].ls].v+s[s[k].rs].v;
}
void add2(int &k,int l,int r){
	s2[++cnt2]=s2[k],k=cnt2;
	if(l==r){++s2[k].sz;return ;}
	int mi=(l+r)>>1;
	if(x<=mi) add2(s2[k].ls,l,mi);
	else add2(s2[k].rs,mi+1,r);
	s2[k].sz=s2[s2[k].ls].sz+s2[s2[k].rs].sz;
}
re find(int k1,int k2,int l,int r){
	if(l==r) return (re){l,s[k1].sz-s[k2].sz,s[k1].v-s[k2].v};
	int mi=(l+r)>>1;
	if(s[s[k1].rs].v-s[s[k2].rs].v>=kk) return find(s[k1].rs,s[k2].rs,mi+1,r);
	return kk-=s[s[k1].rs].v-s[s[k2].rs].v,find(s[k1].ls,s[k2].ls,l,mi);
}

			if(x) while(!(x&1)) ++z,x>>=1;
			else z=1;
			add(rt[i],0,M);
		}
	}
	for(int i=1;i<=q;++i){
		scanf("%d%d%d",&x,&y,&kk);

	}
	return 0;
}
```

## [爆炸（boom）](http://xsy.gdgzez.com.cn/JudgeOnline/problem.php?cid=1997&pid=1)
![[Pasted image 20230322090614.png]]
![[Pasted image 20230322090627.png]]

如果一个炸弹连通块里有环，那么整个连通块内的水晶都可以引爆。否则，需要放弃有炸弹一行或一列的水晶。

由于炸弹的攻击范围为一整行一整列，可以把每行每列变成一个点进行并查集。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=3005,M=6005;
int n,m,k,b,in[M],fx,fy,f[M],cir[M],s[M],del[M],ans;
char a[N][N];
int find(int x){return x==f[x]?x:f[x]=find(f[x]);}
int main(){
	scanf("%d%d%d%d",&n,&m,&k,&b);
	for(int i=1;i<=n+m;++i) f[i]=i;
	for(int i=1;i<=n;++i)for(int j=1;j<=m;++j){
		scanf(" %c",&a[i][j]);
		if(a[i][j]=='b'){
			++in[i],++in[j+n],fx=find(i),fy=find(j+n);
			if(fx!=fy) cir[fx]|=cir[fy],f[fy]=fx;
			else cir[fx]=1;
		}
	}
	for(int i=1;i<=n;++i)for(int j=1;j<=m;++j)
		if(a[i][j]=='k'){
			fx=find(i),fy=find(j+n),++s[fx];
			if(fx!=fy){
				++s[fy];
				if(!cir[fx]&&in[i]==1) ++del[i];
				if(!cir[fy]&&in[j+n]==1) ++del[j+n];
			}
		}
	for(int i=1;i<=n+m;++i){
		if(in[i]==1&&!cir[find(i)]) ans=max(ans,s[find(i)]-del[i]);
		if(f[i]==i&&cir[i]) ans=max(ans,s[i]);
	}
	printf("%d",ans);
	return 0;
}
```

## [炸弹（bomb）](http://xsy.gdgzez.com.cn/JudgeOnline/problem.php?cid=1998&pid=0)
![[Pasted image 20230322093819.png]]
![[Pasted image 20230322093832.png]]

**题解：**
![[Pasted image 20230322094315.png]]
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=55,M=2505,p=1e9+7;
#define ll long long
int n,d,a[N],sum,mu=1,inv,c[M],f[N][N][M],ans;
int poww(int x,int y){
	int re=1;
	while(y){if(y&1)re=1ll*re*x%p; x=1ll*x*x%p,y>>=1;}
	return re;
}
int main(){
	scanf("%d%d",&d,&n);
	for(int i=1;i<=n;++i) scanf("%d",&a[i]),sum=max(a[i],sum),mu=1ll*mu*i%p;
	inv=poww(mu,p-2),sum=sum*(n-1);
	for(int i=n-1;i<=sum;++i){
		c[i]=1;
		for(int j=0;j<n;++j) c[i]=1ll*c[i]*(d-i-j)%p;
	}
	sort(a+1,a+1+n);f[0][0][0]=1;
	for(int i=1;i<=n;++i)
		for(int j=1;j<=n;++j)
			for(int k=0;k<=sum;++k){
				f[i][j][k]=(f[i][j][k]+f[i-1][j-1][k])%p;
				if(k>=a[i]) f[i][j][k]=(f[i][j][k]+1ll*f[i-1][j][k-a[i]]*2*j%p)%p;
				if(k>=2*a[i]) f[i][j][k]=(f[i][j][k]+1ll*f[i-1][j+1][k-2*a[i]]*j*(j+1)%p)%p;
			}
	for(int k=0;k<=sum;++k) ans=(ans+1ll*f[n][1][k]*c[k]%p*inv%p)%p;
	printf("%d",ans);
	return 0;
}
```

## [咒语（spell）](http://xsy.gdgzez.com.cn/JudgeOnline/problem.php?cid=1998&pid=2)
**题意：**

字符集为 $S$，长度为 $n$ 的字符串中有多少种字符串含有长度大于 $1$ 的前缀回文串。

**题解：**
![[Pasted image 20230322095638.png]]
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=1e7+5;
int n,s,p,mi,po[N],f[N],sum,ans;
int main(){
	scanf("%d%d%d",&n,&s,&p);po[0]=f[0]=1;

	for(int i=2;i<=n;++i){
```

## [货物运输（transfer）](http://xsy.gdgzez.com.cn/JudgeOnline/problem.php?cid=1999&pid=1)
![[Pasted image 20230322100659.png]]
![[Pasted image 20230322100720.png]]

对于树：从下往上 DP

对于环：
![[Pasted image 20230322100924.png]]

由于“任意一条道路最多在一个简单环内”，可以用把每个环先处理后缩成点（圆方树），再在树（森林）上 DP。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=1e5+5,M=12e5;
#define ll long long
int n,m,x,a[N],sum,tim,dfn[N],low[N],cr[N],t,sw,ave,s[N],w[N];
ll ans;
struct edge{int v,d;}g;
vector<edge>e[N];
priority_queue<pair<int,int>>q;
void tarjan(int u,int fa){
	dfn[u]=low[u]=++tim;
	int len=e[u].size();
	for(int i=0;i<len;++i){
		int v=e[u][i].v;
		if(v==fa) continue;
		if(!dfn[v]){
			tarjan(v,u);
			if(low[v]==dfn[v]) a[u]+=a[v],ans+=1ll*e[u][i].d*abs(a[v]);//v不在u可能在的环中 
			if(low[v]<dfn[u]) low[u]=low[v],cr[u]=i;//v在u所在的环中  
			if(low[v]==dfn[u]){//"不存在一条边出现在多个简单环中",u为此环的根 
				while(!q.empty()) q.pop();
				t=sw=0;
				while(v!=u){
					++t,s[t]=s[t-1]+a[v],w[t]=e[v][cr[v]].d,sw+=w[t];
					q.push(make_pair(s[t],w[t])),v=e[v][cr[v]].v;
				}
				a[u]+=s[t],s[++t]=0,w[t]=e[u][i].d,sw+=w[t];
				q.push(make_pair(0,w[t])),sw>>=1;
				while(!q.empty()){
					sw-=q.top().second;
					if(sw<0){ave=q.top().first;break;}
					q.pop();
				}
				for(int j=1;j<=t;++j) ans+=1ll*abs(ave-s[j])*w[j];
			}
		}
		else if(dfn[v]<low[u]) low[u]=dfn[v],cr[u]=i;//v在u所在的环中 
	}
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<=n;++i) scanf("%d",&a[i]),sum+=a[i],cr[i]=-1;
	sum/=n;
	for(int i=1;i<=n;++i) a[i]-=sum;
	for(int i=1;i<=m;++i)
		scanf("%d%d%d",&x,&g.v,&g.d),e[x].push_back(g),swap(g.v,x),e[x].push_back(g);
	tarjan(1,1);
	printf("%lld\n",ans);
	return 0;
}
```

## [P7294 [USACO21JAN] Minimum Cost Paths P](https://www.luogu.com.cn/problem/P7294)
![[Pasted image 20230320153636.png]]

**题解：**

当 $c_i$ 递减时，最优方案显然是先一直往右走，再一直往下走。

进一步思考，可以发现最优走法是阶梯状的，只需维护出折线，每次询问二分找到上一个拐点即可 $O(1)$ 算出答案。
![[Pasted image 20230320154926.png]]

把询问按 $y$ 递增离线。

记 $d_i$ 表示第 $i$ 列从第 $d_i$ 行开始一直往下走最优，即拐点为 $(d_i,i)$。

新增一列 $i$ 时：

若 $c_i$ 优于 $c_{st[t]}$，则有 $(i−st[t])*d_{st[t]}^2+c_i*D<(i−st[t])*(d_{st[t]}+D)^2+c_{st[t]}*D$，化简得 $\frac{c_−c_{st[t]}}{i-st[t]}<2*d_{st[t]}+D$。

那么当 $\frac{c_−c_{st[t]}}{i-st[t]}<2*d_{st[t]}$ 时，删除 $c_{st[t]}$。

接下来要求出新的拐点，找到第一个 $d_i$ 使得紫色部分小于蓝色部分（当 $x>d_i$ 时，蓝色‘ - 紫色’ 显然比 蓝色 - 紫色更大）。
$$d_i=\lceil{\frac{\frac{c_i-c_{st[t]}}{i-st[t]}-1}2}\rceil$$

写得不错的题解：[click](https://www.cnblogs.com/AmanoKumiko/p/15163688.html)

```cpp
		else r=mi;
	}
```

```cpp
		if(o&2) adde(i<<1|1,en,p);
		else adde(0,i<<1|1,p);
	}
	for(int i=1;i<=n;++i)for(int j=1;j<=n;++j){
		if(x[i]==x[j]||y[i]==y[j]||(x[i]-x[j])*(x[i]-x[j])+(y[i]-y[j])*(y[i]-y[j])>r) continue;
		adde(i<<1,j<<1,g<<1),adde(i<<1|1,j<<1|1,g<<1),ans+=g;
	}
	dinic();
	printf("%d\n",ans);
	return 0;
}
```

## JOI2023D1T3 Passport
![[Pasted image 20230321203601.png]]

发现边是点连上一个区间，考虑线段树优化建图。

我们希望求出每个点到达 $1$ 和 $n$ 所需的最小代价，不妨建反图以 $1$ 和 $n$ 为原点分别跑一遍最短路，记距离分别为 $f_i,g_i$。

但是，答案并不一定等于 $f_i+g_i$，直接相加会出现问题：假设 $u=4$，先从 $3$ 出发，经过 $2$ 到达了 $1$ ；又从 $3$ 出发，经过 $2$ 到达了 $4$ ，而 $2$ 经过了两次，在统计答案时会被重复计算。

若节点 $u$ 的两条最短路均经过它的邻居 $v$ ，不难发现 $dis_u​=dis_v​+1$。因此，把每个点入队再跑一次最短路进行松弛操作即可。

此题加强版：[P7984 [USACO21DEC] Tickets P](https://www.luogu.com.cn/problem/P7984)，将边权从 $1$ 变成 $p$。

```cpp
	while(ch<'0'||ch>'9') ch=getchar();
	while('0'<=ch&&ch<='9') x=x*10+ch-'0',ch=getchar();

}
const int X=2e5+5,N=4e5+5,M=4e6;
int n,l[X],r[X],Q,x,L,R,rt,nd,ls[N],rs[N],cnt,h[N],nx[M],v[M],w[M],f[N],g[N],dp[N],mn=N,id;

	if(l==r){k=l;return ;}
	k=++nd;int mi=(l+r)>>1;
	build(ls[k],l,mi),build(rs[k],mi+1,r);
	add(ls[k],k,0),add(rs[k],k,0);
}
void link(int k,int l,int r){
	if(L<=l&&r<=R){add(k,x,1);return ;}
	int mi=(l+r)>>1;
	if(L<=mi) link(ls[k],l,mi);
	if(R>mi) link(rs[k],mi+1,r);
}

	int p,dis;
	bool operator <(const node &b)const{return dis>b.dis;}
}u;

	while(!q.empty()){
		u=q.top(),q.pop();
		if(u.dis!=d[u.p]) continue;
		for(int i=h[u.p];i;i=nx[i])if(d[v[i]]>u.dis+w[i])
			d[v[i]]=u.dis+w[i],q.push((node){v[i],d[v[i]]});
	}
}
int main(){

	build(rt,1,n);
	for(int i=1;i<=n;++i)
		l[i]=read(),r[i]=read(),x=i,L=l[i],R=r[i],link(rt,1,n);
	for(int i=1;i<=nd;++i) f[i]=g[i]=dp[i]=N;

	for(int i=1;i<=n;++i) dp[i]=f[i]+g[i]-(f[i]&&g[i]),q.push((node){i,dp[i]});
	dij(dp);
	Q=read();
	for(int i=1;i<=Q;++i) x=read(),printf("%d\n",(dp[x]>n?-1:dp[x]));
	return 0;
}
```

```cpp
		if(b&1) e=1ll*e*a%p;
		a=1ll*a*a%p,b>>=1;
	}
	return e;
}
```

## [P9067 [Ynoi Easy Round 2022] 虚空处刑](https://www.luogu.com.cn/problem/P9067)

```cpp
#include<bits/stdc++.h>
using namespace std;
#define pb push_back
const int N=1e6+5;
int n,m,p[N],ty,x,y,a[N],f[N],sz[N];
vector<int>e[N];
int F(int x){return f[x]==x?x:f[x]=F(f[x]);}
map<int,list<int>>s[N];
list<int>t;
void link(int a,int b){sz[F(a)]+=sz[F(b)],f[F(b)]=F(a);}
void merge(int a,int b){
	if(s[a].size()<s[b].size()) swap(s[a],s[b]);
	for(auto i:s[b]) s[a][i.first].splice(s[a][i.first].end(),i.second);
	s[b].clear();
}
void dfs(int u){
	for(int v:e[u]){
		if(a[u]==a[v]) link(u,v); 
		else s[F(u)][a[v]].pb(v);
		dfs(v);
	}
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=2;i<=n;++i) scanf("%d",&p[i]),e[p[i]].pb(i);
	for(int i=1;i<=n;++i) scanf("%d",&a[i]); 
	for(int i=1;i<=n;++i) f[i]=i,sz[i]=1;
	dfs(1);
	for(int i=1;i<=m;++i){
		scanf("%d%d",&ty,&x),x=F(x);
		if(ty==2) printf("%d\n",sz[x]);
		else{
			scanf("%d",&y);
			if(a[x]==y) continue;
			a[x]=y,t=s[x][y];
			for(auto j:t)if(x!=F(j)&&a[j]==y) link(x,j),merge(x,j);
			if(a[F(p[x])]==y) link(F(p[x]),x),merge(F(p[x]),x);
			else if(p[x]) s[F(p[x])][y].pb(x);
			s[F(x)][y].clear();
		}
	}
	return 0;
}
```
