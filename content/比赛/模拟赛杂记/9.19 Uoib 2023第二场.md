## [P1711  造飞机](http://ac.hfoj.net/p/1711)
[[珂朵莉树]]裸题。

```cpp
#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define si set<odt>::iterator
const int N=2e5+5;
struct odt{
	int l,r;ll t;
	bool operator <(const odt &b)const{return l<b.l;}
}tt;
set<odt>s;
si il,ir,it;
int n,m,q,l[N],r[N];

si split(int x){
	it=s.lower_bound((odt){x,0,0});
	if((*it).l==x) return it;
	--it,tt=*it,s.erase(it),s.insert((odt){tt.l,x-1,tt.t});
	return s.insert((odt){x,tt.r,tt.t?tt.t+x-tt.l:0}).first;
}
void add(int k,int v){
	if(k>q) return;

}

			il=s.erase(il);

	}
	for(int i=1;i<=q;++i) printf("%lld ",ask(lower_bound(c+1,c+1+q,a[i])-c));
```

## [P1712  开飞机](http://ac.hfoj.net/p/1712)
**题意：**

给定 $n$ 个点 $m$ 条边的无向图，每个点有限高 $a_i$（在 $i$ 点的高度要大于 $a_i$）,每单位时间可以改变一单位高度和飞一单位距离，起点 $1$ 终点 $n$ 限高为 $0$，求从起点出发并**降落**到终点的最短时间。

**题解：**

赛时只想到最高点那一段平飞，从起点到第一个最高点尽量飞高（移动时到顺便升高），最后一个最高点到终点尽量降落，通过枚举飞机最高高度可以做到 $n^2\log n$。

其实，第一个最高点和最后一个最高点之间可以转化为先飞高再降落，这样只有两种情况，一种是只有一个最高点，花费为 $2*\max(d_{1,i},d_{n,i})$；另一种是有两个相邻的最高点，中间平飞 $1$，花费为 $d_{1,u}+d_{n,v}+1$。

其中 $d$ 为从 $1/n$ 出发到 $i$ 最短花费及最短花费下的最大高度，$d_v=\max\{a_v,d_u+1\}$。

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=2e5+5;
int n,m,a[N],x,y,d1[N],d2[N],ans=1e9;
vector<int>e[N];
struct qwq{int x,d;}u;
struct cmp{bool operator ()(qwq a,qwq b){return a.d>b.d;}};
priority_queue<qwq,vector<qwq>,cmp>q;
void dij(int st,int d[]){
	for(int i=1;i<=n;++i) d[i]=1e9;
	d[st]=0,q.push((qwq){st,0});
	while(!q.empty()){
		u=q.top(),q.pop();
		if(u.d!=d[u.x]) continue;
		for(int v:e[u.x])if(d[v]>max(a[v],u.d+1))
			d[v]=max(a[v],u.d+1),q.push((qwq){v,d[v]});
	}
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<=n;++i) scanf("%d",&a[i]);
	for(int i=1;i<=m;++i) scanf("%d%d",&x,&y),e[x].push_back(y),e[y].push_back(x);
	dij(1,d1),dij(n,d2);
	for(int i=1;i<=n;++i) ans=min(ans,2*max(d1[i],d2[i]));
	for(int u=1;u<=n;++u)for(int v:e[u])if(d1[u]==d2[v]) ans=min(ans,d1[u]+d2[v]+1);
	printf("%d",ans);
	return 0;
}
```

## [P1713  隐身飞机](http://ac.hfoj.net/p/1713)

^4c2ce2

**题意：**

给定 $m$ 个范围 $[1,n]$ 的区间 $[l,r]$，有 $q$ 次询问，问 $[L,R]$ 是否能被恰好覆盖。

**题解：**

记录每个点覆盖其线段的左端点的最大值（没有则为 $0$），把询问和线段按 $r$ 从左到右扫描，每次更新线段 $[l,r]$ 的值（取 $\max$），判断 $[L,R]$ 的最大值是否都大于 $L$ 即可。

需要进行区间最值操作，可用[[吉司机线段树]]。

```cpp
#include<bits/stdc++.h>
using namespace std;
#define pii pair<int,int>
#define ls k<<1
#define rs k<<1|1
const int N=5e5+5,M=2e6+5;
int n,m,q,x,y,s[M],mn[M],se[M],tg[M],ans[N];
vector<int>a[N];
vector<pii>b[N];
void build(int k,int l,int r){
	se[k]=n+1;
	if(l==r) return;
	int mi=(l+r)>>1;build(ls,l,mi),build(rs,mi+1,r);
}
void ptg(int k,int x){if(mn[k]<x) mn[k]=tg[k]=x;}
void pd(int k){if(tg[k]) ptg(ls,tg[k]),ptg(rs,tg[k]),tg[k]=0;}
void pu(int k){
	if(mn[ls]==mn[rs]) mn[k]=mn[ls],se[k]=min(se[ls],se[rs]);
	else if(mn[ls]<mn[rs]) mn[k]=mn[ls],se[k]=min(se[ls],mn[rs]);
	else mn[k]=mn[rs],se[k]=min(se[rs],mn[ls]);
}
void upd(int k,int l,int r,int x,int y){
	if(mn[k]>=x) return;
	if(x<=l&&r<=y&&se[k]>x){ptg(k,x);return;} 
	int mi=(l+r)>>1;pd(k);
	if(x<=mi) upd(ls,l,mi,x,y);
	if(y>mi) upd(rs,mi+1,r,x,y);
	pu(k);
}
int ask(int k,int l,int r,int x,int y){
	if(x<=l&&r<=y) return mn[k];
	int mi=(l+r)>>1,MN=n;pd(k);
	if(x<=mi) MN=min(MN,ask(ls,l,mi,x,y));
	if(y>mi) MN=min(MN,ask(rs,mi+1,r,x,y));
	return MN;
}
int main(){
	scanf("%d%d%d",&n,&m,&q);
	for(int i=1;i<=m;++i) scanf("%d%d",&x,&y),a[y].push_back(x);
	for(int i=1;i<=q;++i) scanf("%d%d",&x,&y),b[y].push_back(make_pair(x,i));
	build(1,1,n);
	for(int i=1;i<=n;++i){
		for(int j:a[i]) upd(1,1,n,j,i);
		for(pii j:b[i]) ans[j.second]=(ask(1,1,n,j.first,i)>=j.first);
	}
	for(int i=1;i<=q;++i) printf((ans[i]?"YES\n":"NO\n"));
	return 0;
}
```
