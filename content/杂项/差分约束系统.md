## 原理
![[Pasted image 20230922151828.png]]

$$V\leq U+dis：U到V连边权dis的边，跑最短路$$

$$V\geq U+dis：U到V连边权dis的边，跑最长路$$

$$总结：什么符号跑什么路（与跑SPFA松弛操作的if里符号相反），后面往前面连边。$$

## 注意
### (1). 解多元一次方程类
注意和“0”连边保证联通。（若要求正解得跑最长路）
### (2). 转换前缀和类
注意和前一位连边保证前缀和的不降性。
$$ s[k]-s[k-1]\geq0 $$
### (3). 关注隐藏的约束条件
如[#10087. 「一本通 3.4 例 1」Intervals](https://loj.ac/p/10087)中的
$$ s[k]-s[k-1]\leq1 $$
即一个数只能选一次。

## 例题
### [P5960 【模板】差分约束算法](https://www.luogu.com.cn/problem/P5960)
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=5005,inf=1e9+5;
int n,m,u,v,w,t,dis[N],b[N],tot[N];
struct qwq{int u,d;};
vector<qwq>vec[N];
queue<int>q;
bool spfa(){
	for(int i=1;i<=n;++i) vec[0].push_back((qwq){i,0}),dis[i]=inf;
	b[0]=1,q.push(0);
	while(!q.empty()){
		t=q.front(),q.pop(),b[t]=0;
		for(qwq i:vec[t])if(dis[i.u]>dis[t]+i.d){
			dis[i.u]=dis[t]+i.d;
			if(!b[i.u]){
				++tot[i.u],b[i.u]=1,q.push(i.u);
				if(tot[i.u]>n) return 1;
			}
		}
	}
	return 0;
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<=m;++i) scanf("%d%d%d",&u,&v,&w),vec[v].push_back((qwq){u,w});
	if(spfa()) printf("NO");
	else for(int i=1;i<=n;++i) printf("%d ",dis[i]);
	return 0;
}
```

### [#10088. 「一本通 3.4 例 2」出纳员问题](https://loj.ac/p/10088)
设 $num[i]$ 表示第 $i$ 时刻可以雇佣 $num[i]$ 人，$r[i]$ 为第i时刻至少需要 $r[i]$ 个人。
设 $s[i]$ 为前缀和，可得：
$$ s[i]-s[i-1]\geq0~(2) $$

$$ s[i-1]-s[i]\geq-num[i] $$

$$ \begin{cases}s[i]-s[i-8]\geq r[i] ~ ~ ~, ~ ~ ~ i>=8 \\  s[i]-s[i+16]\geq r[i]-s[23] ~ ~ ~, ~ ~ ~ i<8 ~ ~* \end{cases}$$

#### 如何破环为链
关注到 " * "中多出现了 $s[23]$ ，考虑枚举 $s[23]$ 的合法值 $m$ 验证。

**注意隐藏条件**：

$$ s[24]-s[0]\leq m $$

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=25,inf=1e9,M=1005;
int T,n,x,z,dis[N],len,r[N],num[M],sum[N];
bool vis[N],fl;
struct edge{
	int d,to;
}t;
vector<edge>vec[N];
queue<int>q;
bool spfa(int m){
	for(int i=1;i<=7;++i){
		if(m) vec[i+16].pop_back();
		t.to=i,t.d=r[i]-m,vec[i+16].push_back(t);
	}
	if(m) vec[0].pop_back();
	t.to=24,t.d=m,vec[0].push_back(t);
	memset(sum,0,sizeof(sum));
	memset(dis,128,sizeof(dis));
	memset(vis,0,sizeof(vis));
	while(!q.empty()) q.pop();
	dis[0]=0,vis[0]=1,q.push(0);
	while(!q.empty()){
		x=q.front(),q.pop(),vis[x]=0;
		len=vec[x].size();
		for(int i=0;i<len;++i){
			t=vec[x][i];
			if(dis[t.to]<dis[x]+t.d){
				dis[t.to]=dis[x]+t.d,++sum[t.to];
				if(sum[t.to]>24) return 0;
				if(!vis[t.to]) vis[t.to]=1,q.push(t.to);
			}
		}
	}
	return 1;
}
int main(){
	scanf("%d",&T);
	while(T--){
		fl=0;
		vector<edge>().swap(vec[0]);
		for(int i=1;i<=24;++i) scanf("%d",&r[i]),vector<edge>().swap(vec[i]),num[i]=0;
		scanf("%d",&n);
		for(int i=1;i<=n;++i) scanf("%d",&z),++z,++num[z];
		for(int i=1;i<=24;++i){
			if(i>=8) t.d=r[i],t.to=i,vec[i-8].push_back(t);
			t.d=0,t.to=i,vec[i-1].push_back(t);
			t.d=-num[i],t.to=i-1,vec[i].push_back(t);
		}
		for(int i=0;i<=n;++i)
			if(spfa(i)&&dis[24]==i){
				printf("%d\n",i),fl=1;
				break;
			}
		if(!fl) printf("No Solution\n");
	}
	return 0;
}
```
