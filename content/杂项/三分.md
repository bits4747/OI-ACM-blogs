## 用途
**多用于当~~工具人~~，用来求解（一定定义域里）单峰函数的最值。**
## 例题
### [#10013. 「一本通 1.2 例 3」曲线](https://loj.ac/p/10013)
```cpp
#include<bits/stdc++.h>
using namespace std;
int t,n;
double l,r,lm,rm,e=1e-11,ans;//注意：eps精度大多数情况下远大于题目精度
//原因: 题目求f(l)，而三分的是l！！！！！ 
struct line{
	double a,b,c;
}s[100005];
double f(double x){
	ans=s[1].a*x*x+s[1].b*x+s[1].c;
	for(int i=2;i<=n;++i) ans=max(ans,s[i].a*x*x+s[i].b*x+s[i].c);
	return ans;
}
int main(){
	scanf("%d",&t);
	while(t--){
		scanf("%d",&n),l=0,r=1001;
		for(int i=1;i<=n;++i) scanf("%lf%lf%lf",&s[i].a,&s[i].b,&s[i].c);
		while(l+e<r){
			lm=l+(r-l)/3,rm=r-(r-l)/3;
			if(f(lm)<f(rm)) r=rm;
			else l=lm;
		}
		printf("%.4lf\n",f(l));
	}
	return 0;
}
```
